 46    def _model_x(
 47        self,
 48        N,
 49        G,
 50        guides_tensor,
 51        x_obs_tensor,
 52        sum_factor_tensor,
 53        mu_x_mean_tensor,
 54        mu_x_sd_tensor,
 55        sigma_eff_mean_tensor,
 56        sigma_eff_sd_tensor,
 57        C=None,
 58        groups_tensor=None,
 59        alpha_x_sample=None,
 60        target_per_guide_tensor=None,
 61        independent_mu_sigma=False,
 62        phi_x_precomputed=None,
 63        rate_alpha_precomputed=None,
 64        rate_beta_precomputed=None,
 65        scale_2_tensor=None,
 66    ):
 67
 68        ###################################
 69        ## Technical covariate modelling ##
 70        ###################################
 71        if alpha_x_sample is not None:
 72            alpha_x = alpha_x_sample
 73        else:
 74            alpha_x = None
 75        
 76        ####################
 77        ## Overdispersion ##
 78        ####################
 79        if phi_x_precomputed is None:
 80            raise ValueError(
 81                "phi_x_precomputed is required. "
 82                "Run fit_ntc() before fit_cis() to estimate cis gene overdispersion from NTC cells."
 83            )
 84        phi_x_used = phi_x_precomputed
 85
 86        ###############################
 87        ## Mixture model for x_eff_g ##
 88        ###############################
 89        if independent_mu_sigma:
 90            unique_targets = torch.unique(target_per_guide_tensor)
 91            mu_targets = {}
 92            sigma_targets = {}
 93            _hc_scale = scale_2_tensor if scale_2_tensor is not None else self._t(2.0)
 94            for t in unique_targets:
 95                mu_targets[int(t.item())] = pyro.sample(f"mu_target_{int(t.item())}", dist.Normal(mu_x_mean_tensor, mu_x_sd_tensor))
 96                sigma_targets[int(t.item())] = pyro.sample(f"sigma_target_{int(t.item())}", dist.HalfCauchy(scale=_hc_scale))
 97            # gather mu and sigma for each guide
 98            mu_target_tensor = torch.stack([mu_targets[int(t.item())] for t in unique_targets], dim=0)
 99            sigma_target_tensor = torch.stack([sigma_targets[int(t.item())] for t in unique_targets], dim=0)
100            mu = mu_target_tensor[target_per_guide_tensor]
101            sigma = sigma_target_tensor[target_per_guide_tensor]
102        else:
103            mu = pyro.sample("mu", dist.Normal(mu_x_mean_tensor, mu_x_sd_tensor))
104            _hc_scale = scale_2_tensor if scale_2_tensor is not None else self._t(2.0)
105            sigma = pyro.sample("sigma", dist.HalfCauchy(scale=_hc_scale))
106            mu = mu.expand(G)
107            sigma = sigma.expand(G)
108
109        # Non-centered parameterization
110        if rate_alpha_precomputed is not None:
111            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Exponential(rate_alpha_precomputed))
112            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Exponential(rate_beta_precomputed))
113        elif (sigma_eff_mean_tensor >= 0.01) and (sigma_eff_sd_tensor >= 0.01):
114            rate_alpha = (sigma_eff_sd_tensor ** 2) / (sigma_eff_mean_tensor ** 2)
115            rate_beta = (sigma_eff_sd_tensor ** 2) / sigma_eff_mean_tensor
116            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Exponential(rate_alpha))
117            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Exponential(rate_beta))
118        else:
119            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Gamma(self._t(1.0), self._t(0.01)))
120            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Gamma(self._t(1.0), self._t(0.01)))
121        with pyro.plate("guides_plate", G):
122            eps_x_eff_g = pyro.sample("eps_x_eff_g", dist.StudentT(df=self._t(3.0), loc=self._t(0.0), scale=self._t(1.0)))
123            log2_x_eff_g = mu + sigma * eps_x_eff_g
124            x_eff_g = pyro.deterministic("x_eff_g", 2.0 ** log2_x_eff_g)
125        
126            sigma_eff = pyro.sample("sigma_eff", dist.Gamma(sigma_eff_alpha, sigma_eff_beta))
127                
128        ##########################
129        ## Cell-level variables ##
130        ##########################
131        if alpha_x is not None:
132            # alpha_x shape [C]: already includes reference group (index 0 = 1.0)
133            # or shape [C-1]: needs reference group prepended
134            if alpha_x.shape[-1] == C:
135                alpha_x_full = alpha_x
136            else:
137                ones_ = torch.ones(1, device=self.model.device)
138                alpha_x_full = torch.cat([ones_, alpha_x], dim=-1)  # shape = [C]
139            alpha_x_used = alpha_x_full[groups_tensor]  # shape = [N]
140        else:
141            alpha_x_used = torch.ones_like(sum_factor_tensor)  # shape = [N] or [S,N]
142
143        # Aggregate guide effects to cells
144        if self.model.is_high_moi:
145            # High MOI: effects are additive in log2FC space.
146            # log2FC for guide g = log2(x_eff_g[g]) - log2(NTC_mean)
147            # For a cell with guides A, B: log2FC_cell = log2FC_A + log2FC_B
148            # => log2(x_mean) = log2(NTC) + log2FC_cell
149            #                 = sum(log2(x_eff_g)) - (n_guides - 1) * log2(NTC)
150            # This is always finite since x_eff_g > 0 and NTC_mean > 0.
151
152            # NTC reference: weighted mean of NTC guide effects
153            # Use pre-computed mask (set in fit_cis before SVI loop to avoid pandas access here)
154            ntc_mask = self.model._ntc_guide_mask
155            cells_per_guide = self.model.guide_assignment_tensor.sum(dim=0)  # [G]
156            weights = cells_per_guide / sigma_eff.clamp(min=1e-6)            # [G]
157            ntc_weights = weights[ntc_mask]
158            ntc_effects  = x_eff_g[ntc_mask]
159            weighted_mean_NTC = (ntc_weights * ntc_effects).sum(dim=-1) / ntc_weights.sum()
160            weighted_mean_NTC = pyro.deterministic("weighted_mean_NTC", weighted_mean_NTC)
161
162            log2_x_eff_g  = torch.log2(x_eff_g.clamp(min=1e-12))                               # [G]
163            log2_NTC      = torch.log2(weighted_mean_NTC.clamp(min=1e-12))                       # scalar
164            guides_per_cell = self.model.guide_assignment_tensor.sum(dim=1).clamp(min=1)         # [N]
165            sum_log2_effects = torch.matmul(self.model.guide_assignment_tensor, log2_x_eff_g)    # [N]
166            log2_x_mean   = sum_log2_effects - (guides_per_cell - 1) * log2_NTC                 # [N]
167            x_mean        = 2.0 ** log2_x_mean                                                   # [N]
168            sigma_mean    = torch.matmul(self.model.guide_assignment_tensor, sigma_eff) / guides_per_cell  # [N]
169        else:
170            # Single guide per cell: use indexing (unchanged)
171            x_mean = x_eff_g[..., guides_tensor]  # [N]
172            sigma_mean = sigma_eff[..., guides_tensor]  # [N]
173
174        ######################
175        ## Cell-level plate ##
176        ######################
177        with pyro.plate("data_plate", N):
178            log_x_true = pyro.sample( # use log2 of xtrue to allow small values of xtrue
179                "log_x_true",
180                dist.Normal(torch.log2(x_mean), sigma_mean)
181            )
182            x_true = pyro.deterministic("x_true", 2.0 ** log_x_true)
183            mu_obs = alpha_x_used * x_true * sum_factor_tensor
184            pyro.sample(
185                "x_obs",
186                dist.NegativeBinomial(total_count=phi_x_used,
187                                      logits=torch.log(mu_obs) - torch.log(phi_x_used),
188                                      validate_args=False
189                                     ), # important to use logits to allow small x_true
190                obs=x_obs_tensor
191            )

193    def fit_cis(
194        self,
195        technical_covariates: list[str] = None,
196        sum_factor_col: str = 'sum_factor',
197        cis_feature: str = None,
198        manual_guide_effects: pd.DataFrame = None,
199        prior_strength: float = 1.0,
200        lr: float = 1e-3,
201        niters: int = 100_000,
202        nsamples: int = 1000,
203        alpha_ewma: float = 0.05,
204        epsilon: float = 1e-6,
205        minibatch_size: int = None,
206        predictive_on_cpu: bool = True,
207        independent_mu_sigma: bool = False,
208        **kwargs
209    ):
210        """
211        Fits the cis effects (model_x) for your gene_of_interest.
212        This step can be repeated multiple times with different priors
213        or hyperparameters.
214
215        Parameters
216        ----------
217        technical_covariates : list, optional
218            Technical covariates for correction
219        sum_factor_col : str
220            Column name for size factors
221        cis_feature : str, optional
222            Feature ID to use as cis proxy from the primary modality.
223            If None, uses self.model.cis_gene (must exist in primary modality).
224            For ATAC: region ID (e.g., 'chr9:132283881-132284881')
225            For genes: gene name
226        manual_guide_effects : pd.DataFrame, optional
227            Manual guide effect estimates as priors. DataFrame with columns:
228            - guide: guide identifier (matches meta['guide'])
229            - log2FC: expected log2 fold-change vs NTC
230        prior_strength : float
231            Weight for manual guide effects (default: 1.0)
232            0 = ignore manual effects, higher = trust more
233        lr : float
234            Learning rate for Adam
235        niters : int
236            Number of SVI iterations
237        nsamples : int
238            Number of posterior samples
239        predictive_on_cpu : bool
240            If True (default), run posterior Predictive sampling on CPU after SVI
241            to reduce GPU memory pressure. Set False to keep Predictive on GPU.
242        alpha_ewma : float
243            Exponential weight for smoothing the ELBO
244        independent_mu_sigma : bool
245            Whether to use independent mu/sigma per target type
246        kwargs :
247            Additional arguments controlling priors, etc.
248        """
249        print("Running fit_cis...")
250
251        if self.model.cis_gene is None:
252            raise ValueError("self.model.cis_gene must be set.")
253
254        # fit_cis ALWAYS uses the 'cis' modality
255        # This modality is created automatically when bayesDREAM is initialized
256
257        # Get cis modality
258        if 'cis' not in self.model.modalities:
259            raise ValueError(
260                "No 'cis' modality found. The 'cis' modality should be created automatically "
261                "when bayesDREAM is initialized with a cis_gene. "
262                "Make sure cis_gene parameter is set during initialization."
263            )
264
265        cis_modality = self.model.get_modality('cis')
266
267        # Determine which feature to use as cis proxy
268        if cis_feature is None:
269            # Default: use the first (and typically only) feature in cis modality
270            cis_feature_idx = cis_modality.feature_meta.index[0]
271            # Get actual feature name from metadata (not just numeric index)
272            if 'gene_name' in cis_modality.feature_meta.columns:
273                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'gene_name']
274            elif 'gene' in cis_modality.feature_meta.columns:
275                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'gene']
276            elif 'feature' in cis_modality.feature_meta.columns:
277                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'feature']
278            else:
279                # Fallback to index if no name column found
280                cis_feature_name = cis_feature_idx
281            cis_feature = cis_feature_idx  # Use numeric index for actual data retrieval
282            print(f"[INFO] Using cis feature '{cis_feature_name}' from 'cis' modality")
283        else:
284            # User specified explicit cis_feature
285            if cis_feature not in cis_modality.feature_meta.index:
286                raise ValueError(
287                    f"cis_feature '{cis_feature}' not found in 'cis' modality.\n"
288                    f"Available features: {cis_modality.feature_meta.index.tolist()}"
289                )
290            print(f"[INFO] Using cis_feature '{cis_feature}' from 'cis' modality")
291
292        # Get counts for cis feature from cis modality
293        if isinstance(cis_modality.counts, pd.DataFrame):
294            cis_counts = cis_modality.counts.loc[cis_feature].values
295        else:
296            # numpy/sparse array - need to find index
297            feature_idx = cis_modality.feature_meta.index.get_loc(cis_feature)
298            if cis_modality.cells_axis == 1:
299                cis_counts = cis_modality.counts[feature_idx, :]
300            else:
301                cis_counts = cis_modality.counts[:, feature_idx]
302            # Densify if sparse (slicing sparse returns sparse row/col matrix)
303            if hasattr(cis_counts, 'toarray'):
304                cis_counts = cis_counts.toarray().ravel()
305
306        # convert to gpu for fitting
307        if self.model.alpha_x_prefit is not None and self.model.alpha_x_prefit.device != self.model.device:
308            self.model.alpha_x_prefit = self.model.alpha_x_prefit.to(self.model.device)
309
310        if technical_covariates:
311            new_technical_group_code = self.model.meta.groupby(technical_covariates).ngroup()
312            if "technical_group_code" in self.model.meta.columns:
313                old_technical_group_code = self.model.meta["technical_group_code"].values
314                groups_changed = not np.array_equal(old_technical_group_code, new_technical_group_code.values)
315                if groups_changed:
316                    warnings.warn("technical_group already set. Overwriting with new covariate grouping.")
317                    if self.model.alpha_x_prefit is not None:
318                        warnings.warn("Technical groups changed; discarding alpha_x_prefit and refitting.")
319                        self.model.alpha_x_prefit = None
320                else:
321                    # Keep loaded alpha_x_prefit when grouping is unchanged (common in stepwise workflows)
322                    pass
323
324            self.model.meta["technical_group_code"] = new_technical_group_code
325            C = self.model.meta['technical_group_code'].nunique()
326            groups_tensor = torch.tensor(self.model.meta['technical_group_code'].values, dtype=torch.long, device=self.model.device)
327
328            # alpha_x_prefit is required when technical_covariates are specified
329            if self.model.alpha_x_prefit is None:
330                raise ValueError(
331                    f"Technical covariates provided but alpha_x_prefit not set. "
332                    f"Run fit_ntc() on the primary modality ('{self.model.primary_modality}') before fit_cis() "
333                    f"to estimate technical effects for the cis gene."
334                )
335
336        elif self.model.alpha_x_prefit is None:
337            C = None
338            groups_tensor = None
339            warnings.warn("no alpha_x_prefit and no technical_covariates provided, assuming no confounding effect.")
340        else:
341            # alpha_x_prefit exists but no new technical_covariates specified
342            # Use existing technical groups
343            C = self.model.meta['technical_group_code'].nunique()
344            groups_tensor = torch.tensor(self.model.meta['technical_group_code'].values, dtype=torch.long, device=self.model.device)
345        
346        N = self.model.meta.shape[0]
347
348        # Handle G (number of guides) differently for high MOI vs single-guide mode
349        if self.model.is_high_moi:
350            G = self.model.guide_assignment.shape[1]  # Number of guides
351            guides_tensor = None  # Not used in high MOI mode
352        else:
353            G = self.model.meta['guide_code'].nunique()
354            guides_tensor = torch.tensor(self.model.meta['guide_code'].values, dtype=torch.long, device=self.model.device)
355
356        # Validate minimum data requirements for stable initialization
357        min_guides = 3
358        min_cells_per_guide = 5
359        if self.model.is_high_moi:
360            cells_per_guide = self.model.guide_assignment.sum(axis=0)  # [G]
361            n_adequate = int((cells_per_guide >= min_cells_per_guide).sum())
362        else:
363            guide_counts = self.model.meta['guide_code'].value_counts()
364            n_adequate = int((guide_counts >= min_cells_per_guide).sum())
365        if n_adequate < min_guides:
366            raise ValueError(
367                f"fit_cis requires at least {min_guides} guides with >= {min_cells_per_guide} cells each, "
368                f"but only {n_adequate} guides meet this criterion. "
369                f"Subsetting to too few cells/guides will produce unreliable initializations."
370            )
371
372        # Use cis_counts from modality-specific lookup (or traditional self.model.counts)
373        x_obs_tensor = torch.tensor(cis_counts, dtype=torch.float32, device=self.model.device)
374
375        # ========================================================================
376        # MANUAL GUIDE EFFECTS INFRASTRUCTURE
377        # ========================================================================
378        # If user provides manual guide effects (log2FC estimates), prepare them
379        # as tensors that can be used as priors in the Pyro model.
380        manual_guide_log2fc_tensor = None
381        manual_guide_mask_tensor = None
382
383        if manual_guide_effects is not None:
384            raise NotImplementedError(
385                "manual_guide_effects is not yet implemented. "
386                "The infrastructure for guide-level priors is planned but not yet integrated into _model_x."
387            )
388        # ========================================================================
389        if independent_mu_sigma:
390            if ('target' not in self.model.meta.columns):
391                raise ValueError("independent_mu_sigma is True, self.model.meta['target'] column not found.")
392            elif self.model.meta['target'].nunique() < 2:
393                raise ValueError("independent_mu_sigma is True, but only 1 target type found in self.model.meta['target'] column.")
394
395            ### BUILD target_per_guide_tensor [G] based on guide → target
396            if self.model.is_high_moi:
397                # High MOI: derive one target label per guide for grouping mu/sigma.
398                # Two sources: guide_meta['target'] (simple) or guide_targets_dict (many-to-many).
399                _ntc_variants = {'ntc', 'NTC', 'non-targeting', 'non-targeting-control', 'Non-Targeting'}
400
401                if 'target' in self.model.guide_meta.columns:
402                    guide_target_labels = self.model.guide_meta['target'].tolist()
403                elif hasattr(self.model, 'guide_targets_dict') and self.model.guide_targets_dict:
404                    # Derive a single representative target per guide.
405                    # Priority: cis_gene > any NTC variant > first target.
406                    def _primary_target(targets):
407                        if self.model.cis_gene and self.model.cis_gene in targets:
408                            return self.model.cis_gene
409                        for t in targets:
410                            if t in _ntc_variants:
411                                return 'ntc'
412                        return targets[0] if targets else 'ntc'
413
414                    guide_target_labels = [
415                        _primary_target(self.model.guide_targets_dict.get(row['guide'], ['ntc']))
416                        for _, row in self.model.guide_meta.iterrows()
417                    ]
418                else:
419                    raise ValueError(
420                        "independent_mu_sigma=True in high MOI mode requires either "
421                        "guide_meta['target'] or guide_targets_dict."
422                    )
423
424                target_factorized, target_unique = pd.factorize(guide_target_labels)
425                target_per_guide_tensor = torch.tensor(target_factorized, dtype=torch.long, device=self.model.device)
426                print(f"[INFO] independent_mu_sigma (high MOI): {len(target_unique)} unique targets")
427            else:
428                # Single-guide mode: one target code per guide (raises if a guide has multiple targets)
429                self.model.meta['target_code'] = pd.factorize(self.model.meta['target'])[0]
430                target_codes_tensor = torch.tensor(self.model.meta['target_code'].values, dtype=torch.long, device=self.model.device)
431
432                target_per_guide_tensor = torch.empty(G, dtype=torch.long, device=self.model.device)
433                for g in range(G):
434                    idx = (guides_tensor == g)
435                    guide_targets = torch.unique(target_codes_tensor[idx])
436                    if guide_targets.shape[0] != 1:
437                        raise ValueError(f"Guide {g} maps to multiple targets: {guide_targets}. independent_mu_sigma=True requires unambiguous target assignment per guide.")
438                    target_per_guide_tensor[g] = guide_targets[0]
439        else:
440            target_per_guide_tensor = None
441        primary_mod = self.model.get_modality(self.model.primary_modality)
442        sum_factor_tensor = torch.tensor(
443            primary_mod.sum_factors.loc[self.model.meta['cell'].values, sum_factor_col].values,
444            dtype=torch.float32, device=self.model.device
445        )
446
447        # Compute guide means (adjusting for alpha_x if provided)
448        x_obs_factored = x_obs_tensor / sum_factor_tensor
449        
450        if self.model.alpha_x_prefit is not None:
451            # alpha_x_prefit is always a [C] point estimate (mean already taken at fit_ntc time)
452            alpha_x_full = self.model.alpha_x_prefit.flatten()
453            # Select the correct alpha_x for each observation (expand for broadcasting)
454            alpha_x_used = alpha_x_full[groups_tensor]  # groups_tensor indexes into (C,)
455
456            # Adjust x_obs_factored by alpha_x_used
457            x_obs_factored /= alpha_x_used  # Ensure correct shape for division
458        
459        # --- Extract o_x from the technical fit (cis modality posterior) ---
460        cis_modality = self.model.get_modality('cis')
461        if (cis_modality.posterior_samples_ntc is None
462                or 'o_x' not in cis_modality.posterior_samples_ntc):
463            raise ValueError(
464                "Cis gene overdispersion not found. Run fit_ntc() before fit_cis().\n"
465                "fit_ntc() estimates o_x for the cis gene from NTC cells, which is required "
466                "to set a data-driven overdispersion prior instead of the generic Gamma(9,3)."
467            )
468        o_x_ntc = float(cis_modality.posterior_samples_ntc['o_x'].mean().item())
469        print(f"[INFO] fit_cis: using technical o_x = {o_x_ntc:.4f} (phi = {1/o_x_ntc**2:.2f})")
470
471        # Compute guide-level means and MADs.
472        # Uses numpy on CPU for per-guide median computation (faster than G GPU boolean masks).
473        x_np = x_obs_factored.detach().cpu().numpy().astype(np.float32)
474
475        if self.model.is_high_moi:
476            # High MOI: vectorized means via matrix-vector multiply, MADs via sorted numpy slices
477            assign_np = self.model.guide_assignment  # [N, G] numpy array
478            cell_counts_np = assign_np.sum(axis=0).clip(min=1)  # [G]
479            guide_sums_np = x_np @ assign_np  # [G]
480            guide_means_np = np.log2(np.clip(guide_sums_np / cell_counts_np, epsilon, None))
481
482            log2_x_np = np.log2(np.clip(x_np, epsilon, None))
483            guide_mads_np = np.zeros(G, dtype=np.float32)
484            for g in range(G):
485                cell_mask = assign_np[:, g].astype(bool)
486                if cell_mask.sum() > 0:
487                    vals = log2_x_np[cell_mask]
488                    med = np.median(vals)
489                    guide_mads_np[g] = float(np.median(np.abs(vals - med)))
490
491            guide_means = torch.from_numpy(guide_means_np).to(self.model.device)
492            guide_mads_tensor = torch.from_numpy(guide_mads_np).to(self.model.device)
493        else:
494            # Single-guide mode: vectorized means via bincount, MADs via sorted numpy slices
495            guides_np = guides_tensor.cpu().numpy()
496            n_per_guide_np = np.bincount(guides_np, minlength=G).clip(min=1).astype(np.float32)
497            sum_per_guide_np = np.bincount(guides_np, weights=x_np, minlength=G).astype(np.float32)
498            guide_means_np = np.log2(np.clip(sum_per_guide_np / n_per_guide_np, epsilon, None))
499
500            log2_x_np = np.log2(np.clip(x_np, epsilon, None))
501            sort_idx = np.argsort(guides_np, kind='stable')
502            guides_sorted = guides_np[sort_idx]
503            log2_x_sorted = log2_x_np[sort_idx]
504            boundaries = np.searchsorted(guides_sorted, np.arange(G + 1))
505
506            guide_mads_np = np.zeros(G, dtype=np.float32)
507            for g in range(G):
508                s, e = boundaries[g], boundaries[g + 1]
509                if e > s:
510                    vals = log2_x_sorted[s:e]
511                    med = np.median(vals)
512                    guide_mads_np[g] = float(np.median(np.abs(vals - med)))
513
514            guide_means = torch.from_numpy(guide_means_np).to(self.model.device)
515            guide_mads_tensor = torch.from_numpy(guide_mads_np).to(self.model.device)
516
517        mu_x_mean_tensor = torch.mean(guide_means)
518        mu_x_sd_tensor = torch.std(guide_means)
519
520        guide_mads_tensor = guide_mads_tensor * 1.4826  # Gaussian-equivalent spread
521        sigma_eff_mean_tensor = torch.mean(guide_mads_tensor)#.to(self.model.device)
522        sigma_eff_sd_tensor = torch.std(guide_mads_tensor)#.to(self.model.device)
523
524        def init_loc_fn(site):
525            name = site["name"]
526        
527            if name == "mu":
528                return mu_x_mean_tensor
529            elif name == "sigma":
530                return mu_x_sd_tensor.clamp(min=1e-3)
531        
532            elif name == "sigma_eff_alpha":
533                # Corresponds to (mean^2 / var)
534                sigma_eff_mean_tensor_tmp = sigma_eff_mean_tensor.clamp(min=1e-2)
535                sigma_eff_sd_tensor_tmp = sigma_eff_sd_tensor.clamp(min=1e-2)
536                return ((sigma_eff_mean_tensor_tmp ** 2) / (sigma_eff_sd_tensor_tmp ** 2))
537            elif name == "sigma_eff_beta":
538                # Corresponds to (mean / var)
539                sigma_eff_mean_tensor_tmp = sigma_eff_mean_tensor.clamp(min=1e-2)
540                sigma_eff_sd_tensor_tmp = sigma_eff_sd_tensor.clamp(min=1e-2)
541                return (sigma_eff_mean_tensor_tmp / (sigma_eff_sd_tensor_tmp ** 2))
542        
543            elif name == "sigma_eff":
544                # Assume you’re sampling per-guide (e.g. G = 37)
545                return sigma_eff_mean_tensor.clamp(min=1e-2).expand(G)
546        
547            return pyro.infer.autoguide.initialization.init_to_median(site)
548
549        # Pre-compute NTC guide mask for high MOI mode (avoids pandas access inside Pyro model)
550        if self.model.is_high_moi:
551            _ntc_variants = {'ntc', 'NTC', 'non-targeting', 'non-targeting-control', 'Non-Targeting'}
552            if 'target' in self.model.guide_meta.columns:
553                ntc_flags = [self.model.guide_meta.iloc[g]['target'] in _ntc_variants
554                             for g in range(G)]
555            elif hasattr(self.model, 'guide_targets_dict') and self.model.guide_targets_dict:
556                ntc_flags = [
557                    any(t in _ntc_variants for t in self.model.guide_targets_dict.get(row['guide'], []))
558                    for _, row in self.model.guide_meta.iterrows()
559                ]
560            else:
561                raise ValueError(
562                    "High MOI mode requires either guide_meta['target'] or guide_targets_dict "
563                    "to identify NTC guides."
564                )
565            self.model._ntc_guide_mask = torch.tensor(ntc_flags, dtype=torch.bool,
566                                                       device=self.model.device)
567
568        # Precompute model invariants once to avoid repeated computation in every SVI step
569        phi_x_precomputed = torch.tensor(1.0 / (o_x_ntc ** 2), dtype=torch.float32, device=self.model.device)
570        scale_2_tensor = self._t(2.0)
571        _seff_mean = sigma_eff_mean_tensor.clamp(min=1e-2)
572        _seff_sd = sigma_eff_sd_tensor.clamp(min=1e-2)
573        if (_seff_mean >= 0.01) and (_seff_sd >= 0.01):
574            rate_alpha_precomputed = (_seff_sd ** 2) / (_seff_mean ** 2)
575            rate_beta_precomputed = (_seff_sd ** 2) / _seff_mean
576        else:
577            rate_alpha_precomputed = None
578            rate_beta_precomputed = None
579
580        guide_x = pyro.infer.autoguide.AutoNormalMessenger(self._model_x, init_loc_fn=init_loc_fn)
581        guide_x.to(self.model.device)
582        optimizer = pyro.optim.Adam({"lr": lr})
583        svi = pyro.infer.SVI(self._model_x, guide_x, optimizer,
584                             loss=pyro.infer.Trace_ELBO())
585
586        original_device = self.model.device
587        losses = []
588        smoothed_loss = None
589        alpha_x_sample_loop = self.model.alpha_x_prefit
590
591        for step in range(niters):
592            loss = svi.step(
593                N,
594                G,
595                guides_tensor,
596                x_obs_tensor,
597                sum_factor_tensor,
598                mu_x_mean_tensor,
599                mu_x_sd_tensor,
600                _seff_mean,
601                _seff_sd,
602                C=C,
603                groups_tensor=groups_tensor,
604                alpha_x_sample=alpha_x_sample_loop,
605                target_per_guide_tensor=target_per_guide_tensor,
606                independent_mu_sigma=independent_mu_sigma,
607                phi_x_precomputed=phi_x_precomputed,
608                rate_alpha_precomputed=rate_alpha_precomputed,
609                rate_beta_precomputed=rate_beta_precomputed,
610                scale_2_tensor=scale_2_tensor,
611            )
612            losses.append(loss)
613            if step % 1000 == 0:
614                print(f"Step {step} : loss = {loss:.5e}, device: {mu_x_mean_tensor.device}")
615            if smoothed_loss is None:
616                smoothed_loss = loss
617            else:
618                smoothed_loss = alpha_ewma * loss + (1 - alpha_ewma) * smoothed_loss
619
620        # Move to CPU if using too much GPU memory for Predictive
621        run_on_cpu = predictive_on_cpu and self.model.device.type != "cpu"
622
623        if run_on_cpu:
624            print(f"[INFO] SVI completed on {original_device}. Running Predictive on CPU to reduce GPU memory pressure...")
625            guide_x.to("cpu")
626            self.model.device = torch.device("cpu")
627            # Move high-MOI tensors that are accessed directly in _model_x
628            if self.model.is_high_moi:
629                self.model.guide_assignment_tensor = self.model.guide_assignment_tensor.cpu()
630                self.model._ntc_guide_mask = self.model._ntc_guide_mask.cpu()
631
632            model_inputs = {
633                "N": N,
634                "G": G,
635                "guides_tensor": self._to_cpu(guides_tensor),
636                "x_obs_tensor": self._to_cpu(x_obs_tensor),
637                "sum_factor_tensor": self._to_cpu(sum_factor_tensor),
638                "mu_x_mean_tensor": self._to_cpu(mu_x_mean_tensor),
639                "mu_x_sd_tensor": self._to_cpu(mu_x_sd_tensor),
640                "sigma_eff_mean_tensor": self._to_cpu(_seff_mean),
641                "sigma_eff_sd_tensor": self._to_cpu(_seff_sd),
642                "C": C,
643                "groups_tensor": self._to_cpu(groups_tensor),

645                "target_per_guide_tensor": self._to_cpu(target_per_guide_tensor),
646                "independent_mu_sigma": independent_mu_sigma,
647                "phi_x_precomputed": self._to_cpu(phi_x_precomputed),
648                "rate_alpha_precomputed": self._to_cpu(rate_alpha_precomputed),
649                "rate_beta_precomputed": self._to_cpu(rate_beta_precomputed),
650                "scale_2_tensor": self._to_cpu(scale_2_tensor),
651            }
652        else:
653            model_inputs = {
654                "N": N,
655                "G": G,
656                "guides_tensor": guides_tensor,
657                "x_obs_tensor": x_obs_tensor,
658                "sum_factor_tensor": sum_factor_tensor,
659                "mu_x_mean_tensor": mu_x_mean_tensor,
660                "mu_x_sd_tensor": mu_x_sd_tensor,
661                "sigma_eff_mean_tensor": _seff_mean,
662                "sigma_eff_sd_tensor": _seff_sd,
663                "C": C,
664                "groups_tensor": groups_tensor,
665                "alpha_x_sample": self.model.alpha_x_prefit,
666                "target_per_guide_tensor": target_per_guide_tensor,
667                "independent_mu_sigma": independent_mu_sigma,
668                "phi_x_precomputed": phi_x_precomputed,
669                "rate_alpha_precomputed": rate_alpha_precomputed,
670                "rate_beta_precomputed": rate_beta_precomputed,
671                "scale_2_tensor": scale_2_tensor,
672            }
673
674        if self.model.device.type == "cuda":
675            torch.cuda.empty_cache()
676        gc.collect()
677
678        max_samples = nsamples
679        keep_sites = kwargs.get("keep_sites", lambda name, site: site["value"].ndim <= 2 or name != "x_obs")
680
681        if minibatch_size is not None:
682            from collections import defaultdict
683            print(f"[INFO] Running Predictive in minibatches of {minibatch_size}...")
684            predictive_x = pyro.infer.Predictive(
685                self._model_x,
686                guide=guide_x,
687                num_samples=minibatch_size,
688                parallel=True
689            )
690            all_samples = defaultdict(list)
691            with torch.no_grad():
692                for i in range(0, max_samples, minibatch_size):
693                    samples = predictive_x(**model_inputs)
694                    for k, v in samples.items():
695                        if keep_sites(k, {"value": v}):
696                            all_samples[k].append(self._to_cpu(v))
697                    if self.model.device.type == "cuda":
698                        torch.cuda.empty_cache()
699                    gc.collect()
700            posterior_samples_x = {k: torch.cat(v, dim=0) for k, v in all_samples.items()}
701        else:
702            predictive_x = pyro.infer.Predictive(
703                self._model_x,
704                guide=guide_x,
705                num_samples=nsamples
706            )
707            with torch.no_grad():
708                posterior_samples_x = predictive_x(**model_inputs)
709                if self.model.device.type == "cuda":
710                    torch.cuda.empty_cache()
711                gc.collect()
712
713        if run_on_cpu:
714            self.model.device = original_device
715            print("[INFO] Reset self.model.device to:", self.model.device)
716            if self.model.is_high_moi:
717                self.model.guide_assignment_tensor = self.model.guide_assignment_tensor.to(original_device)
718                self.model._ntc_guide_mask = self.model._ntc_guide_mask.to(original_device)
719
720        for k, v in posterior_samples_x.items():
721            posterior_samples_x[k] = self._to_cpu(v)
722
723        self.model.loss_x = losses
724        # Store full posterior on model (not just on CisFitter)
725        self.model.posterior_samples_cis = posterior_samples_x
726        self.model.x_true = posterior_samples_x['x_true'].median(dim=0).values
727        self.model.log2_x_true = posterior_samples_x['log_x_true'].median(dim=0).values
728
729        if self.model.device.type == "cuda":
730            torch.cuda.empty_cache()
731        gc.collect()
732        pyro.clear_param_store()
733
734        print("Finished fit_cis.")

 45    # Step 2: Fit cis effects (model_x)

 47    def _model_x(
 48        self,
 49        N,
 50        G,
 51        guides_tensor,
 52        x_obs_tensor,
 53        sum_factor_tensor,
 54        mu_x_mean_tensor,
 55        mu_x_sd_tensor,
 56        sigma_eff_mean_tensor,
 57        sigma_eff_sd_tensor,
 58        C=None,
 59        groups_tensor=None,
 60        alpha_x_sample=None,
 61        target_per_guide_tensor=None,
 62        independent_mu_sigma=False,
 63        phi_x_precomputed=None,
 64        rate_alpha_precomputed=None,
 65        rate_beta_precomputed=None,
 66        scale_2_tensor=None,
 67    ):
 68
 69        ###################################
 70        ## Technical covariate modelling ##
 71        ###################################
 72        if alpha_x_sample is not None:
 73            alpha_x = alpha_x_sample
 74        else:
 75            alpha_x = None
 76        
 77        ####################
 78        ## Overdispersion ##
 79        ####################
 80        if phi_x_precomputed is None:
 81            raise ValueError(
 82                "phi_x_precomputed is required. "
 83                "Run fit_ntc() before fit_cis() to estimate cis gene overdispersion from NTC cells."
 84            )
 85        phi_x_used = phi_x_precomputed
 86
 87        ###############################
 88        ## Mixture model for x_eff_g ##
 89        ###############################
 90        if independent_mu_sigma:
 91            unique_targets = torch.unique(target_per_guide_tensor)
 92            mu_targets = {}
 93            sigma_targets = {}
 94            _hc_scale = scale_2_tensor if scale_2_tensor is not None else self._t(2.0)
 95            for t in unique_targets:
 96                mu_targets[int(t.item())] = pyro.sample(f"mu_target_{int(t.item())}", dist.Normal(mu_x_mean_tensor, mu_x_sd_tensor))
 97                sigma_targets[int(t.item())] = pyro.sample(f"sigma_target_{int(t.item())}", dist.HalfCauchy(scale=_hc_scale))
 98            # gather mu and sigma for each guide
 99            mu_target_tensor = torch.stack([mu_targets[int(t.item())] for t in unique_targets], dim=0)
100            sigma_target_tensor = torch.stack([sigma_targets[int(t.item())] for t in unique_targets], dim=0)
101            mu = mu_target_tensor[target_per_guide_tensor]
102            sigma = sigma_target_tensor[target_per_guide_tensor]
103        else:
104            mu = pyro.sample("mu", dist.Normal(mu_x_mean_tensor, mu_x_sd_tensor))
105            _hc_scale = scale_2_tensor if scale_2_tensor is not None else self._t(2.0)
106            sigma = pyro.sample("sigma", dist.HalfCauchy(scale=_hc_scale))
107            mu = mu.expand(G)
108            sigma = sigma.expand(G)
109
110        # Non-centered parameterization
111        if rate_alpha_precomputed is not None:
112            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Exponential(rate_alpha_precomputed))
113            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Exponential(rate_beta_precomputed))
114        elif (sigma_eff_mean_tensor >= 0.01) and (sigma_eff_sd_tensor >= 0.01):
115            rate_alpha = (sigma_eff_sd_tensor ** 2) / (sigma_eff_mean_tensor ** 2)
116            rate_beta = (sigma_eff_sd_tensor ** 2) / sigma_eff_mean_tensor
117            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Exponential(rate_alpha))
118            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Exponential(rate_beta))
119        else:
120            sigma_eff_alpha = pyro.sample("sigma_eff_alpha", dist.Gamma(self._t(1.0), self._t(0.01)))
121            sigma_eff_beta = pyro.sample("sigma_eff_beta", dist.Gamma(self._t(1.0), self._t(0.01)))
122        with pyro.plate("guides_plate", G):
123            eps_x_eff_g = pyro.sample("eps_x_eff_g", dist.StudentT(df=self._t(3.0), loc=self._t(0.0), scale=self._t(1.0)))
124            log2_x_eff_g = mu + sigma * eps_x_eff_g
125            x_eff_g = pyro.deterministic("x_eff_g", 2.0 ** log2_x_eff_g)
126        
127            sigma_eff = pyro.sample("sigma_eff", dist.Gamma(sigma_eff_alpha, sigma_eff_beta))
128                
129        ##########################
130        ## Cell-level variables ##
131        ##########################
132        if alpha_x is not None:
133            # alpha_x shape [C]: already includes reference group (index 0 = 1.0)
134            # or shape [C-1]: needs reference group prepended
135            if alpha_x.shape[-1] == C:
136                alpha_x_full = alpha_x
137            else:
138                ones_ = torch.ones(1, device=self.model.device)
139                alpha_x_full = torch.cat([ones_, alpha_x], dim=-1)  # shape = [C]
140            alpha_x_used = alpha_x_full[groups_tensor]  # shape = [N]
141        else:
142            alpha_x_used = torch.ones_like(sum_factor_tensor)  # shape = [N] or [S,N]
143
144        # Aggregate guide effects to cells
145        if self.model.is_high_moi:
146            # High MOI: effects are additive in log2FC space.
147            # log2FC for guide g = log2(x_eff_g[g]) - log2(NTC_mean)
148            # For a cell with guides A, B: log2FC_cell = log2FC_A + log2FC_B
149            # => log2(x_mean) = log2(NTC) + log2FC_cell
150            #                 = sum(log2(x_eff_g)) - (n_guides - 1) * log2(NTC)
151            # This is always finite since x_eff_g > 0 and NTC_mean > 0.
152
153            # NTC reference: weighted mean of NTC guide effects
154            # Use pre-computed mask (set in fit_cis before SVI loop to avoid pandas access here)
155            ntc_mask = self.model._ntc_guide_mask
156            cells_per_guide = self.model.guide_assignment_tensor.sum(dim=0)  # [G]
157            weights = cells_per_guide / sigma_eff.clamp(min=1e-6)            # [G]
158            ntc_weights = weights[ntc_mask]
159            ntc_effects  = x_eff_g[ntc_mask]
160            weighted_mean_NTC = (ntc_weights * ntc_effects).sum(dim=-1) / ntc_weights.sum()
161            weighted_mean_NTC = pyro.deterministic("weighted_mean_NTC", weighted_mean_NTC)
162
163            log2_x_eff_g  = torch.log2(x_eff_g.clamp(min=1e-12))                               # [G]
164            log2_NTC      = torch.log2(weighted_mean_NTC.clamp(min=1e-12))                       # scalar
165            guides_per_cell = self.model.guide_assignment_tensor.sum(dim=1).clamp(min=1)         # [N]
166            sum_log2_effects = torch.matmul(self.model.guide_assignment_tensor, log2_x_eff_g)    # [N]
167            log2_x_mean   = sum_log2_effects - (guides_per_cell - 1) * log2_NTC                 # [N]
168            x_mean        = 2.0 ** log2_x_mean                                                   # [N]
169            sigma_mean    = torch.matmul(self.model.guide_assignment_tensor, sigma_eff) / guides_per_cell  # [N]
170        else:
171            # Single guide per cell: use indexing (unchanged)
172            x_mean = x_eff_g[..., guides_tensor]  # [N]
173            sigma_mean = sigma_eff[..., guides_tensor]  # [N]
174
175        ######################
176        ## Cell-level plate ##
177        ######################
178        with pyro.plate("data_plate", N):
179            log_x_true = pyro.sample( # use log2 of xtrue to allow small values of xtrue
180                "log_x_true",
181                dist.Normal(torch.log2(x_mean), sigma_mean)
182            )
183            x_true = pyro.deterministic("x_true", 2.0 ** log_x_true)
184            mu_obs = alpha_x_used * x_true * sum_factor_tensor
185            pyro.sample(
186                "x_obs",
187                dist.NegativeBinomial(total_count=phi_x_used,
188                                      logits=torch.log(mu_obs) - torch.log(phi_x_used),
189                                      validate_args=False
190                                     ), # important to use logits to allow small x_true
191                obs=x_obs_tensor
192            )

206        minibatch_size: int = None,
207        predictive_on_cpu: bool = True,
208        independent_mu_sigma: bool = False,
209        **kwargs
210    ):
211        """
212        Fits the cis effects (model_x) for your gene_of_interest.
213        This step can be repeated multiple times with different priors
214        or hyperparameters.
215
216        Parameters
217        ----------
218        technical_covariates : list, optional
219            Technical covariates for correction
220        sum_factor_col : str
221            Column name for size factors
222        cis_feature : str, optional
223            Feature ID to use as cis proxy from the primary modality.
224            If None, uses self.model.cis_gene (must exist in primary modality).
225            For ATAC: region ID (e.g., 'chr9:132283881-132284881')
226            For genes: gene name
227        manual_guide_effects : pd.DataFrame, optional
228            Manual guide effect estimates as priors. DataFrame with columns:
229            - guide: guide identifier (matches meta['guide'])
230            - log2FC: expected log2 fold-change vs NTC
231        prior_strength : float
232            Weight for manual guide effects (default: 1.0)
233            0 = ignore manual effects, higher = trust more
234        lr : float
235            Learning rate for Adam
236        niters : int
237            Number of SVI iterations
238        nsamples : int
239            Number of posterior samples
240        predictive_on_cpu : bool
241            If True (default), run posterior Predictive sampling on CPU after SVI
242            to reduce GPU memory pressure. Set False to keep Predictive on GPU.
243        alpha_ewma : float
244            Exponential weight for smoothing the ELBO
245        independent_mu_sigma : bool
246            Whether to use independent mu/sigma per target type
247        kwargs :
248            Additional arguments controlling priors, etc.
249        """
250        print("Running fit_cis...")
251
252        if self.model.cis_gene is None:
253            raise ValueError("self.model.cis_gene must be set.")
254
255        # fit_cis ALWAYS uses the 'cis' modality
256        # This modality is created automatically when bayesDREAM is initialized
257
258        # Get cis modality
259        if 'cis' not in self.model.modalities:
260            raise ValueError(
261                "No 'cis' modality found. The 'cis' modality should be created automatically "
262                "when bayesDREAM is initialized with a cis_gene. "
263                "Make sure cis_gene parameter is set during initialization."
264            )
265
266        cis_modality = self.model.get_modality('cis')
267
268        # Determine which feature to use as cis proxy
269        if cis_feature is None:
270            # Default: use the first (and typically only) feature in cis modality
271            cis_feature_idx = cis_modality.feature_meta.index[0]
272            # Get actual feature name from metadata (not just numeric index)
273            if 'gene_name' in cis_modality.feature_meta.columns:
274                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'gene_name']
275            elif 'gene' in cis_modality.feature_meta.columns:
276                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'gene']
277            elif 'feature' in cis_modality.feature_meta.columns:
278                cis_feature_name = cis_modality.feature_meta.loc[cis_feature_idx, 'feature']
279            else:
280                # Fallback to index if no name column found
281                cis_feature_name = cis_feature_idx
282            cis_feature = cis_feature_idx  # Use numeric index for actual data retrieval
283            print(f"[INFO] Using cis feature '{cis_feature_name}' from 'cis' modality")
284        else:
285            # User specified explicit cis_feature
286            if cis_feature not in cis_modality.feature_meta.index:
287                raise ValueError(
288                    f"cis_feature '{cis_feature}' not found in 'cis' modality.\n"
289                    f"Available features: {cis_modality.feature_meta.index.tolist()}"
290                )
291            print(f"[INFO] Using cis_feature '{cis_feature}' from 'cis' modality")
292
293        # Get counts for cis feature from cis modality
294        if isinstance(cis_modality.counts, pd.DataFrame):
295            cis_counts = cis_modality.counts.loc[cis_feature].values
296        else:
297            # numpy/sparse array - need to find index
298            feature_idx = cis_modality.feature_meta.index.get_loc(cis_feature)
299            if cis_modality.cells_axis == 1:
300                cis_counts = cis_modality.counts[feature_idx, :]
301            else:
302                cis_counts = cis_modality.counts[:, feature_idx]
303            # Densify if sparse (slicing sparse returns sparse row/col matrix)
304            if hasattr(cis_counts, 'toarray'):
305                cis_counts = cis_counts.toarray().ravel()
306
307        # convert to gpu for fitting
308        if self.model.alpha_x_prefit is not None and self.model.alpha_x_prefit.device != self.model.device:
309            self.model.alpha_x_prefit = self.model.alpha_x_prefit.to(self.model.device)
310
311        if technical_covariates:
312            new_technical_group_code = self.model.meta.groupby(technical_covariates).ngroup()
313            if "technical_group_code" in self.model.meta.columns:
314                old_technical_group_code = self.model.meta["technical_group_code"].values
315                groups_changed = not np.array_equal(old_technical_group_code, new_technical_group_code.values)
316                if groups_changed:
317                    warnings.warn("technical_group already set. Overwriting with new covariate grouping.")
318                    if self.model.alpha_x_prefit is not None:
319                        warnings.warn("Technical groups changed; discarding alpha_x_prefit and refitting.")
320                        self.model.alpha_x_prefit = None
321                else:
322                    # Keep loaded alpha_x_prefit when grouping is unchanged (common in stepwise workflows)
323                    pass
324
325            self.model.meta["technical_group_code"] = new_technical_group_code
326            C = self.model.meta['technical_group_code'].nunique()
327            groups_tensor = torch.tensor(self.model.meta['technical_group_code'].values, dtype=torch.long, device=self.model.device)
328
329            # alpha_x_prefit is required when technical_covariates are specified
330            if self.model.alpha_x_prefit is None:
331                raise ValueError(
332                    f"Technical covariates provided but alpha_x_prefit not set. "
333                    f"Run fit_ntc() on the primary modality ('{self.model.primary_modality}') before fit_cis() "
334                    f"to estimate technical effects for the cis gene."
335                )
336
337        elif self.model.alpha_x_prefit is None:
338            C = None
339            groups_tensor = None
340            warnings.warn("no alpha_x_prefit and no technical_covariates provided, assuming no confounding effect.")
341        else:
342            # alpha_x_prefit exists but no new technical_covariates specified
343            # Use existing technical groups
344            C = self.model.meta['technical_group_code'].nunique()
345            groups_tensor = torch.tensor(self.model.meta['technical_group_code'].values, dtype=torch.long, device=self.model.device)
346        
347        N = self.model.meta.shape[0]
348
349        # Handle G (number of guides) differently for high MOI vs single-guide mode
350        if self.model.is_high_moi:
351            G = self.model.guide_assignment.shape[1]  # Number of guides
352            guides_tensor = None  # Not used in high MOI mode
353        else:
354            G = self.model.meta['guide_code'].nunique()
355            guides_tensor = torch.tensor(self.model.meta['guide_code'].values, dtype=torch.long, device=self.model.device)
356
357        # Validate minimum data requirements for stable initialization
358        min_guides = 3
359        min_cells_per_guide = 5
360        if self.model.is_high_moi:
361            cells_per_guide = self.model.guide_assignment.sum(axis=0)  # [G]
362            n_adequate = int((cells_per_guide >= min_cells_per_guide).sum())
363        else:
364            guide_counts = self.model.meta['guide_code'].value_counts()
365            n_adequate = int((guide_counts >= min_cells_per_guide).sum())
366        if n_adequate < min_guides:
367            raise ValueError(
368                f"fit_cis requires at least {min_guides} guides with >= {min_cells_per_guide} cells each, "
369                f"but only {n_adequate} guides meet this criterion. "
370                f"Subsetting to too few cells/guides will produce unreliable initializations."
371            )
372
373        # Use cis_counts from modality-specific lookup (or traditional self.model.counts)
374        x_obs_tensor = torch.tensor(cis_counts, dtype=torch.float32, device=self.model.device)
375
376        # ========================================================================
377        # MANUAL GUIDE EFFECTS INFRASTRUCTURE
378        # ========================================================================
379        # If user provides manual guide effects (log2FC estimates), prepare them
380        # as tensors that can be used as priors in the Pyro model.
381        manual_guide_log2fc_tensor = None
382        manual_guide_mask_tensor = None
383
384        if manual_guide_effects is not None:
385            raise NotImplementedError(
386                "manual_guide_effects is not yet implemented. "
387                "The infrastructure for guide-level priors is planned but not yet integrated into _model_x."
388            )
389        # ========================================================================
390        if independent_mu_sigma:
391            if ('target' not in self.model.meta.columns):
392                raise ValueError("independent_mu_sigma is True, self.model.meta['target'] column not found.")
393            elif self.model.meta['target'].nunique() < 2:
394                raise ValueError("independent_mu_sigma is True, but only 1 target type found in self.model.meta['target'] column.")
395
396            ### BUILD target_per_guide_tensor [G] based on guide → target
397            if self.model.is_high_moi:
398                # High MOI: derive one target label per guide for grouping mu/sigma.
399                # Two sources: guide_meta['target'] (simple) or guide_targets_dict (many-to-many).
400                _ntc_variants = {'ntc', 'NTC', 'non-targeting', 'non-targeting-control', 'Non-Targeting'}
401
402                if 'target' in self.model.guide_meta.columns:
403                    guide_target_labels = self.model.guide_meta['target'].tolist()
404                elif hasattr(self.model, 'guide_targets_dict') and self.model.guide_targets_dict:
405                    # Derive a single representative target per guide.
406                    # Priority: cis_gene > any NTC variant > first target.
407                    def _primary_target(targets):
408                        if self.model.cis_gene and self.model.cis_gene in targets:
409                            return self.model.cis_gene
410                        for t in targets:
411                            if t in _ntc_variants:
412                                return 'ntc'
413                        return targets[0] if targets else 'ntc'
414
415                    guide_target_labels = [
416                        _primary_target(self.model.guide_targets_dict.get(row['guide'], ['ntc']))
417                        for _, row in self.model.guide_meta.iterrows()
418                    ]
419                else:
420                    raise ValueError(
421                        "independent_mu_sigma=True in high MOI mode requires either "
422                        "guide_meta['target'] or guide_targets_dict."
423                    )
424
425                target_factorized, target_unique = pd.factorize(guide_target_labels)
426                target_per_guide_tensor = torch.tensor(target_factorized, dtype=torch.long, device=self.model.device)
427                print(f"[INFO] independent_mu_sigma (high MOI): {len(target_unique)} unique targets")
428            else:
429                # Single-guide mode: one target code per guide (raises if a guide has multiple targets)
430                self.model.meta['target_code'] = pd.factorize(self.model.meta['target'])[0]
431                target_codes_tensor = torch.tensor(self.model.meta['target_code'].values, dtype=torch.long, device=self.model.device)
432
433                target_per_guide_tensor = torch.empty(G, dtype=torch.long, device=self.model.device)
434                for g in range(G):
435                    idx = (guides_tensor == g)
436                    guide_targets = torch.unique(target_codes_tensor[idx])
437                    if guide_targets.shape[0] != 1:
438                        raise ValueError(f"Guide {g} maps to multiple targets: {guide_targets}. independent_mu_sigma=True requires unambiguous target assignment per guide.")
439                    target_per_guide_tensor[g] = guide_targets[0]
440        else:
441            target_per_guide_tensor = None
442        primary_mod = self.model.get_modality(self.model.primary_modality)
443        sum_factor_tensor = torch.tensor(
444            primary_mod.sum_factors.loc[self.model.meta['cell'].values, sum_factor_col].values,
445            dtype=torch.float32, device=self.model.device
446        )
447
448        # Compute guide means (adjusting for alpha_x if provided)
449        x_obs_factored = x_obs_tensor / sum_factor_tensor
450        
451        if self.model.alpha_x_prefit is not None:
452            # alpha_x_prefit is always a [C] point estimate (mean already taken at fit_ntc time)
453            alpha_x_full = self.model.alpha_x_prefit.flatten()
454            # Select the correct alpha_x for each observation (expand for broadcasting)
455            alpha_x_used = alpha_x_full[groups_tensor]  # groups_tensor indexes into (C,)
456
457            # Adjust x_obs_factored by alpha_x_used
458            x_obs_factored /= alpha_x_used  # Ensure correct shape for division
459        
460        # --- Extract o_x from the technical fit (cis modality posterior) ---
461        cis_modality = self.model.get_modality('cis')
462        if (cis_modality.posterior_samples_ntc is None
463                or 'o_x' not in cis_modality.posterior_samples_ntc):
464            raise ValueError(
465                "Cis gene overdispersion not found. Run fit_ntc() before fit_cis().\n"
466                "fit_ntc() estimates o_x for the cis gene from NTC cells, which is required "
467                "to set a data-driven overdispersion prior instead of the generic Gamma(9,3)."
468            )
469        o_x_ntc = float(cis_modality.posterior_samples_ntc['o_x'].mean().item())
470        print(f"[INFO] fit_cis: using technical o_x = {o_x_ntc:.4f} (phi = {1/o_x_ntc**2:.2f})")
471
472        # Compute guide-level means and MADs.
473        # Uses numpy on CPU for per-guide median computation (faster than G GPU boolean masks).
474        x_np = x_obs_factored.detach().cpu().numpy().astype(np.float32)
475
476        if self.model.is_high_moi:
477            # High MOI: vectorized means via matrix-vector multiply, MADs via sorted numpy slices
478            assign_np = self.model.guide_assignment  # [N, G] numpy array
479            cell_counts_np = assign_np.sum(axis=0).clip(min=1)  # [G]
480            guide_sums_np = x_np @ assign_np  # [G]
481            guide_means_np = np.log2(np.clip(guide_sums_np / cell_counts_np, epsilon, None))
482
483            log2_x_np = np.log2(np.clip(x_np, epsilon, None))
484            guide_mads_np = np.zeros(G, dtype=np.float32)
485            for g in range(G):
486                cell_mask = assign_np[:, g].astype(bool)
487                if cell_mask.sum() > 0:
488                    vals = log2_x_np[cell_mask]
489                    med = np.median(vals)
490                    guide_mads_np[g] = float(np.median(np.abs(vals - med)))
491
492            guide_means = torch.from_numpy(guide_means_np).to(self.model.device)
493            guide_mads_tensor = torch.from_numpy(guide_mads_np).to(self.model.device)
494        else:
495            # Single-guide mode: vectorized means via bincount, MADs via sorted numpy slices
496            guides_np = guides_tensor.cpu().numpy()
497            n_per_guide_np = np.bincount(guides_np, minlength=G).clip(min=1).astype(np.float32)
498            sum_per_guide_np = np.bincount(guides_np, weights=x_np, minlength=G).astype(np.float32)
499            guide_means_np = np.log2(np.clip(sum_per_guide_np / n_per_guide_np, epsilon, None))
500
501            log2_x_np = np.log2(np.clip(x_np, epsilon, None))
502            sort_idx = np.argsort(guides_np, kind='stable')
503            guides_sorted = guides_np[sort_idx]
504            log2_x_sorted = log2_x_np[sort_idx]
505            boundaries = np.searchsorted(guides_sorted, np.arange(G + 1))
506
507            guide_mads_np = np.zeros(G, dtype=np.float32)
508            for g in range(G):
509                s, e = boundaries[g], boundaries[g + 1]
510                if e > s:
511                    vals = log2_x_sorted[s:e]
512                    med = np.median(vals)
513                    guide_mads_np[g] = float(np.median(np.abs(vals - med)))
514
515            guide_means = torch.from_numpy(guide_means_np).to(self.model.device)
516            guide_mads_tensor = torch.from_numpy(guide_mads_np).to(self.model.device)
517
518        mu_x_mean_tensor = torch.mean(guide_means)
519        mu_x_sd_tensor = torch.std(guide_means)
520
521        guide_mads_tensor = guide_mads_tensor * 1.4826  # Gaussian-equivalent spread
522        sigma_eff_mean_tensor = torch.mean(guide_mads_tensor)#.to(self.model.device)
523        sigma_eff_sd_tensor = torch.std(guide_mads_tensor)#.to(self.model.device)
524
525        def init_loc_fn(site):
526            name = site["name"]
527        
528            if name == "mu":
529                return mu_x_mean_tensor
530            elif name == "sigma":
531                return mu_x_sd_tensor.clamp(min=1e-3)
532        
533            elif name == "sigma_eff_alpha":
534                # Corresponds to (mean^2 / var)
535                sigma_eff_mean_tensor_tmp = sigma_eff_mean_tensor.clamp(min=1e-2)
536                sigma_eff_sd_tensor_tmp = sigma_eff_sd_tensor.clamp(min=1e-2)
537                return ((sigma_eff_mean_tensor_tmp ** 2) / (sigma_eff_sd_tensor_tmp ** 2))
538            elif name == "sigma_eff_beta":
539                # Corresponds to (mean / var)
540                sigma_eff_mean_tensor_tmp = sigma_eff_mean_tensor.clamp(min=1e-2)
541                sigma_eff_sd_tensor_tmp = sigma_eff_sd_tensor.clamp(min=1e-2)
542                return (sigma_eff_mean_tensor_tmp / (sigma_eff_sd_tensor_tmp ** 2))
543        
544            elif name == "sigma_eff":
545                # Assume you’re sampling per-guide (e.g. G = 37)
546                return sigma_eff_mean_tensor.clamp(min=1e-2).expand(G)
547        
548            return pyro.infer.autoguide.initialization.init_to_median(site)
549
550        # Pre-compute NTC guide mask for high MOI mode (avoids pandas access inside Pyro model)
551        if self.model.is_high_moi:
552            _ntc_variants = {'ntc', 'NTC', 'non-targeting', 'non-targeting-control', 'Non-Targeting'}
553            if 'target' in self.model.guide_meta.columns:
554                ntc_flags = [self.model.guide_meta.iloc[g]['target'] in _ntc_variants
555                             for g in range(G)]
556            elif hasattr(self.model, 'guide_targets_dict') and self.model.guide_targets_dict:
557                ntc_flags = [
558                    any(t in _ntc_variants for t in self.model.guide_targets_dict.get(row['guide'], []))
559                    for _, row in self.model.guide_meta.iterrows()
560                ]
561            else:
562                raise ValueError(
563                    "High MOI mode requires either guide_meta['target'] or guide_targets_dict "
564                    "to identify NTC guides."
565                )
566            self.model._ntc_guide_mask = torch.tensor(ntc_flags, dtype=torch.bool,
567                                                       device=self.model.device)
568
569        # Precompute model invariants once to avoid repeated computation in every SVI step
570        phi_x_precomputed = torch.tensor(1.0 / (o_x_ntc ** 2), dtype=torch.float32, device=self.model.device)
571        scale_2_tensor = self._t(2.0)
572        _seff_mean = sigma_eff_mean_tensor.clamp(min=1e-2)
573        _seff_sd = sigma_eff_sd_tensor.clamp(min=1e-2)
574        if (_seff_mean >= 0.01) and (_seff_sd >= 0.01):
575            rate_alpha_precomputed = (_seff_sd ** 2) / (_seff_mean ** 2)
576            rate_beta_precomputed = (_seff_sd ** 2) / _seff_mean
577        else:
578            rate_alpha_precomputed = None
579            rate_beta_precomputed = None
580
581        guide_x = pyro.infer.autoguide.AutoNormalMessenger(self._model_x, init_loc_fn=init_loc_fn)
582        guide_x.to(self.model.device)
583        optimizer = pyro.optim.Adam({"lr": lr})
584        svi = pyro.infer.SVI(self._model_x, guide_x, optimizer,
585                             loss=pyro.infer.Trace_ELBO())
586
587        original_device = self.model.device
588        losses = []
589        smoothed_loss = None
590        alpha_x_sample_loop = self.model.alpha_x_prefit
591
592        for step in range(niters):
593            loss = svi.step(
594                N,
595                G,
596                guides_tensor,
597                x_obs_tensor,
598                sum_factor_tensor,
599                mu_x_mean_tensor,
600                mu_x_sd_tensor,
601                _seff_mean,
602                _seff_sd,
603                C=C,
604                groups_tensor=groups_tensor,
605                alpha_x_sample=alpha_x_sample_loop,
606                target_per_guide_tensor=target_per_guide_tensor,
607                independent_mu_sigma=independent_mu_sigma,
608                phi_x_precomputed=phi_x_precomputed,
609                rate_alpha_precomputed=rate_alpha_precomputed,
610                rate_beta_precomputed=rate_beta_precomputed,
611                scale_2_tensor=scale_2_tensor,
612            )
613            losses.append(loss)
614            if step % 1000 == 0:
615                print(f"Step {step} : loss = {loss:.5e}, device: {mu_x_mean_tensor.device}")
616            if smoothed_loss is None:
617                smoothed_loss = loss
618            else:
619                smoothed_loss = alpha_ewma * loss + (1 - alpha_ewma) * smoothed_loss
620
621        # Move to CPU if using too much GPU memory for Predictive
622        run_on_cpu = predictive_on_cpu and self.model.device.type != "cpu"
623
624        if run_on_cpu:
625            print(f"[INFO] SVI completed on {original_device}. Running Predictive on CPU to reduce GPU memory pressure...")
626            guide_x.to("cpu")
627            self.model.device = torch.device("cpu")
628            # Move high-MOI tensors that are accessed directly in _model_x
629            if self.model.is_high_moi:
630                self.model.guide_assignment_tensor = self.model.guide_assignment_tensor.cpu()
631                self.model._ntc_guide_mask = self.model._ntc_guide_mask.cpu()
632
633            model_inputs = {
634                "N": N,
635                "G": G,
636                "guides_tensor": self._to_cpu(guides_tensor),
637                "x_obs_tensor": self._to_cpu(x_obs_tensor),
638                "sum_factor_tensor": self._to_cpu(sum_factor_tensor),
639                "mu_x_mean_tensor": self._to_cpu(mu_x_mean_tensor),
640                "mu_x_sd_tensor": self._to_cpu(mu_x_sd_tensor),
641                "sigma_eff_mean_tensor": self._to_cpu(_seff_mean),
642                "sigma_eff_sd_tensor": self._to_cpu(_seff_sd),
643                "C": C,
644                "groups_tensor": self._to_cpu(groups_tensor),
645                "alpha_x_sample": self._to_cpu(self.model.alpha_x_prefit),
646                "target_per_guide_tensor": self._to_cpu(target_per_guide_tensor),
647                "independent_mu_sigma": independent_mu_sigma,
648                "phi_x_precomputed": self._to_cpu(phi_x_precomputed),
649                "rate_alpha_precomputed": self._to_cpu(rate_alpha_precomputed),
650                "rate_beta_precomputed": self._to_cpu(rate_beta_precomputed),
651                "scale_2_tensor": self._to_cpu(scale_2_tensor),
652            }
653        else:
654            model_inputs = {
655                "N": N,
656                "G": G,
657                "guides_tensor": guides_tensor,
658                "x_obs_tensor": x_obs_tensor,
659                "sum_factor_tensor": sum_factor_tensor,
660                "mu_x_mean_tensor": mu_x_mean_tensor,
661                "mu_x_sd_tensor": mu_x_sd_tensor,
662                "sigma_eff_mean_tensor": _seff_mean,
663                "sigma_eff_sd_tensor": _seff_sd,
664                "C": C,
665                "groups_tensor": groups_tensor,
666                "alpha_x_sample": self.model.alpha_x_prefit,
667                "target_per_guide_tensor": target_per_guide_tensor,
668                "independent_mu_sigma": independent_mu_sigma,
669                "phi_x_precomputed": phi_x_precomputed,
670                "rate_alpha_precomputed": rate_alpha_precomputed,
671                "rate_beta_precomputed": rate_beta_precomputed,
672                "scale_2_tensor": scale_2_tensor,
673            }
674
675        if self.model.device.type == "cuda":
676            torch.cuda.empty_cache()
677        gc.collect()
678
679        max_samples = nsamples
680        keep_sites = kwargs.get("keep_sites", lambda name, site: site["value"].ndim <= 2 or name != "x_obs")
681
682        if minibatch_size is not None:
683            from collections import defaultdict
684            print(f"[INFO] Running Predictive in minibatches of {minibatch_size}...")
685            predictive_x = pyro.infer.Predictive(
686                self._model_x,
