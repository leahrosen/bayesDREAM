 26    def save_ntc_fit(self, output_dir: str = None, modalities: list = None, verbose: bool = False,
 27                          save_model_level: bool = None):
 28        """
 29        Save fitted NTC parameters from fit_ntc().
 30
 31        Parameters
 32        ----------
 33        output_dir : str, optional
 34            Directory to save to. If None, uses self.model.output_dir.
 35        modalities : list of str, optional
 36            List of modality names to save. If None, saves all modalities.
 37            Example: ['gene', 'atac']
 38        verbose : bool
 39            If True, print detailed save information. Default False (summary only).
 40
 41        Returns
 42        -------
 43        dict
 44            Paths to saved files
 45
 46        Notes
 47        -----
 48        Saves per-modality:
 49        - alpha_y_prefit_{modality}.pt: Appropriate alpha_y for distribution (add or mult)
 50        - posterior_samples_ntc_{modality}.pt: Full posterior samples
 51
 52        Saves model-level (automatically when primary modality is included):
 53        - alpha_x_prefit.pt: Cis gene overdispersion (if set)
 54        - alpha_y_prefit.pt: Trans gene overdispersion (from primary modality)
 55        """
 56        if output_dir is None:
 57            output_dir = os.path.join(self.model.output_dir, self.model.label)
 58
 59        os.makedirs(output_dir, exist_ok=True)
 60
 61        saved_files = {}
 62        saved_summary = []
 63
 64        # Determine which modalities to save
 65        if modalities is None:
 66            modalities_to_save = list(self.model.modalities.keys())
 67        else:
 68            # Validate requested modalities
 69            invalid = set(modalities) - set(self.model.modalities.keys())
 70            if invalid:
 71                raise ValueError(f"Unknown modalities: {invalid}. Available: {list(self.model.modalities.keys())}")
 72            modalities_to_save = modalities
 73
 74        # Determine whether to save model-level parameters
 75        if save_model_level is None:
 76            should_save_model_level = self.model.primary_modality in modalities_to_save
 77        else:
 78            should_save_model_level = save_model_level
 79
 80        # Save model-level parameters (when primary modality is being saved)
 81        if should_save_model_level:
 82            if hasattr(self.model, 'alpha_x_prefit') and self.model.alpha_x_prefit is not None:
 83                path = os.path.join(output_dir, 'alpha_x_prefit.pt')
 84                torch.save(self.model.alpha_x_prefit, path)
 85                saved_files['alpha_x_prefit'] = path
 86                saved_summary.append('alpha_x')
 87                if verbose:
 88                    print(f"[SAVE] alpha_x_prefit → {path}")
 89
 90        # Save per-modality alpha_y_prefit and posterior_samples_ntc
 91        for mod_name in modalities_to_save:
 92            mod = self.model.modalities[mod_name]
 93            mod_saved = []
 94
 95            # Save distribution-appropriate alpha_y
 96            if hasattr(mod, 'alpha_y_prefit') and mod.alpha_y_prefit is not None:
 97                # Determine which alpha_y to save based on distribution
 98                if mod.distribution == 'negbinom':
 99                    # Save multiplicative for negbinom
100                    if hasattr(mod, 'alpha_y_prefit_mult') and mod.alpha_y_prefit_mult is not None:
101                        alpha_to_save = mod.alpha_y_prefit_mult
102                        alpha_type = 'mult'
103                    else:
104                        alpha_to_save = mod.alpha_y_prefit
105                        alpha_type = 'mult'
106                else:
107                    # Save additive for normal, binomial, multinomial
108                    if hasattr(mod, 'alpha_y_prefit_add') and mod.alpha_y_prefit_add is not None:
109                        alpha_to_save = mod.alpha_y_prefit_add
110                        alpha_type = 'add'
111                    else:
112                        alpha_to_save = mod.alpha_y_prefit
113                        alpha_type = 'add'
114
115                path = os.path.join(output_dir, f'alpha_y_prefit_{mod_name}.pt')
116                torch.save(alpha_to_save, path)
117                saved_files[f'alpha_y_prefit_{mod_name}'] = path
118                mod_saved.append('alpha_y')
119                if verbose:
120                    print(f"[SAVE] {mod_name}.alpha_y_prefit ({alpha_type}) → {path}")
121
122            # Save modality-specific posterior_samples_ntc
123            if hasattr(mod, 'posterior_samples_ntc') and mod.posterior_samples_ntc is not None:
124                # Remove large observation arrays before saving
125                # Ensure alpha_y_add and alpha_y_mult are present for downstream loading
126                posterior_clean = {k: v for k, v in mod.posterior_samples_ntc.items()
127                                 if k not in ['y_obs_ntc', 'y_obs']}
128
129                # Verify critical keys are present for downstream loading
130                if mod.distribution != 'negbinom' and 'alpha_y_add' not in posterior_clean:
131                    if hasattr(mod, 'alpha_y_prefit_add') and mod.alpha_y_prefit_add is not None:
132                        posterior_clean['alpha_y_add'] = mod.alpha_y_prefit_add
133
134                if mod.distribution == 'negbinom' and 'alpha_y_mult' not in posterior_clean and 'alpha_y' not in posterior_clean:
135                    if hasattr(mod, 'alpha_y_prefit_mult') and mod.alpha_y_prefit_mult is not None:
136                        posterior_clean['alpha_y_mult'] = mod.alpha_y_prefit_mult
137
138                # Add feature metadata (including full DataFrame for excluded features tracking)
139                # Use the modality's authoritative feature count (same reason as trans save)
140                n_features = mod.dims.get('n_features', None)
141                if n_features is None:
142                    _sample_tensor = next((v for v in posterior_clean.values()
143                                           if isinstance(v, torch.Tensor) and v.ndim >= 2), None)
144                    n_features = _sample_tensor.shape[-1] if _sample_tensor is not None else None
145                posterior_with_meta = {
146                    'posterior_samples': posterior_clean,
147                    'modality_name': mod_name,
148                    'distribution': mod.distribution,
149                    'feature_names': mod.feature_names if hasattr(mod, 'feature_names') else None,
150                    'n_features': n_features,
151                    'feature_meta': mod.feature_meta.to_dict('records') if hasattr(mod, 'feature_meta') and mod.feature_meta is not None else None,
152                    'loss_ntc': mod.loss_ntc if hasattr(mod, 'loss_ntc') else None
153                }
154
155                path = os.path.join(output_dir, f'posterior_samples_ntc_{mod_name}.pt')
156                torch.save(posterior_with_meta, path)
157                saved_files[f'posterior_samples_ntc_{mod_name}'] = path
158                mod_saved.append(f'posterior({n_features} features)')
159                if verbose:
160                    print(f"[SAVE] {mod_name}.posterior_samples_ntc ({n_features} features) → {path}")
161
162            if mod_saved:
163                saved_summary.append(f"{mod_name}: {', '.join(mod_saved)}")
164
165        # Print summary
166        print(f"[SAVE] NTC fit to {output_dir}")
167        if saved_summary:
168            print(f"[SAVE] Saved: {'; '.join(saved_summary)}")
169
170        return saved_files

173    def save_cis_fit(self, output_dir: str = None, verbose: bool = False):
174        """
175        Save fitted cis parameters from fit_cis().
176
177        Saves:
178        - x_true: True cis gene expression (posterior samples or point estimate)
179        - log2_x_true: Log2-transformed x_true (posterior samples or point estimate)
180        - posterior_samples_cis: Full posterior samples
181
182        Parameters
183        ----------
184        output_dir : str, optional
185            Directory to save to. If None, uses self.model.output_dir.
186        verbose : bool
187            If True, print detailed save information. Default False (summary only).
188
189        Returns
190        -------
191        dict
192            Paths to saved files
193        """
194        if output_dir is None:
195            output_dir = os.path.join(self.model.output_dir, self.model.label)
196
197        os.makedirs(output_dir, exist_ok=True)
198
199        saved_files = {}
200        saved_summary = []
201
202        # Save x_true
203        if hasattr(self.model, 'x_true') and self.model.x_true is not None:
204            path = os.path.join(output_dir, 'x_true.pt')
205            torch.save(self.model.x_true, path)
206            saved_files['x_true'] = path
207            saved_summary.append('x_true')
208            if verbose:
209                print(f"[SAVE] x_true → {path}")
210
211        # Save log2_x_true
212        if hasattr(self.model, 'log2_x_true') and self.model.log2_x_true is not None:
213            path = os.path.join(output_dir, 'log2_x_true.pt')
214            torch.save(self.model.log2_x_true, path)
215            saved_files['log2_x_true'] = path
216            saved_summary.append('log2_x_true')
217            if verbose:
218                print(f"[SAVE] log2_x_true → {path}")
219
220        # Save posterior samples
221        if hasattr(self.model, 'posterior_samples_cis') and self.model.posterior_samples_cis is not None:
222            # Remove large observation arrays
223            posterior_clean = {k: v for k, v in self.model.posterior_samples_cis.items()
224                             if k not in ['x_obs', 'y_obs']}
225
226            # Add cis gene metadata (including feature_meta from cis modality)
227            cis_mod = self.model.get_modality('cis')
228            posterior_with_meta = {
229                'posterior_samples': posterior_clean,
230                'cis_gene': self.model.cis_gene,
231                'modality_name': 'cis',  # Cis always uses 'cis' modality
232                'feature_meta': cis_mod.feature_meta.to_dict('records') if hasattr(cis_mod, 'feature_meta') and cis_mod.feature_meta is not None else None,
233                'cell_names': self.model.meta['cell'].tolist() if 'cell' in self.model.meta.columns else None,
234                'loss_x': self.model.loss_x if hasattr(self.model, 'loss_x') else None
235            }
236
237            path = os.path.join(output_dir, 'posterior_samples_cis.pt')
238            torch.save(posterior_with_meta, path)
239            saved_files['posterior_samples_cis'] = path
240            saved_summary.append(f'posterior_cis ({self.model.cis_gene})')
241            if verbose:
242                print(f"[SAVE] posterior_samples_cis (cis_gene: {self.model.cis_gene}) → {path}")
243
244        # Print summary
245        print(f"[SAVE] Cis fit to {output_dir}")
246        if saved_summary:
247            print(f"[SAVE] Saved: {', '.join(saved_summary)}")
248
249        return saved_files

252    def save_trans_fit(self, output_dir: str = None, modalities: list = None, verbose: bool = False):
253        """
254        Save fitted trans parameters from fit_trans().
255
256        Parameters
257        ----------
258        output_dir : str, optional
259            Directory to save to. If None, uses self.model.output_dir.
260        modalities : list of str, optional
261            List of modality names to save. If None, saves all modalities.
262            Example: ['gene', 'atac']
263        verbose : bool
264            If True, print detailed save information. Default False (summary only).
265
266        Returns
267        -------
268        dict
269            Paths to saved files
270
271        Notes
272        -----
273        Saves per-modality:
274        - posterior_samples_trans_{modality}.pt: Full posterior samples for each modality
275
276        Saves model-level (automatically when primary modality is included):
277        - posterior_samples_trans.pt: Model-level posterior samples
278        """
279        if output_dir is None:
280            output_dir = os.path.join(self.model.output_dir, self.model.label)
281
282        os.makedirs(output_dir, exist_ok=True)
283
284        saved_files = {}
285        saved_summary = []
286
287        # Determine which modalities to save
288        if modalities is None:
289            modalities_to_save = list(self.model.modalities.keys())
290        else:
291            # Validate requested modalities
292            invalid = set(modalities) - set(self.model.modalities.keys())
293            if invalid:
294                raise ValueError(f"Unknown modalities: {invalid}. Available: {list(self.model.modalities.keys())}")
295            modalities_to_save = modalities
296
297        # Automatically save model-level parameters if primary modality is included
298        should_save_model_level = self.model.primary_modality in modalities_to_save
299
300        # Save model-level posterior samples (when primary modality is being saved)
301        if should_save_model_level:
302            if hasattr(self.model, 'posterior_samples_trans') and self.model.posterior_samples_trans is not None:
303                # Remove large observation arrays and add metadata
304                posterior_clean = {k: v for k, v in self.model.posterior_samples_trans.items()
305                                 if k not in ['y_obs', 'x_obs']}
306
307                # Get modality for feature info
308                primary_mod = self.model.get_modality(self.model.primary_modality)
309
310                # Add modality and feature metadata (including full feature_meta DataFrame)
311                # Use the modality's authoritative feature count (not sample tensor shape[-1],
312                # which can pick a non-feature tensor such as alpha_y [S, C-1, T] whose
313                # last dim could coincidentally be 1 when C-1=1, saving wrong n_features).
314                n_features_primary = primary_mod.dims.get('n_features', None)
315                if n_features_primary is None:
316                    _sample_tensor = next((v for v in posterior_clean.values()
317                                           if isinstance(v, torch.Tensor) and v.ndim >= 2), None)
318                    n_features_primary = _sample_tensor.shape[-1] if _sample_tensor is not None else None
319                posterior_with_meta = {
320                    'posterior_samples': posterior_clean,
321                    'modality_name': self.model.primary_modality,
322                    'distribution': primary_mod.distribution,
323                    'feature_names': primary_mod.feature_names if hasattr(primary_mod, 'feature_names') else None,
324                    'n_features': n_features_primary,
325                    'feature_meta': primary_mod.feature_meta.to_dict('records') if hasattr(primary_mod, 'feature_meta') and primary_mod.feature_meta is not None else None,
326                    'cis_gene': self.model.cis_gene,
327                    'losses_trans': self.model.losses_trans if hasattr(self.model, 'losses_trans') else None,
328                    'trans_prior_params': self.model.trans_prior_params if hasattr(self.model, 'trans_prior_params') else None,
329                }
330
331                # Include modality name in filename to prevent overwrites
332                path = os.path.join(output_dir, f'posterior_samples_trans_{self.model.primary_modality}.pt')
333                torch.save(posterior_with_meta, path)
334                saved_files['posterior_samples_trans'] = path
335                if verbose:
336                    print(f"[SAVE] posterior_samples_trans (modality: {self.model.primary_modality}, {n_features_primary} features) → {path}")
337
338        # Save per-modality posterior samples
339        for mod_name in modalities_to_save:
340            mod = self.model.modalities[mod_name]
341            if hasattr(mod, 'posterior_samples_trans') and mod.posterior_samples_trans is not None:
342                posterior_clean = {k: v for k, v in mod.posterior_samples_trans.items()
343                                 if k not in ['y_obs', 'x_obs']}
344
345                # Use the modality's authoritative feature count (same reason as primary modality)
346                n_features = mod.dims.get('n_features', None)
347                if n_features is None:
348                    _sample_tensor = next((v for v in posterior_clean.values()
349                                           if isinstance(v, torch.Tensor) and v.ndim >= 2), None)
350                    n_features = _sample_tensor.shape[-1] if _sample_tensor is not None else None
351
352                # Add modality and feature metadata (including full feature_meta DataFrame)
353                posterior_with_meta = {
354                    'posterior_samples': posterior_clean,
355                    'modality_name': mod_name,
356                    'distribution': mod.distribution,
357                    'feature_names': mod.feature_names if hasattr(mod, 'feature_names') else None,
358                    'n_features': n_features,
359                    'feature_meta': mod.feature_meta.to_dict('records') if hasattr(mod, 'feature_meta') and mod.feature_meta is not None else None,
360                    'cis_gene': self.model.cis_gene,
361                    'losses_trans': mod.losses_trans if hasattr(mod, 'losses_trans') else None,
362                    'trans_prior_params': mod.trans_prior_params if hasattr(mod, 'trans_prior_params') else None,
363                }
364
365                path = os.path.join(output_dir, f'posterior_samples_trans_{mod_name}.pt')
366                torch.save(posterior_with_meta, path)
367                saved_files[f'posterior_samples_trans_{mod_name}'] = path
368                saved_summary.append(f"{mod_name}: posterior({n_features} features)")
369                if verbose:
370                    print(f"[SAVE] {mod_name}.posterior_samples_trans (distribution: {mod.distribution}, {n_features} features) → {path}")

373        print(f"[SAVE] Trans fit to {output_dir}")
374        if saved_summary:
375            print(f"[SAVE] Saved: {'; '.join(saved_summary)}")
376
377        return saved_files

332                # Include modality name in filename to prevent overwrites
333                path = os.path.join(output_dir, f'posterior_samples_trans_{self.model.primary_modality}.pt')
334                torch.save(posterior_with_meta, path)
335                saved_files['posterior_samples_trans'] = path
336                if verbose:
337                    print(f"[SAVE] posterior_samples_trans (modality: {self.model.primary_modality}, {n_features_primary} features) → {path}")
338
339        # Save per-modality posterior samples
340        for mod_name in modalities_to_save:
341            mod = self.model.modalities[mod_name]
342            if hasattr(mod, 'posterior_samples_trans') and mod.posterior_samples_trans is not None:
343                posterior_clean = {k: v for k, v in mod.posterior_samples_trans.items()
344                                 if k not in ['y_obs', 'x_obs']}
345
346                # Use the modality's authoritative feature count (same reason as primary modality)
347                n_features = mod.dims.get('n_features', None)
348                if n_features is None:
349                    _sample_tensor = next((v for v in posterior_clean.values()
350                                           if isinstance(v, torch.Tensor) and v.ndim >= 2), None)
351                    n_features = _sample_tensor.shape[-1] if _sample_tensor is not None else None
352
353                # Add modality and feature metadata (including full feature_meta DataFrame)
354                posterior_with_meta = {
355                    'posterior_samples': posterior_clean,
356                    'modality_name': mod_name,
357                    'distribution': mod.distribution,
358                    'feature_names': mod.feature_names if hasattr(mod, 'feature_names') else None,
359                    'n_features': n_features,
360                    'feature_meta': mod.feature_meta.to_dict('records') if hasattr(mod, 'feature_meta') and mod.feature_meta is not None else None,
361                    'cis_gene': self.model.cis_gene,
362                    'losses_trans': mod.losses_trans if hasattr(mod, 'losses_trans') else None,
363                    'trans_prior_params': mod.trans_prior_params if hasattr(mod, 'trans_prior_params') else None,
364                }
365
366                path = os.path.join(output_dir, f'posterior_samples_trans_{mod_name}.pt')
367                torch.save(posterior_with_meta, path)
368                saved_files[f'posterior_samples_trans_{mod_name}'] = path
369                saved_summary.append(f"{mod_name}: posterior({n_features} features)")
370                if verbose:
371                    print(f"[SAVE] {mod_name}.posterior_samples_trans (distribution: {mod.distribution}, {n_features} features) → {path}")
372
373        # Print summary
374        print(f"[SAVE] Trans fit to {output_dir}")
375        if saved_summary:
376            print(f"[SAVE] Saved: {'; '.join(saved_summary)}")
377
378        return saved_files
