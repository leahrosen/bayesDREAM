 17def scatter_with_smooth_by_group(
 18    x,
 19    y,
 20    group,
 21    frac=0.2,
 22    s=1,
 23    alpha=0.3,
 24    xlabel=None,
 25    ylabel=None,
 26    title=None,
 27    ax=None,
 28    show=True,
 29):
 30    """
 31    Scatter plot with per-group lowess smoothed lines, drawn on the current axes.
 32
 33    Parameters
 34    ----------
 35    x : array-like or torch.Tensor
 36        X-axis values.
 37    y : array-like or torch.Tensor
 38        Y-axis values.
 39    group : array-like
 40        Group labels; one smoothed line is drawn per unique value.
 41    frac : float, default 0.2
 42        Lowess smoothing fraction.
 43    s : float, default 1
 44        Scatter marker size.
 45    alpha : float, default 0.3
 46        Scatter marker transparency.
 47    xlabel : str, optional
 48        X-axis label.
 49    ylabel : str, optional
 50        Y-axis label.
 51    title : str, optional
 52        Plot title.
 53    ax : matplotlib axes, optional
 54        Axes to plot on. If None, uses current axes.
 55    show : bool, default True
 56        Whether to call ``plt.show()``.
 57
 58    Returns
 59    -------
 60    ax : matplotlib axes
 61    """
 62    if hasattr(x, "detach"):
 63        x = x.detach().cpu().numpy()
 64    else:
 65        x = np.asarray(x)
 66
 67    if hasattr(y, "detach"):
 68        y = y.detach().cpu().numpy()
 69    else:
 70        y = np.asarray(y)
 71
 72    group = np.asarray(group)
 73
 74    x = np.ravel(x)
 75    y = np.ravel(y)
 76    group = np.ravel(group)
 77
 78    mask = np.isfinite(x) & np.isfinite(y)
 79    x = x[mask]
 80    y = y[mask]
 81    group = group[mask]
 82
 83    if ax is None:
 84        ax = plt.gca()
 85
 86    for g in np.unique(group):
 87        idx = group == g
 88        x_g = x[idx]
 89        y_g = y[idx]
 90
 91        order = np.argsort(x_g)
 92        x_g = x_g[order]
 93        y_g = y_g[order]
 94
 95        ax.scatter(x_g, y_g, s=s, alpha=alpha, label=str(g))
 96        smoothed = lowess(y_g, x_g, frac=frac, return_sorted=True)
 97        (line,) = ax.plot(smoothed[:, 0], smoothed[:, 1], linewidth=2.5, zorder=10)
 98        line.set_path_effects([
 99            pe.Stroke(linewidth=4, foreground="white"),
100            pe.Normal(),
101        ])
102
103    if xlabel is not None:
104        ax.set_xlabel(xlabel)
105    if ylabel is not None:
106        ax.set_ylabel(ylabel)
107    if title is not None:
108        ax.set_title(title)
109
110    ax.legend()
111    if show:
112        plt.show()
113    return ax

116def plot_x_true_residuals_vs_sumfactor(
117    model,
118    group_col,
119    facet_col,
120    sum_factor_col="sum_factor",
121    frac=0.2,
122    s=1,
123    alpha=0.3,
124    figsize=(12, 5),
125    min_cells_for_smooth=10,
126    show=True,
127):
128    """
129    Diagnostic: log2(sum_factor) vs normalised log2_x_true, faceted by a
130    user-specified column, with a lowess line per group.
131
132    For each cell assigned to guide g, the normalised x_true is::
133
134        (log2_x_true_cell - log2_x_eff_g_guide) / sigma_eff_guide
135
136    where log2_x_eff_g and sigma_eff are posterior-mean guide-level parameters
137    from the cis fit.  Under the model this should be N(0, 1) with no dependence
138    on sum_factor; a systematic trend reveals residual confounding between
139    sequencing depth and inferred expression.
140
141    Only single-guide cells are included (all cells in low-MOI; a subset in
142    high-MOI mode).
143
144    Parameters
145    ----------
146    model : bayesDREAM
147        A fitted model (fit_cis must have been called).
148    group_col : str
149        Column in model.meta to colour and smooth separately (e.g. ``'lane'``,
150        ``'batch'``).
151    facet_col : str
152        Column in model.meta to facet by; one panel per unique value
153        (e.g. ``'target'``, ``'cell_line'``).
154    sum_factor_col : str, default ``'sum_factor'``
155        Column in the primary modality's sum_factors to use as y-axis.
156    frac : float, default 0.2
157        Lowess smoothing fraction.
158    s : float, default 1
159        Scatter marker size.
160    alpha : float, default 0.3
161        Scatter marker transparency.
162    figsize : tuple, default (12, 5)
163        Figure size.
164    min_cells_for_smooth : int, default 10
165        Minimum cells in a (facet × group) slice to draw a smoothed line.
166    show : bool, default True
167        Whether to call ``plt.show()``.
168
169    Returns
170    -------
171    fig : matplotlib.figure.Figure
172    """
173    import torch
174
175    if not hasattr(model, "posterior_samples_cis") or model.posterior_samples_cis is None:
176        raise RuntimeError("fit_cis() must be called before this plot.")
177
178    # --- posterior-mean guide-level quantities ---
179    x_eff_g_mean = model.posterior_samples_cis["x_eff_g"].mean(dim=0)     # [G]
180    sigma_eff_mean = model.posterior_samples_cis["sigma_eff"].mean(dim=0)  # [G]
181
182    log2_x_eff_g_np = torch.log2(x_eff_g_mean.clamp(min=1e-12)).cpu().numpy()
183    sigma_eff_np = sigma_eff_mean.cpu().numpy()
184
185    # --- cell-level arrays (positionally aligned with model.meta rows) ---
186    meta = model.meta
187    cell_col = meta["cell"].values  # cell barcodes for sum-factor lookup
188
189    log2_x_true = model.log2_x_true
190    if hasattr(log2_x_true, "detach"):
191        log2_x_true = log2_x_true.detach().cpu().numpy()
192    else:
193        log2_x_true = np.asarray(log2_x_true, dtype=float)
194
195    sf_series = model.get_modality(model.primary_modality).sum_factors[sum_factor_col]
196    sf_values = sf_series.loc[cell_col].values.astype(float)
197    log2_sf = np.log2(sf_values)
198
199    groups = meta[group_col].values
200    facets_raw = meta[facet_col].values
201
202    # --- single-guide mask + guide index per cell ---
203    is_high_moi = getattr(model, "is_high_moi", False)
204    if is_high_moi:
205        ga = model.guide_assignment_tensor  # [N, G]
206        n_guides_per_cell = ga.sum(dim=1).cpu().numpy()
207        single_mask = n_guides_per_cell == 1
208        guide_idx = ga.argmax(dim=1).cpu().numpy()
209    else:
210        single_mask = np.ones(len(meta), dtype=bool)
211        guide_idx = meta["guide_code"].values.astype(int)
212
213    # --- normalised residual ---
214    log2_x_eff_per_cell = log2_x_eff_g_np[guide_idx]
215    sigma_per_cell = sigma_eff_np[guide_idx]
216    normalized = (log2_x_true - log2_x_eff_per_cell) / np.maximum(sigma_per_cell, 1e-6)
217
218    # apply single-guide mask
219    sel = single_mask
220    normalized = normalized[sel]
221    log2_sf = log2_sf[sel]
222    groups = groups[sel]
223    facets_raw = facets_raw[sel]
224    n_single = int(sel.sum())
225
226    # --- facet panels ---
227    facet_values = np.unique(facets_raw)
228    n_facets = len(facet_values)
229
230    all_groups = np.unique(groups)
231    cmap = plt.cm.tab10(np.linspace(0, 1, max(len(all_groups), 1)))
232    color_map = {g: cmap[i] for i, g in enumerate(all_groups)}
233
234    fig, axes = plt.subplots(1, n_facets, figsize=figsize, sharey=True)
235    if n_facets == 1:
236        axes = [axes]
237
238    for ax, fval in zip(axes, facet_values):
239        facet_mask = facets_raw == fval
240        n_facet = int(facet_mask.sum())
241
242        for g in all_groups:
243            g_mask = facet_mask & (groups == g)
244            if g_mask.sum() < 2:
245                continue
246
247            x_g = normalized[g_mask]
248            y_g = log2_sf[g_mask]
249            order = np.argsort(x_g)
250            x_g = x_g[order]
251            y_g = y_g[order]
252
253            color = color_map[g]
254            ax.scatter(x_g, y_g, s=s, alpha=alpha, color=color, label=str(g), rasterized=True)
255
256            if g_mask.sum() >= min_cells_for_smooth:
257                smoothed = lowess(y_g, x_g, frac=frac, return_sorted=True)
258                (line,) = ax.plot(
259                    smoothed[:, 0], smoothed[:, 1],
260                    linewidth=2.5, color=color, zorder=10,
261                )
262                line.set_path_effects([
263                    pe.Stroke(linewidth=4, foreground="white"),
264                    pe.Normal(),
265                ])
266
267        ax.axvline(0, color="gray", linewidth=1, linestyle="--", zorder=5, alpha=0.7)
268        ax.set_xlabel(
269            "Normalised log2_x_true\n"
270            r"$(log_2 x_{true} - log_2 x_{eff,g}) \,/\, \sigma_{eff,g}$"
271        )
272        ax.set_title(f"{facet_col}={fval}  (n={n_facet:,})")
273        ax.legend(title=group_col, markerscale=4, fontsize=7)
274
275    axes[0].set_ylabel(f"log2({sum_factor_col})")
276    fig.suptitle(
277        f"log2({sum_factor_col}) vs normalised log2_x_true\n"
278        f"single-guide cells only  (n={n_single:,})"
279    )
280    plt.tight_layout()
281    if show:
282        plt.show()
283    return fig

286def plot_sum_factor_comparison(model, sf_col1, sf_col2, cis_gene=None,
287                               color_scheme=None, show=True):
288    """
289    Plot pairwise comparison of sum factors (e.g., original vs adjusted).
290
291    Parameters
292    ----------
293    model : bayesDREAM
294        Fitted bayesDREAM model.
295    sf_col1 : str
296        First sum factor column name (checked in model.meta then primary modality
297        sum_factors).
298    sf_col2 : str
299        Second sum factor column name (checked in model.meta then primary modality
300        sum_factors).
301    cis_gene : str, optional
302        Cis gene name for plot title. Defaults to ``model.cis_gene``.
303    color_scheme : ColorScheme, optional
304        Custom color scheme. Defaults to a plain ``ColorScheme()``.
305    show : bool, default True
306        Whether to call ``plt.show()``.
307
308    Returns
309    -------
310    fig : matplotlib figure
311    """
312    if color_scheme is None:
313        color_scheme = ColorScheme()
314
315    if cis_gene is None:
316        cis_gene = getattr(model, 'cis_gene', 'cis')
317
318    fig, ax = plt.subplots(figsize=(5, 4))
319
320    df = model.meta[['cell', 'guide']].copy() if 'guide' in model.meta.columns \
321        else model.meta[['cell']].copy()
322
323    # Resolve each sum factor column: check model.meta first, then primary modality
324    def _resolve_sf_col(col):
325        if col in model.meta.columns:
326            return model.meta.set_index('cell')[col]
327        primary_mod = model.get_modality(model.primary_modality)
328        if (primary_mod.sum_factors is not None
329                and col in primary_mod.sum_factors.columns):
330            return primary_mod.sum_factors[col]
331        return None
332
333    s1 = _resolve_sf_col(sf_col1)
334    s2 = _resolve_sf_col(sf_col2)
335
336    if s1 is None or s2 is None:
337        missing = [c for c, s in [(sf_col1, s1), (sf_col2, s2)] if s is None]
338        primary_mod = model.get_modality(model.primary_modality)
339        available_meta = list(model.meta.columns)
340        available_mod  = (list(primary_mod.sum_factors.columns)
341                          if primary_mod.sum_factors is not None else [])
342        print(f"Missing sum factor column(s): {missing}. "
343              f"Available in meta: {available_meta}. "
344              f"Available in modality sum_factors: {available_mod}")
345        return fig
346
347    df[sf_col1] = s1.loc[model.meta['cell'].values].values
348    df[sf_col2] = s2.loc[model.meta['cell'].values].values
349
350    # Filter to positive values
351    df = df[(df[sf_col1] > 0) & (df[sf_col2] > 0)]
352
353    # Plot by guide
354    for guide, sub in df.groupby('guide'):
355        color = color_scheme.get_guide_color(guide, 'black')
356        ax.scatter(
357            sub[sf_col1],
358            sub[sf_col2],
359            s=12,
360            alpha=0.8,
361            color=color,
362            label=guide
363        )
364
365    # Identity line
366    all_sf = np.concatenate([df[sf_col1].values, df[sf_col2].values])
367    sf_min, sf_max = all_sf.min(), all_sf.max()
368    ax.plot([sf_min, sf_max], [sf_min, sf_max], 'k--', linewidth=1, alpha=0.6)
369
370    ax.set_xlabel(sf_col1, fontsize=10)
371    ax.set_ylabel(sf_col2, fontsize=10)
372    ax.set_title(f'{cis_gene}: sum factor comparison', fontsize=11)
373    ax.legend(fontsize=8, markerscale=1.2, frameon=False)
374    ax.grid(True, linewidth=0.5, alpha=0.3)
375
376    plt.tight_layout()
377
378    if show:
379        plt.show()
380
381    return fig

401def _detect_group_col(res, group_col):
402    """
403    Resolve the grouping column for check_systematic_shift() results.
404
405    If ``group_col`` is given, returns it unchanged. Otherwise prefers
406    ``'technical_group_code'`` (check_systematic_shift's default ``tech_col``
407    name) if present; else falls back to the single non-boolean column in
408    ``res`` that isn't one of the standard result columns, raising
409    ``ValueError`` if that's ambiguous (e.g. user-added derived columns).
410    """
411    if group_col is not None:
412        return group_col
413    if "technical_group_code" in res.columns:
414        return "technical_group_code"
415    extra = [
416        c for c in res.columns
417        if c not in _SHIFT_RESULT_COLS and res[c].dtype != bool
418    ]
419    if len(extra) != 1:
420        raise ValueError(
421            f"Could not auto-detect group_col (candidates: {extra}). "
422            "Pass group_col explicitly."
423        )
424    return extra[0]

427def _declutter_texts(fig, ax, texts, iterations=200, pad_px=1.5):
428    """
429    Iteratively nudge overlapping ``Text`` labels apart in display-pixel
430    space, converting shifts back to data coordinates for repositioning.
431
432    Lightweight fallback for when ``adjustText`` isn't installed: not as
433    good, but resolves direct overlaps within a panel.
434    """
435    if len(texts) < 2:
436        return
437    fig.canvas.draw()
438    renderer = fig.canvas.get_renderer()
439
440    x0, x1 = ax.get_xlim()
441    y0, y1 = ax.get_ylim()
442    bbox = ax.get_window_extent(renderer=renderer)
443    px_per_xdata = bbox.width / (x1 - x0)
444    px_per_ydata = bbox.height / (y1 - y0)
445    if px_per_xdata <= 0 or px_per_ydata <= 0:
446        return
447
448    for _ in range(iterations):
449        boxes = [t.get_window_extent(renderer=renderer) for t in texts]
450        moved = False
451        for i in range(len(texts)):
452            for j in range(i + 1, len(texts)):
453                bi, bj = boxes[i], boxes[j]
454                if not bi.overlaps(bj):
455                    continue
456                moved = True
457                overlap_x = min(bi.x1, bj.x1) - max(bi.x0, bj.x0)
458                overlap_y = min(bi.y1, bj.y1) - max(bi.y0, bj.y0)
459                cix, cjx = (bi.x0 + bi.x1) / 2, (bj.x0 + bj.x1) / 2
460                ciy, cjy = (bi.y0 + bi.y1) / 2, (bj.y0 + bj.y1) / 2
461                xi, yi = texts[i].get_position()
462                xj, yj = texts[j].get_position()
463                if overlap_x < overlap_y:
464                    shift = (overlap_x / 2 + pad_px) * (1.0 if cix >= cjx else -1.0)
465                    dx = shift / px_per_xdata
466                    texts[i].set_position((xi + dx, yi))
467                    texts[j].set_position((xj - dx, yj))
468                else:
469                    shift = (overlap_y / 2 + pad_px) * (1.0 if ciy >= cjy else -1.0)
470                    dy = shift / px_per_ydata
471                    texts[i].set_position((xi, yi + dy))
472                    texts[j].set_position((xj, yj - dy))
473        if not moved:
474            break
475        fig.canvas.draw()
476        renderer = fig.canvas.get_renderer()

479def _draw_leader_lines(ax, texts, orig_xy, min_px=3.0, color="gray", lw=0.5):
480    """Draw a thin line from each label's original point to its (possibly
481    decluttered) final position, when it moved more than ``min_px`` pixels."""
482    fig = ax.figure
483    fig.canvas.draw()
484    for txt, (x0, y0) in zip(texts, orig_xy):
485        x1, y1 = txt.get_position()
486        d0 = ax.transData.transform((x0, y0))
487        d1 = ax.transData.transform((x1, y1))
488        if np.hypot(*(d1 - d0)) > min_px:
489            ax.plot([x0, x1], [y0, y1], color=color, lw=lw, zorder=3.5)

492def plot_systematic_shift_volcano(
493    res,
494    group_col=None,
495    p_col="p_adj",
496    effect_col="shift_est",
497    label_col="feature",
498    fdr_threshold=0.1,
499    top_n=20,
500    sig_color="firebrick",
501    nonsig_color="#999999",
502    label_fontsize=7,
503    figsize=None,
504    show=True,
505):
506    """
507    Volcano plot of ``check_systematic_shift()`` results, faceted by group.
508
509    One panel per unique value of ``group_col`` (the grouping column produced
510    by ``check_systematic_shift(tech_col=...)`` -- named after whatever
511    ``tech_col`` was passed, e.g. ``'technical_group_code'`` by default).
512    Each panel plots ``effect_col`` (x) against ``-log10(p_col)`` (y),
513    highlights points below ``fdr_threshold``, and labels the ``top_n`` most
514    significant features per panel.
515
516    Parameters
517    ----------
518    res : pd.DataFrame
519        Output of ``model.check_systematic_shift()``.
520    group_col : str, optional
521        Column to facet by. If None: uses ``'technical_group_code'`` if present
522        (``check_systematic_shift``'s default ``tech_col`` name), otherwise
523        auto-detected as the single non-boolean column in ``res`` that isn't
524        one of the standard result columns (raises ``ValueError`` if that's
525        ambiguous, e.g. if you added your own derived columns to ``res`` --
526        pass it explicitly in that case).
527    p_col : str, default ``'p_adj'``
528        Which BH-corrected p-value column to plot: ``'p_adj'`` (likelihood-ratio
529        test) or ``'shift_p_adj'`` (Wald test on the ``targeted`` coefficient).
530        See ``check_systematic_shift``'s docstring for the difference.
531    effect_col : str, default ``'shift_est'``
532        Column for the x-axis (effect size).
533    label_col : str, default ``'feature'``
534        Column supplying the text used to label top hits.
535    fdr_threshold : float, default 0.1
536        Significance cutoff; also drawn as a horizontal dashed line.
537    top_n : int, default 20
538        Number of most-significant (lowest ``p_col``) features to label,
539        per panel, among rows below ``fdr_threshold``.
540    sig_color, nonsig_color : str
541        Colors for significant / non-significant points.
542    label_fontsize : int, default 7
543        Font size for point labels.
544    figsize : tuple, optional
545        Figure size. Defaults to ``(5 * n_panels, 5)``.
546    show : bool, default True
547        Whether to call ``plt.show()``.
548
549    Returns
550    -------
551    fig : matplotlib.figure.Figure
552
553    Raises
554    ------
555    ValueError
556        If ``group_col`` is None and cannot be unambiguously auto-detected
557        (see ``group_col`` above), or if no rows have ``ok == True`` with
558        finite ``effect_col``/``p_col`` values to plot.
559
560    Notes
561    -----
562    Only rows with ``ok == True`` and finite ``effect_col``/``p_col`` are
563    plotted; skipped (``ok == False``) rows are silently dropped.
564
565    If the `adjustText <https://github.com/Phlya/adjustText>`_ package is
566    installed, labels are auto-repelled using it (best quality). Otherwise a
567    built-in pixel-space decluttering pass nudges overlapping labels apart
568    and draws thin leader lines back to their points.
569    """
570    group_col = _detect_group_col(res, group_col)
571
572    df = res[res["ok"] & np.isfinite(res[effect_col]) & np.isfinite(res[p_col])].copy()
573    if df.empty:
574        raise ValueError("No rows with ok=True and finite values to plot.")
575
576    # Avoid -log10(0) = inf for exact-zero p-values
577    tiny = np.finfo(float).tiny
578    df["_neg_log10_p"] = -np.log10(df[p_col].clip(lower=tiny))
579    df["_sig"] = df[p_col] < fdr_threshold
580
581    group_values = sorted(df[group_col].unique())
582    n_panels = len(group_values)
583    if figsize is None:
584        figsize = (5 * n_panels, 5)
585
586    fig, axes = plt.subplots(1, n_panels, figsize=figsize, sharex=True, sharey=True)
587    if n_panels == 1:
588        axes = [axes]
589
590    hline_y = -np.log10(fdr_threshold)
591
592    for ax, gval in zip(axes, group_values):
593        sub = df[df[group_col] == gval]
594        nonsig = sub[~sub["_sig"]]
595        sig = sub[sub["_sig"]]
596

956    model.meta[window_col] = model.meta["cell"].isin(all_window_cells)
957
958    figs = {}

962            subset_meta[window_col] = True
963            figs[feature] = model.plot_xy_data(feature, subset_meta=subset_meta, **plot_xy_kwargs)
964    finally:
965        if had_col:
966            model.meta[window_col] = prev_values
967        else:
968            del model.meta[window_col]
969
970    return figs
