  65# Base column names (before the _median/_lower/_upper/_log2fc suffix) that are
  66# specific to one Hill component of an additive_hill fit -- see the "For
  67# additive_hill" column list in ModelSummarizer.export_trans_summary's
  68# docstring (io/summary.py). A feature classified 'single_positive'/
  69# 'single_negative' has only ONE of these two components active (the other's

  80def _component_for_value_col(value_col):
  81    """Return 'a'/'b' if value_col is a component-specific additive_hill
  82    parameter (see _COMPONENT_*_BASE_COLS), else None."""
  83    for base in _COMPONENT_A_BASE_COLS:
  84        if value_col == base or value_col.startswith(base + '_'):
  85            return 'a'
  86    for base in _COMPONENT_B_BASE_COLS:
  87        if value_col == base or value_col.startswith(base + '_'):
  88            return 'b'
  89    return None

  92def _prepare_trans_group_data(fdr_df, gene_col, classification_col, dependent_col,
  93                                color_by, require_dependent, colors):
  94    """
  95    Shared setup for plot_trans_hits_by_gene / plot_trans_values_by_gene:
  96    validate columns, resolve the direction-vs-classification grouping, and
  97    apply the is_dependent / classification filters.
  98
  99    Returns
 100    -------
 101    d : pd.DataFrame
 102        Filtered rows (subset of fdr_df, index preserved).
 103    group : pd.Series
 104        Per-row group label (direction or classification), aligned to d.index.
 105    group_order : list of str
 106    color_map : dict
 107    legend_title : str
 108    """
 109    if classification_col not in fdr_df.columns:
 110        raise ValueError(
 111            f"'{classification_col}' not found in fdr_df. This column is only "
 112            "populated for function_type='additive_hill' trans fits."
 113        )
 114    if gene_col not in fdr_df.columns:
 115        raise ValueError(f"'{gene_col}' not found in fdr_df; pass gene_col= explicitly.")
 116
 117    if color_by == "direction":
 118        group_map = _TRANS_CLASSIFICATION_TO_DIRECTION
 119        group_order = _TRANS_DIRECTION_ORDER
 120        color_map = dict(_TRANS_DIRECTION_COLORS)
 121        legend_title = "Direction"
 122    elif color_by == "classification":
 123        group_map = {c: c for c in _TRANS_CLASSIFICATION_ORDER}
 124        group_order = _TRANS_CLASSIFICATION_ORDER
 125        color_map = dict(_TRANS_CLASSIFICATION_COLORS)
 126        legend_title = "Classification"
 127    else:
 128        raise ValueError(f"color_by must be 'direction' or 'classification', got {color_by!r}")
 129    if colors:
 130        color_map.update(colors)
 131
 132    d = fdr_df
 133    if require_dependent:
 134        if dependent_col not in d.columns:
 135            raise ValueError(
 136                f"'{dependent_col}' not found in fdr_df; pass require_dependent=False "
 137                "to skip the significance filter."
 138            )
 139        d = d[d[dependent_col] == True]  # noqa: E712 -- may be object/NaN dtype
 140
 141    group = d[classification_col].map(group_map)
 142    d = d.loc[group.notna() & d[gene_col].notna()]
 143    group = group.loc[d.index]
 144
 145    return d, group, group_order, color_map, legend_title

 148def plot_trans_hits_by_gene(
 149    fdr_df,
 150    gene_col="gene",
 151    classification_col="classification",
 152    dependent_col="is_dependent",
 153    color_by="direction",
 154    require_dependent=True,
 155    top_n=None,
 156    colors=None,
 157    cis_gene=None,
 158    figsize=None,
 159    show=True,
 160):
 161    """
 162    Stacked bar chart of the number of trans-dependent features per gene,
 163    colored by shape of the dose-response curve.
 164
 165    Meant for feature-level modalities where several features can map to the
 166    same gene (e.g. ``splicing_sj``/``splicing_donor`` -- several SJs per
 167    gene, or ``transcript`` -- several transcripts per gene). Groups
 168    ``fdr_df`` by ``gene_col`` and draws one stacked bar per gene.
 169
 170    Parameters
 171    ----------
 172    fdr_df : pd.DataFrame
 173        Trans summary dataframe, e.g. the output of
 174        ``model.export_trans_summary()`` / ``save_trans_summary``. Must have
 175        ``classification_col`` (only produced for ``function_type='additive_hill'``)
 176        and ``gene_col``.
 177    gene_col : str, default ``'gene'``
 178        Column giving the gene each row (feature) belongs to. For splicing
 179        modalities this is aliased from ``gene_for_denominator`` by the
 180        splicing loader (see ``docs/SPLICING_LOADER_GUIDE.md``).
 181    classification_col : str, default ``'classification'``
 182        Column with the per-feature shape classification (``'single_positive'``,
 183        ``'additive_negative'``, ``'non_monotonic_min'``, ``'flat'``, ...).
 184    dependent_col : str, default ``'is_dependent'``
 185        Column flagging FDR-significant, non-degenerate dose-response features.
 186    color_by : {'direction', 'classification'}, default ``'direction'``
 187        ``'direction'`` collapses ``classification_col`` into 3 groups --
 188        ``positive`` (``single_positive``/``additive_positive``), ``negative``
 189        (``single_negative``/``additive_negative``), ``non_monotonic``
 190        (``non_monotonic_min``/``non_monotonic_max``).
 191        ``'classification'`` keeps all 6 non-``'flat'`` values distinct, so
 192        single-Hill and additive-Hill features are shown separately within
 193        each direction.
 194    require_dependent : bool, default True
 195        If True, restrict to rows with ``dependent_col == True`` before
 196        counting (dropping ``'flat'``/non-significant rows in addition to the
 197        ``classification_col`` mapping already excluding ``'flat'``).
 198    top_n : int, optional
 199        Only show the ``top_n`` genes with the most trans-dependent features
 200        (sorted descending by total). Default: show all genes.
 201    colors : dict, optional
 202        Override the default color mapping. Keys are ``'positive'``,
 203        ``'negative'``, ``'non_monotonic'`` when ``color_by='direction'``, or
 204        the classification values when ``color_by='classification'``.
 205    cis_gene : str, optional
 206        Name of the cis gene these trans features were fit against (e.g.
 207        ``model.cis_gene``). Used only to make the y-axis label explicit
 208        about what "dependent" means here -- every row in a trans summary is
 209        a feature whose response was modeled as a function of this one cis
 210        gene, so a bar's height is "number of features dependent on
 211        ``cis_gene``", not some property intrinsic to the x-axis gene itself.
 212        If omitted, the label falls back to generic wording.
 213    figsize : tuple, optional
 214        Figure size. Defaults to ``(max(6, 0.4 * n_genes), 5)``.
 215    show : bool, default True
 216        Whether to call ``plt.show()``.
 217
 218    Returns
 219    -------
 220    fig : matplotlib.figure.Figure
 221
 222    Raises
 223    ------
 224    ValueError
 225        If ``classification_col`` or ``gene_col`` is missing from ``fdr_df``
 226        (e.g. ``fdr_df`` was exported with ``function_type='polynomial'``,
 227        which has no Hill components to classify), if ``color_by`` is not one
 228        of ``'direction'``/``'classification'``, or if no rows remain after
 229        filtering.
 230    """
 231    d, group, group_order, color_map, legend_title = _prepare_trans_group_data(
 232        fdr_df, gene_col, classification_col, dependent_col,
 233        color_by, require_dependent, colors
 234    )
 235
 236    if len(d) == 0:
 237        raise ValueError("No rows left to plot after filtering (require_dependent / classification).")
 238
 239    counts = (
 240        pd.crosstab(d[gene_col], group)
 241        .reindex(columns=group_order, fill_value=0)
 242    )
 243    counts = counts.loc[counts.sum(axis=1).sort_values(ascending=False).index]
 244    if top_n is not None:
 245        counts = counts.iloc[:top_n]
 246
 247    if figsize is None:
 248        figsize = (max(6, 0.4 * len(counts)), 5)
 249    fig, ax = plt.subplots(figsize=figsize)
 250
 251    x = np.arange(len(counts))
 252    bottom = np.zeros(len(counts))
 253    for group_name in group_order:
 254        vals = counts[group_name].values
 255        ax.bar(x, vals, bottom=bottom, width=0.7, color=color_map[group_name],
 256               label=group_name.replace('_', ' '), edgecolor='white', linewidth=1)
 257        bottom += vals
 258
 259    ax.set_xticks(x)
 260    ax.set_xticklabels(counts.index, rotation=90, fontsize=8)
 261    ylabel = (f"Number of features dependent on {cis_gene}" if cis_gene
 262              else "Number of features dependent on cis gene")
 263    ax.set_ylabel(ylabel)
 264    ax.set_xlabel(gene_col)
 265    ax.spines['top'].set_visible(False)
 266    ax.spines['right'].set_visible(False)
 267    ax.legend(frameon=False, title=legend_title)
 268    fig.tight_layout()
 269
 270    if show:
 271        plt.show()
 272
 273    return fig

 276def plot_trans_values_by_gene(
 277    fdr_df,
 278    value_col,
 279    gene_col="gene",
 280    classification_col="classification",
 281    dependent_col="is_dependent",
 282    color_by="direction",
 283    require_dependent=True,
 284    top_n=None,
 285    gene_order=None,
 286    colors=None,
 287    exclude_inactive_component=True,
 288    active_col="which_active",
 289    jitter=0.3,
 290    point_size=20,
 291    alpha=0.8,
 292    show_zero_line=True,
 293    show_gene_bands=True,
 294    gene_band_color="#000000",
 295    gene_band_alpha=0.06,
 296    figsize=None,
 297    show=True,
 298):
 299    """
 300    Dot (strip) plot of a per-feature value column, grouped by gene along the
 301    x-axis and colored by direction/classification.
 302
 303    Same gene grouping and direction/classification coloring as
 304    ``plot_trans_hits_by_gene``, but instead of counting hits, plots one point
 305    per feature at its actual value of ``value_col`` -- e.g.
 306    ``'observed_delta_p_median'``, ``'full_delta_p_median'``, ``'n_a_median'``,
 307    ``'n_b_median'``, ``'EC50_a_median'``, ``'EC50_b_median'``.
 308
 309    Parameters
 310    ----------
 311    fdr_df : pd.DataFrame
 312        Trans summary dataframe, e.g. the output of
 313        ``model.export_trans_summary()`` / ``save_trans_summary``. Must have
 314        ``classification_col``, ``gene_col``, and ``value_col``.
 315    value_col : str
 316        Column in ``fdr_df`` to plot on the y-axis. Rows with a missing/NaN
 317        value are dropped.
 318    gene_col, classification_col, dependent_col, color_by, require_dependent, colors :
 319        See ``plot_trans_hits_by_gene``.
 320    exclude_inactive_component : bool, default True
 321        ``value_col`` columns specific to one Hill component (``alpha``,
 322        ``Vmax_a``/``Vmax_b``, ``K_a``/``K_b``, ``EC50_a``/``EC50_b``,
 323        ``n_a``/``n_b``, ``inflection_a``/``inflection_b``, and their
 324        ``_median``/``_lower``/``_upper``/``_log2fc`` variants) are meaningless
 325        for a feature whose ``which_active`` doesn't include that component --
 326        e.g. ``n_a_median`` for a feature classified ``single_negative`` via
 327        component B, where component A's alpha absorbed a constant offset
 328        while ``n_a`` is completely unidentified. When True (default) and
 329        ``value_col`` is recognized as component-specific, rows where the
 330        corresponding component isn't active (per ``active_col``) are dropped
 331        before plotting. Has no effect on component-agnostic columns (e.g.
 332        ``observed_delta_p_median``, ``A_median``, ``full_delta_p_median``).
 333    active_col : str, default ``'which_active'``
 334        Column naming which component(s) are active per feature (``'a'``,
 335        ``'b'``, or ``'both'``). Only consulted when
 336        ``exclude_inactive_component=True`` and ``value_col`` is
 337        component-specific; if missing from ``fdr_df`` in that case, a warning
 338        is issued and the extra filtering is skipped.
 339    top_n : int, optional
 340        Only show the ``top_n`` genes with the most dependent features
 341        (sorted descending by count, same criterion as
 342        ``plot_trans_hits_by_gene``). Ignored if ``gene_order`` is given.
 343    gene_order : list of str, optional
 344        Explicit x-axis gene order -- e.g. reuse ``counts.index.tolist()``
 345        from a prior ``plot_trans_hits_by_gene()`` call so the two plots line
 346        up side by side. Genes not in ``gene_order`` are dropped from the data;
 347        genes in ``gene_order`` with no surviving rows appear as empty columns.
 348    jitter : float, default 0.3
 349        Half-width of horizontal jitter applied within each gene's column
 350        (uniform, seeded for reproducibility) to separate overlapping points.
 351    point_size : float, default 20
 352        Marker size (``ax.scatter``'s ``s``).
 353    alpha : float, default 0.8
 354        Marker transparency.
 355    show_zero_line : bool, default True
 356        Draw a grey dotted horizontal line at y=0. Meaningful for signed
 357        columns (``*_delta_p_*``, ``n_a``/``n_b``, log2fc columns); harmless
 358        otherwise -- set False if it's not a useful reference for your column.
 359    show_gene_bands : bool, default True
 360        Shade every other gene's column with a faint background band (zebra
 361        striping), so it's clear which gene a jittered point belongs to even
 362        with many genes packed tightly on the x-axis. Purely visual --
 363        doesn't affect ``gene_order``/alternation, just alternates starting
 364        from the first gene.
 365    gene_band_color : str, default ``'#000000'``
 366        Fill color for the shaded bands (only visible via ``gene_band_alpha``).
 367    gene_band_alpha : float, default 0.06
 368        Opacity of the shaded bands. Keep low -- these are a background
 369        reference, not data ink.
 370    figsize : tuple, optional
 371        Figure size. Defaults to ``(max(6, 0.4 * n_genes), 5)``.
 372    show : bool, default True
 373        Whether to call ``plt.show()``.
 374
 375    Returns
 376    -------
 377    fig : matplotlib.figure.Figure
 378
 379    Raises
 380    ------
 381    ValueError
 382        If ``classification_col``/``gene_col``/``value_col`` is missing from
 383        ``fdr_df``, if ``color_by`` is invalid, or if no rows remain after
 384        filtering.
 385    """
 386    if value_col not in fdr_df.columns:
 387        raise ValueError(f"'{value_col}' not found in fdr_df.")
 388
 389    d, group, group_order, color_map, legend_title = _prepare_trans_group_data(
 390        fdr_df, gene_col, classification_col, dependent_col,
 391        color_by, require_dependent, colors
 392    )
 393
 394    if exclude_inactive_component:
 395        component = _component_for_value_col(value_col)
 396        if component is not None:
 397            if active_col in d.columns:
 398                d = d.loc[d[active_col].isin([component, 'both'])]
 399                group = group.loc[d.index]
 400            else:
 401                warnings.warn(
 402                    f"'{value_col}' looks like a component-{component} parameter, but "
 403                    f"'{active_col}' is not in fdr_df -- cannot exclude rows where "
 404                    "that component is inactive (values may include degenerate/"
 405                    "meaningless fits for the inactive component). Add "
 406                    f"'{active_col}' (from save_trans_summary) or pass "
 407                    "exclude_inactive_component=False to silence this warning."
 408                )
 409
 410    d = d.loc[d[value_col].notna()]
 411    group = group.loc[d.index]
 412
 413    if len(d) == 0:
 414        raise ValueError("No rows left to plot after filtering (require_dependent / classification / value_col).")
 415
 416    if gene_order is None:
 417        gene_order_list = (
 418            d[gene_col].value_counts().sort_values(ascending=False).index.tolist()
 419        )
 420        if top_n is not None:
 421            gene_order_list = gene_order_list[:top_n]
 422    else:
 423        gene_order_list = list(gene_order)
 424
 425    d = d.loc[d[gene_col].isin(gene_order_list)]
 426    group = group.loc[d.index]
 427
 428    if len(d) == 0:
 429        raise ValueError("No rows left to plot for the requested genes (check gene_order/top_n).")
 430
 431    gene_pos = {g: i for i, g in enumerate(gene_order_list)}
 432
 433    if figsize is None:
 434        figsize = (max(6, 0.4 * len(gene_order_list)), 5)
 435    fig, ax = plt.subplots(figsize=figsize)
 436
 437    if show_gene_bands:
 438        for i in range(1, len(gene_order_list), 2):
 439            ax.axvspan(i - 0.5, i + 0.5, color=gene_band_color,
 440                       alpha=gene_band_alpha, zorder=0, linewidth=0)
 441
 442    rng = np.random.RandomState(0)
 443    base_x = d[gene_col].map(gene_pos).values.astype(float)
 444    jittered_x = base_x + rng.uniform(-jitter, jitter, size=len(d))
 445    values = d[value_col].values
 446
 447    for group_name in group_order:
 448        mask = (group == group_name).values
 449        if not mask.any():
 450            continue
 451        ax.scatter(jittered_x[mask], values[mask], s=point_size, alpha=alpha,
 452                   color=color_map[group_name], label=group_name.replace('_', ' '),
 453                   edgecolor='none', zorder=3)
 454
 455    if show_zero_line:
 456        ax.axhline(0, color='gray', linestyle=':', linewidth=0.8, alpha=0.6, zorder=1)
 457
 458    ax.set_xticks(range(len(gene_order_list)))
 459    ax.set_xticklabels(gene_order_list, rotation=90, fontsize=8)
 460    ax.set_xlim(-0.5, len(gene_order_list) - 0.5)
 461    ax.set_ylabel(value_col)
 462    ax.set_xlabel(gene_col)
 463    ax.spines['top'].set_visible(False)
 464    ax.spines['right'].set_visible(False)
 465    ax.legend(frameon=False, title=legend_title)
 466    fig.tight_layout()
 467
 468    if show:
 469        plt.show()
 470
 471    return fig

 474def scatter_with_smooth_by_group(
 475    x,
 476    y,
 477    group,
 478    frac=0.2,
 479    s=1,
 480    alpha=0.3,
 481    xlabel=None,
 482    ylabel=None,
 483    title=None,
 484    ax=None,
 485    show=True,
 486):
 487    """
 488    Scatter plot with per-group lowess smoothed lines, drawn on the current axes.
 489
 490    Parameters
 491    ----------
 492    x : array-like or torch.Tensor
 493        X-axis values.
 494    y : array-like or torch.Tensor
 495        Y-axis values.
 496    group : array-like
 497        Group labels; one smoothed line is drawn per unique value.
 498    frac : float, default 0.2
 499        Lowess smoothing fraction.
 500    s : float, default 1
 501        Scatter marker size.
 502    alpha : float, default 0.3
 503        Scatter marker transparency.
 504    xlabel : str, optional
 505        X-axis label.
 506    ylabel : str, optional
 507        Y-axis label.
 508    title : str, optional
 509        Plot title.
 510    ax : matplotlib axes, optional
 511        Axes to plot on. If None, uses current axes.
 512    show : bool, default True
 513        Whether to call ``plt.show()``.
 514
 515    Returns
 516    -------
 517    ax : matplotlib axes
 518    """
 519    if hasattr(x, "detach"):
 520        x = x.detach().cpu().numpy()
 521    else:
 522        x = np.asarray(x)
 523
 524    if hasattr(y, "detach"):
 525        y = y.detach().cpu().numpy()
 526    else:
 527        y = np.asarray(y)
 528
 529    group = np.asarray(group)
 530
 531    x = np.ravel(x)
 532    y = np.ravel(y)
 533    group = np.ravel(group)
 534
 535    mask = np.isfinite(x) & np.isfinite(y)
 536    x = x[mask]
 537    y = y[mask]
 538    group = group[mask]
 539
 540    if ax is None:
 541        ax = plt.gca()
 542
 543    for g in np.unique(group):
 544        idx = group == g
 545        x_g = x[idx]
 546        y_g = y[idx]
 547
 548        order = np.argsort(x_g)
 549        x_g = x_g[order]
 550        y_g = y_g[order]
 551
 552        ax.scatter(x_g, y_g, s=s, alpha=alpha, label=str(g))
 553        smoothed = lowess(y_g, x_g, frac=frac, return_sorted=True)
 554        (line,) = ax.plot(smoothed[:, 0], smoothed[:, 1], linewidth=2.5, zorder=10)
 555        line.set_path_effects([

 573def plot_x_true_residuals_vs_sumfactor(
 574    model,
 575    group_col,
 576    facet_col,
 577    sum_factor_col="sum_factor",
 578    frac=0.2,
 579    s=1,
 580    alpha=0.3,
 581    figsize=(12, 5),
 582    min_cells_for_smooth=10,
 583    show=True,
 584):
 585    """
 586    Diagnostic: log2(sum_factor) vs normalised log2_x_true, faceted by a
 587    user-specified column, with a lowess line per group.
 588
 589    For each cell assigned to guide g, the normalised x_true is::
 590
 591        (log2_x_true_cell - log2_x_eff_g_guide) / sigma_eff_guide
 592
 593    where log2_x_eff_g and sigma_eff are posterior-mean guide-level parameters
 594    from the cis fit.  Under the model this should be N(0, 1) with no dependence
 595    on sum_factor; a systematic trend reveals residual confounding between
 596    sequencing depth and inferred expression.
 597
 598    Only single-guide cells are included (all cells in low-MOI; a subset in
 599    high-MOI mode).
 600
 601    Parameters
 602    ----------
 603    model : bayesDREAM
 604        A fitted model (fit_cis must have been called).
 605    group_col : str
 606        Column in model.meta to colour and smooth separately (e.g. ``'lane'``,
 607        ``'batch'``).
 608    facet_col : str
 609        Column in model.meta to facet by; one panel per unique value
 610        (e.g. ``'target'``, ``'cell_line'``).
 611    sum_factor_col : str, default ``'sum_factor'``
 612        Column in the primary modality's sum_factors to use as y-axis.
 613    frac : float, default 0.2
 614        Lowess smoothing fraction.
 615    s : float, default 1
 616        Scatter marker size.
 617    alpha : float, default 0.3
 618        Scatter marker transparency.
 619    figsize : tuple, default (12, 5)
 620        Figure size.
 621    min_cells_for_smooth : int, default 10
 622        Minimum cells in a (facet × group) slice to draw a smoothed line.
 623    show : bool, default True
 624        Whether to call ``plt.show()``.
 625
 626    Returns
 627    -------
 628    fig : matplotlib.figure.Figure
 629    """
 630    import torch
 631
 632    if not hasattr(model, "posterior_samples_cis") or model.posterior_samples_cis is None:
 633        raise RuntimeError("fit_cis() must be called before this plot.")
 634
 635    # --- posterior-mean guide-level quantities ---
 636    x_eff_g_mean = model.posterior_samples_cis["x_eff_g"].mean(dim=0)     # [G]
 637    sigma_eff_mean = model.posterior_samples_cis["sigma_eff"].mean(dim=0)  # [G]
 638
 639    log2_x_eff_g_np = torch.log2(x_eff_g_mean.clamp(min=1e-12)).cpu().numpy()
 640    sigma_eff_np = sigma_eff_mean.cpu().numpy()
 641
 642    # --- cell-level arrays (positionally aligned with model.meta rows) ---
 643    meta = model.meta
 644    cell_col = meta["cell"].values  # cell barcodes for sum-factor lookup
 645
 646    log2_x_true = model.log2_x_true
 647    if hasattr(log2_x_true, "detach"):
 648        log2_x_true = log2_x_true.detach().cpu().numpy()
 649    else:
 650        log2_x_true = np.asarray(log2_x_true, dtype=float)
 651
 652    sf_series = model.get_modality(model.primary_modality).sum_factors[sum_factor_col]
 653    sf_values = sf_series.loc[cell_col].values.astype(float)
 654    log2_sf = np.log2(sf_values)
 655
 656    groups = meta[group_col].values
 657    facets_raw = meta[facet_col].values
 658
 659    # --- single-guide mask + guide index per cell ---
 660    is_high_moi = getattr(model, "is_high_moi", False)
 661    if is_high_moi:
 662        ga = model.guide_assignment_tensor  # [N, G]
 663        n_guides_per_cell = ga.sum(dim=1).cpu().numpy()
 664        single_mask = n_guides_per_cell == 1
 665        guide_idx = ga.argmax(dim=1).cpu().numpy()
 666    else:
 667        single_mask = np.ones(len(meta), dtype=bool)
 668        guide_idx = meta["guide_code"].values.astype(int)
 669
 670    # --- normalised residual ---
 671    log2_x_eff_per_cell = log2_x_eff_g_np[guide_idx]
 672    sigma_per_cell = sigma_eff_np[guide_idx]
 673    normalized = (log2_x_true - log2_x_eff_per_cell) / np.maximum(sigma_per_cell, 1e-6)
 674
 675    # apply single-guide mask
 676    sel = single_mask
 677    normalized = normalized[sel]
 678    log2_sf = log2_sf[sel]
 679    groups = groups[sel]
 680    facets_raw = facets_raw[sel]
 681    n_single = int(sel.sum())
 682
 683    # --- facet panels ---
 684    facet_values = np.unique(facets_raw)
 685    n_facets = len(facet_values)
 686
 687    all_groups = np.unique(groups)
 688    cmap = plt.cm.tab10(np.linspace(0, 1, max(len(all_groups), 1)))
 689    color_map = {g: cmap[i] for i, g in enumerate(all_groups)}
 690
 691    fig, axes = plt.subplots(1, n_facets, figsize=figsize, sharey=True)
 692    if n_facets == 1:
 693        axes = [axes]
 694
 695    for ax, fval in zip(axes, facet_values):
 696        facet_mask = facets_raw == fval
 697        n_facet = int(facet_mask.sum())
 698
 699        for g in all_groups:
 700            g_mask = facet_mask & (groups == g)
 701            if g_mask.sum() < 2:
 702                continue
 703
 704            x_g = normalized[g_mask]
 705            y_g = log2_sf[g_mask]
 706            order = np.argsort(x_g)
 707            x_g = x_g[order]
 708            y_g = y_g[order]
 709
 710            color = color_map[g]
 711            ax.scatter(x_g, y_g, s=s, alpha=alpha, color=color, label=str(g), rasterized=True)
 712
 713            if g_mask.sum() >= min_cells_for_smooth:
 714                smoothed = lowess(y_g, x_g, frac=frac, return_sorted=True)
 715                (line,) = ax.plot(
 716                    smoothed[:, 0], smoothed[:, 1],
 717                    linewidth=2.5, color=color, zorder=10,
 718                )
 719                line.set_path_effects([
 720                    pe.Stroke(linewidth=4, foreground="white"),
 721                    pe.Normal(),
 722                ])
 723
 724        ax.axvline(0, color="gray", linewidth=1, linestyle="--", zorder=5, alpha=0.7)
 725        ax.set_xlabel(
 726            "Normalised log2_x_true\n"
 727            r"$(log_2 x_{true} - log_2 x_{eff,g}) \,/\, \sigma_{eff,g}$"
 728        )
 729        ax.set_title(f"{facet_col}={fval}  (n={n_facet:,})")
 730        ax.legend(title=group_col, markerscale=4, fontsize=7)
 731
 732    axes[0].set_ylabel(f"log2({sum_factor_col})")
 733    fig.suptitle(
 734        f"log2({sum_factor_col}) vs normalised log2_x_true\n"
 735        f"single-guide cells only  (n={n_single:,})"
 736    )
 737    plt.tight_layout()
 738    if show:
 739        plt.show()
 740    return fig

 884def _declutter_texts(fig, ax, texts, iterations=200, pad_px=1.5):
 885    """
 886    Iteratively nudge overlapping ``Text`` labels apart in display-pixel
 887    space, converting shifts back to data coordinates for repositioning.
 888
 889    Lightweight fallback for when ``adjustText`` isn't installed: not as
 890    good, but resolves direct overlaps within a panel.
 891    """
 892    if len(texts) < 2:
 893        return
 894    fig.canvas.draw()
 895    renderer = fig.canvas.get_renderer()
 896
 897    x0, x1 = ax.get_xlim()
 898    y0, y1 = ax.get_ylim()
 899    bbox = ax.get_window_extent(renderer=renderer)
 900    px_per_xdata = bbox.width / (x1 - x0)
 901    px_per_ydata = bbox.height / (y1 - y0)
 902    if px_per_xdata <= 0 or px_per_ydata <= 0:
 903        return
 904
 905    for _ in range(iterations):
 906        boxes = [t.get_window_extent(renderer=renderer) for t in texts]
 907        moved = False
 908        for i in range(len(texts)):
 909            for j in range(i + 1, len(texts)):
 910                bi, bj = boxes[i], boxes[j]
 911                if not bi.overlaps(bj):
 912                    continue
 913                moved = True
 914                overlap_x = min(bi.x1, bj.x1) - max(bi.x0, bj.x0)
 915                overlap_y = min(bi.y1, bj.y1) - max(bi.y0, bj.y0)
 916                cix, cjx = (bi.x0 + bi.x1) / 2, (bj.x0 + bj.x1) / 2
 917                ciy, cjy = (bi.y0 + bi.y1) / 2, (bj.y0 + bj.y1) / 2
 918                xi, yi = texts[i].get_position()
 919                xj, yj = texts[j].get_position()
 920                if overlap_x < overlap_y:
 921                    shift = (overlap_x / 2 + pad_px) * (1.0 if cix >= cjx else -1.0)
 922                    dx = shift / px_per_xdata
 923                    texts[i].set_position((xi + dx, yi))
 924                    texts[j].set_position((xj - dx, yj))
 925                else:
 926                    shift = (overlap_y / 2 + pad_px) * (1.0 if ciy >= cjy else -1.0)
 927                    dy = shift / px_per_ydata
 928                    texts[i].set_position((xi, yi + dy))
 929                    texts[j].set_position((xj, yj - dy))
 930        if not moved:
 931            break
 932        fig.canvas.draw()
 933        renderer = fig.canvas.get_renderer()
 934

 936def _draw_leader_lines(ax, texts, orig_xy, min_px=3.0, color="gray", lw=0.5):
 937    """Draw a thin line from each label's original point to its (possibly
 938    decluttered) final position, when it moved more than ``min_px`` pixels."""
 939    fig = ax.figure
 940    fig.canvas.draw()
 941    for txt, (x0, y0) in zip(texts, orig_xy):
 942        x1, y1 = txt.get_position()
 943        d0 = ax.transData.transform((x0, y0))
 944        d1 = ax.transData.transform((x1, y1))
 945        if np.hypot(*(d1 - d0)) > min_px:
 946            ax.plot([x0, x1], [y0, y1], color=color, lw=lw, zorder=3.5)

 949def plot_systematic_shift_volcano(
 950    res,
 951    group_col=None,
 952    p_col="p_adj",
 953    effect_col="shift_est",
 954    label_col="feature",
 955    fdr_threshold=0.1,
 956    top_n=20,
 957    highlight_features=None,
 958    highlight_color="tab:blue",
 959    highlight_label_all=False,
 960    sig_color="firebrick",
 961    nonsig_color="#999999",
 962    label_fontsize=7,
 963    figsize=None,
 964    show=True,
 965):
 966    """
 967    Volcano plot of ``check_systematic_shift()`` results, faceted by group.
 968
 969    One panel per unique value of ``group_col`` (the grouping column produced
 970    by ``check_systematic_shift(tech_col=...)`` -- named after whatever
 971    ``tech_col`` was passed, e.g. ``'technical_group_code'`` by default).
 972    Each panel plots ``effect_col`` (x) against ``-log10(p_col)`` (y),
 973    highlights points below ``fdr_threshold``, and labels the ``top_n`` most
 974    significant features per panel.
 975
 976    Parameters
 977    ----------
 978    res : pd.DataFrame
 979        Output of ``model.check_systematic_shift()``.
 980    group_col : str, optional
 981        Column to facet by. If None: uses ``'technical_group_code'`` if present
 982        (``check_systematic_shift``'s default ``tech_col`` name), otherwise
 983        auto-detected as the single non-boolean column in ``res`` that isn't
 984        one of the standard result columns (raises ``ValueError`` if that's
 985        ambiguous, e.g. if you added your own derived columns to ``res`` --
 986        pass it explicitly in that case).
 987    p_col : str, default ``'p_adj'``
 988        Which BH-corrected p-value column to plot: ``'p_adj'`` (likelihood-ratio
 989        test) or ``'shift_p_adj'`` (Wald test on the ``targeted`` coefficient).
 990        See ``check_systematic_shift``'s docstring for the difference.
 991    effect_col : str, default ``'shift_est'``
 992        Column for the x-axis (effect size).
 993    label_col : str, default ``'feature'``
 994        Column supplying the text used to label top hits.
 995    fdr_threshold : float, default 0.1
 996        Significance cutoff; also drawn as a horizontal dashed line.
 997    top_n : int, default 20
 998        Number of most-significant (lowest ``p_col``) features to label per
 999        panel. Without ``highlight_features``, this is the top_n among rows
1000        below ``fdr_threshold``. With ``highlight_features``, this is instead
1001        the top_n most significant *of the highlighted matches* (regardless
1002        of ``fdr_threshold``) -- e.g. ``top_n=5`` labels only the 5 most
1003        significant heat shock proteins. Ignored if ``highlight_label_all=True``.
1004    highlight_features : list of str, optional
1005        Feature names (matched against ``label_col``) to highlight, e.g. a
1006        curated gene set such as heat shock proteins. When given, every
1007        matching row in each panel is drawn with an open ``highlight_color``
1008        ring (regardless of significance, so you can see the whole set), and
1009        the ``top_n`` most significant of them are text-labeled (see
1010        ``top_n`` and ``highlight_label_all``).
1011    highlight_color : str, default ``'tab:blue'``
1012        Ring color used to mark ``highlight_features`` points.
1013    highlight_label_all : bool, default ``False``
1014        If True, text-label every ``highlight_features`` match instead of
1015        just the ``top_n`` most significant. Only meaningful when
1016        ``highlight_features`` is given -- expect heavy overlap if the set
1017        is large and clustered.
1018    sig_color, nonsig_color : str
1019        Colors for significant / non-significant points.
1020    label_fontsize : int, default 7
1021        Font size for point labels.
1022    figsize : tuple, optional
1023        Figure size. Defaults to ``(5 * n_panels, 5)``.
1024    show : bool, default True
1025        Whether to call ``plt.show()``.
1026
1027    Returns
1028    -------
1029    fig : matplotlib.figure.Figure
1030
1031    Raises
1032    ------
1033    ValueError
1034        If ``group_col`` is None and cannot be unambiguously auto-detected
1035        (see ``group_col`` above), or if no rows have ``ok == True`` with
1036        finite ``effect_col``/``p_col`` values to plot.
1037
1038    Notes
1039    -----
1040    Only rows with ``ok == True`` and finite ``effect_col``/``p_col`` are
1041    plotted; skipped (``ok == False``) rows are silently dropped.
1042
1043    If the `adjustText <https://github.com/Phlya/adjustText>`_ package is
1044    installed, labels are auto-repelled using it (best quality). Otherwise a
1045    built-in pixel-space decluttering pass nudges overlapping labels apart
1046    and draws thin leader lines back to their points.
1047    """
1048    group_col = _detect_group_col(res, group_col)
1049
1050    df = res[res["ok"] & np.isfinite(res[effect_col]) & np.isfinite(res[p_col])].copy()
1051    if df.empty:
1052        raise ValueError("No rows with ok=True and finite values to plot.")
1053
1054    # Avoid -log10(0) = inf for exact-zero p-values
1055    tiny = np.finfo(float).tiny
1056    df["_neg_log10_p"] = -np.log10(df[p_col].clip(lower=tiny))
1057    df["_sig"] = df[p_col] < fdr_threshold
1058
1059    group_values = sorted(df[group_col].unique())
1060    n_panels = len(group_values)
1061    if figsize is None:
1062        figsize = (5 * n_panels, 5)
1063
1064    fig, axes = plt.subplots(1, n_panels, figsize=figsize, sharex=True, sharey=True)
1065    if n_panels == 1:
1066        axes = [axes]
1067
1068    hline_y = -np.log10(fdr_threshold)
1069
1070    for ax, gval in zip(axes, group_values):
1071        sub = df[df[group_col] == gval]
1072        nonsig = sub[~sub["_sig"]]
1073        sig = sub[sub["_sig"]]
1074
1075        ax.axvline(0, linestyle="--", color="gray", linewidth=1, zorder=1)
1076        ax.axhline(hline_y, linestyle="--", color=sig_color, linewidth=1, zorder=1)
1077
1078        ax.scatter(
1079            nonsig[effect_col], nonsig["_neg_log10_p"],
1080            s=10, alpha=0.4, color=nonsig_color, label="Not significant", zorder=2,
1081        )
1082        ax.scatter(
1083            sig[effect_col], sig["_neg_log10_p"],
1084            s=14, alpha=0.8, color=sig_color,
1085            label=f"{_P_COL_LABELS.get(p_col, p_col)} < {fdr_threshold}", zorder=3,
1086        )
1087
1088        if highlight_features is not None:
1089            highlighted = sub[sub[label_col].isin(set(highlight_features))]
1090            if not highlighted.empty:
1091                ax.scatter(

1121        if texts:
1122            try:
1123                from adjustText import adjust_text
1124                adjust_text(
1125                    texts, ax=ax,
1126                    arrowprops=dict(arrowstyle="-", color="gray", lw=0.5),
1127                )
1128            except ImportError:
1129                # Built-in fallback: nudge overlapping labels apart, then
1130                # draw thin leader lines back to their points.

1135        ax.set_title(f"{group_col}={gval}  (n={len(sub):,}, sig={len(sig):,})")

1145def plot_shift_est_group_correlation(
1146    res,
1147    group_col=None,
1148    groups=None,
1149    effect_col="shift_est",

1152    fdr_threshold=0.1,
1153    highlight="either",

1156    nonsig_color="#999999",
1157    label_fontsize=7,
1158    figsize=(5, 5),
1159    show=True,
1160):

1162    Scatter ``effect_col`` for one group against another, matched by feature.
1163
1164    Checks whether the residual shift detected by ``check_systematic_shift()``
1165    is consistent across groups (e.g. two technical groups, or CRISPRi vs
1166    CRISPRa) rather than being an artefact of one group's fit -- a real

1171    ----------
1172    res : pd.DataFrame

1203
1204    Returns
1205    -------
1206    fig : matplotlib.figure.Figure

1216        join on ``label_col`` between the two groups produces no shared,
1217        valid features; or if ``highlight`` isn't one of ``'either'``,
1218        ``'both'``, ``'none'``.
1219
1220    Notes
1221    -----
1222    Only features with ``ok == True`` and finite ``effect_col``/``p_col`` in
1223    *both* groups are plotted (inner join on ``label_col``) -- features that
1224    were skipped or untested in one group are silently dropped. Pearson r
1225    and Spearman rho (computed on the matched pairs) are annotated on the
1226    plot. Labels use the same adjustText-or-built-in decluttering as
1227    ``plot_systematic_shift_volcano``.
1228    """

1230
1231    ok = res[res["ok"] & np.isfinite(res[effect_col]) & np.isfinite(res[p_col])]
1232    unique_groups = sorted(ok[group_col].unique())
1233
1234    if groups is None:
1235        if len(unique_groups) != 2:
1236            all_groups = sorted(res[group_col].unique())
1237            empty_groups = sorted(set(all_groups) - set(unique_groups))
1238            if empty_groups:
1239                lines = [
1240                    f"Group(s) {empty_groups} have zero rows with ok=True in `res` -- "
1241                    f"nothing to correlate for {'it' if len(empty_groups) == 1 else 'them'}. "
1242                    f"Groups with usable rows: {unique_groups}."
1243                ]

1246                    if not reason_counts.empty:
1247                        lines.append(f"  Skip reasons for group {g!r}:")
1248                        lines.extend(f"    {r}: {c}" for r, c in reason_counts.items())
1249                lines.append(
1250                    "This usually means min_cells_per_group (in check_systematic_shift()) "
1251                    "was too high for this group's cell counts after x_true-window matching. "
1252                    "Consider lowering min_cells_per_group and re-running, or pass "
1253                    f"groups={tuple(unique_groups)} explicitly if you only meant to plot those."
1254                )
1255                raise ValueError("\n".join(lines))
1256            raise ValueError(
1257                f"Expected exactly 2 groups, found {unique_groups}. "
1258                "Pass groups=(a, b) explicitly."
1259            )
1260        groups = tuple(unique_groups)
1261    ga, gb = groups
1262
1263    da = ok[ok[group_col] == ga][[label_col, effect_col, p_col]]
1264    db = ok[ok[group_col] == gb][[label_col, effect_col, p_col]]
1265    merged = pd.merge(da, db, on=label_col, suffixes=("_a", "_b"))
1266    if merged.empty:
1267        raise ValueError(
1268            f"No shared features with ok=True between groups {ga!r} and {gb!r}."
1269        )
1270
1271    x = merged[f"{effect_col}_a"].to_numpy()
1272    y = merged[f"{effect_col}_b"].to_numpy()
1273    pa = merged[f"{p_col}_a"].to_numpy()
1274    pb = merged[f"{p_col}_b"].to_numpy()
1275
1276    if highlight == "either":
1277        sig_mask = (pa < fdr_threshold) | (pb < fdr_threshold)
1278    elif highlight == "both":
1279        sig_mask = (pa < fdr_threshold) & (pb < fdr_threshold)
1280    elif highlight == "none":
1281        sig_mask = np.zeros(len(merged), dtype=bool)
1282    else:
1283        raise ValueError("highlight must be 'either', 'both', or 'none'.")
1284
1285    r, r_p = pearsonr(x, y)
1286    rho, _ = spearmanr(x, y)
1287    n = len(merged)
1288
1289    fig, ax = plt.subplots(figsize=figsize)
1290
1291    ax.scatter(
1292        x[~sig_mask], y[~sig_mask],
1293        s=10, alpha=0.4, color=nonsig_color, label="Not significant", zorder=2,
1294    )
1295    if sig_mask.any():
1296        ax.scatter(
1297            x[sig_mask], y[sig_mask],
1298            s=14, alpha=0.8, color=sig_color,
1299            label=f"{_P_COL_LABELS.get(p_col, p_col)} < {fdr_threshold} ({highlight})", zorder=3,
1300        )
1301
1302    lo, hi = min(x.min(), y.min()), max(x.max(), y.max())
1303    pad = 0.05 * (hi - lo) if hi > lo else 1.0
1304    ax.plot([lo - pad, hi + pad], [lo - pad, hi + pad],
1305            linestyle="--", color="gray", linewidth=1, zorder=1)
1306    ax.axhline(0, linestyle=":", color="gray", linewidth=0.7, zorder=1)
1307    ax.axvline(0, linestyle=":", color="gray", linewidth=0.7, zorder=1)
1308
1309    ax.text(
1310        0.03, 0.97,
1311        f"Pearson r = {r:.2f} (p={r_p:.1e})\nSpearman ρ = {rho:.2f}\nn = {n:,}",
1312        transform=ax.transAxes, va="top", ha="left", fontsize=8,
1313    )
1314
1315    if top_n > 0 and sig_mask.any():
1316        min_p = np.minimum(pa, pb)
1317        sig_idx = np.flatnonzero(sig_mask)
1318        top_idx = sig_idx[np.argsort(min_p[sig_idx])][:top_n]
1319
1320        texts, orig_xy = [], []
1321        x_span, y_span = x.max() - x.min(), y.max() - y.min()
1322        dx0 = 0.01 * x_span if x_span > 0 else 0.01
1323        dy0 = 0.02 * y_span if y_span > 0 else 0.02
1324        for i in top_idx:
1325            xi, yi = x[i], y[i]
1326            txt = ax.text(
1327                xi + dx0, yi + dy0, str(merged[label_col].iloc[i]),
1328                fontsize=label_fontsize, zorder=4,
1329            )
1330            txt.set_path_effects([pe.Stroke(linewidth=2, foreground="white"), pe.Normal()])
1331            texts.append(txt)
1332            orig_xy.append((xi, yi))
1333
1334        try:
1335            from adjustText import adjust_text
1336            adjust_text(texts, ax=ax, arrowprops=dict(arrowstyle="-", color="gray", lw=0.5))
1337        except ImportError:
1338            _declutter_texts(fig, ax, texts)
1339            _draw_leader_lines(ax, texts, orig_xy)
1340
1341    ax.set_xlabel(f"{effect_col} ({group_col}={ga})")
1342    ax.set_ylabel(f"{effect_col} ({group_col}={gb})")
1343    ax.set_title(f"{effect_col} correlation across {group_col}")
1344    ax.legend(fontsize=7, frameon=False, markerscale=1.5, loc="lower right")
1345
1346    plt.tight_layout()
1347    if show:
1348        plt.show()
1349    return fig

1355    value_col="shift_est",
1356    group_col_a=None,
1357    group_col_b=None,
1358    label_col="feature",
1359    p_col="p_adj",
1360    transform=None,
1361    fdr_threshold=0.1,
1362    highlight="either",
1363    top_n=0,
1364    dataset_names=("Dataset A", "Dataset B"),
1365    sig_color="firebrick",
1366    sig_color_a="tab:blue",
1367    sig_color_b="tab:orange",
1368    nonsig_color="#999999",
1369    label_fontsize=7,
1370    figsize=None,
1371    show=True,

1373    """
1374    Grid of scatter plots correlating ``value_col`` between two independently
1375    produced ``check_systematic_shift()`` result tables (e.g. from different
1376    datasets/experiments), matched by feature.
1377
1378    Columns = technical groups in ``res_a``; rows = technical groups in
1379    ``res_b`` -- i.e. every (group_a, group_b) combination gets its own
1380    panel (a full cross-tabulation, not a 1:1 group match, since group codes
1381    from two independently fit models aren't assumed to correspond). Each
1382    panel inner-joins ``res_a`` and ``res_b`` on ``label_col`` for that pair
1383    of groups and scatters ``res_a[value_col]`` (x) against
1384    ``res_b[value_col]`` (y).
1385
1386    Call this once with ``value_col='shift_est'`` and once with
1387    ``value_col='p_adj'`` (or ``'shift_p_adj'``) to get the shift-estimate
1388    and p-value correlation grids.
1389
1390    Parameters
1391    ----------
1392    res_a, res_b : pd.DataFrame
1393        Two ``check_systematic_shift()`` outputs to compare. Must share at
1394        least some ``label_col`` values within a given (group_a, group_b)
1395        pair for that panel to be non-empty.
1396    value_col : str, default ``'shift_est'``
1397        Column to correlate. Known p-value columns (``'p_adj'``,
1398        ``'shift_p_adj'``, ``'pval'``, ``'p_lrt'``, ``'shift_p'``) are
1399        ``-log10``-transformed automatically unless ``transform`` overrides
1400        this.
1401    group_col_a, group_col_b : str, optional
1402        Grouping column in ``res_a`` / ``res_b`` respectively. Auto-detected
1403        independently for each dataframe (same logic as
1404        ``plot_systematic_shift_volcano``), so they need not share a name.
1405    label_col : str, default ``'feature'``
1406        Column used to match rows across the two dataframes within a panel.
1407    p_col : str, default ``'p_adj'``
1408        Column name (assumed shared by both dataframes) used for
1409        significance highlighting, regardless of what ``value_col`` is.
1410    transform : {None, '-log10'}, optional
1411        Force the axis transform. If None (default), auto-detected from
1412        ``value_col`` (see above).
1413    fdr_threshold : float, default 0.1
1414        Significance cutoff applied to ``p_col`` in both dataframes.
1415    highlight : {'either', 'both', 'none'}, default ``'either'``
1416        - ``'either'``: color points significant in dataset A only, dataset B
1417          only, and both, each with a distinct color (``sig_color_a``,
1418          ``sig_color_b``, ``sig_color``) -- so you can see whether hits
1419          replicate or are dataset-specific.
1420        - ``'both'``: single-color highlight (``sig_color``) for points
1421          significant in both datasets; everything else grey.
1422        - ``'none'``: no significance-based coloring.
1423    top_n : int, default 0
1424        Label the top N highlighted points per panel (any of the
1425        significant categories under ``highlight='either'``) by ``min(p_col)``
1426        across the two datasets. 0 = no labels.
1427    dataset_names : tuple of 2 str, default ``('Dataset A', 'Dataset B')``
1428        Names used in the shared x/y axis labels and in the "significant in
1429        <name> only" legend entries.
1430    sig_color : str, default ``'firebrick'``
1431        Color for points significant in both datasets.
1432    sig_color_a, sig_color_b : str
1433        Colors for points significant in dataset A only / dataset B only
1434        (only used when ``highlight='either'``).
1435    nonsig_color : str
1436        Color for non-significant (or, if ``highlight='none'``, all) points.
1437    label_fontsize : int, default 7
1438    figsize : tuple, optional
1439        Defaults to ``(4 * n_groups_a, 4 * n_groups_b)``.
1440    show : bool, default True
1441
1442    Returns
1443    -------
1444    fig : matplotlib.figure.Figure
1445
1446    Raises
1447    ------
1448    ValueError
1449        If ``highlight`` isn't one of ``'either'``, ``'both'``, ``'none'``;
1450        or if either dataframe has zero groups with usable (``ok == True``,
1451        finite ``value_col``/``p_col``) rows.
1452
1453    Notes
1454    -----
1455    Panels with no shared features between the two datasets for that
1456    (group_a, group_b) pair are left blank with an annotation rather than
1457    raising -- expected when the two group grids don't align 1:1.
1458
1459    Each panel draws a dashed reference crosshair (both axes) at the null
1460    value for whatever's plotted: 0 for effect-size columns (gray dotted),
1461    or the FDR threshold on the -log10 scale for p-value columns
1462    (``sig_color`` dashed, matching ``plot_systematic_shift_volcano``'s
1463    threshold line).
1464
1465    Examples
1466    --------
1467    >>> fig_shift = plot_cross_dataset_correlation(
1468    ...     res, res_Domingo, value_col='shift_est',
1469    ...     dataset_names=('This run', 'Domingo'))
1470    >>> fig_p = plot_cross_dataset_correlation(
1471    ...     res, res_Domingo, value_col='p_adj',
1472    ...     dataset_names=('This run', 'Domingo'))
1473    """
1474    if highlight not in ("either", "both", "none"):
1475        raise ValueError("highlight must be 'either', 'both', or 'none'.")
1476
1477    group_col_a = _detect_group_col(res_a, group_col_a)
1478    group_col_b = _detect_group_col(res_b, group_col_b)
1479
1480    p_known = {"p_adj", "shift_p_adj", "pval", "p_lrt", "shift_p"}
1481    do_log = (value_col in p_known) if transform is None else (transform == "-log10")
1482
1483    ok_a = res_a[res_a["ok"] & np.isfinite(res_a[value_col]) & np.isfinite(res_a[p_col])]
1484    ok_b = res_b[res_b["ok"] & np.isfinite(res_b[value_col]) & np.isfinite(res_b[p_col])]
1485
1486    groups_a = sorted(ok_a[group_col_a].unique())
1487    groups_b = sorted(ok_b[group_col_b].unique())
1488    if not groups_a or not groups_b:
1489        raise ValueError(
1490            f"No usable rows (ok=True, finite {value_col}/{p_col}) in "
1491            f"{'res_a' if not groups_a else 'res_b'}."
1492        )
1493
1494    n_cols, n_rows = len(groups_a), len(groups_b)
1495    if figsize is None:
1496        figsize = (4 * n_cols, 4 * n_rows)
1497    fig, axes = plt.subplots(n_rows, n_cols, figsize=figsize, squeeze=False)
1498
1499    axis_label = f"$-\\log_{{10}}$({value_col})" if do_log else value_col
1500    tiny = np.finfo(float).tiny
1501
1502    # Build legend proxies up front (rather than harvesting handles from
1503    # whichever panel happens to be plotted first) so the legend always
1504    # shows every category ``highlight`` can produce, even if some category
1505    # is empty in every panel.
1506    def _dot(color, alpha, label):
1507        return mlines.Line2D([], [], marker="o", linestyle="None",
1508                              color=color, alpha=alpha, markersize=6, label=label)
1509    if highlight == "none":
1510        legend_elements = [_dot(nonsig_color, 0.4, "Data")]
1511    elif highlight == "both":
1512        legend_elements = [
1513            _dot(nonsig_color, 0.4, "Not significant"),
1514            _dot(sig_color, 0.9, f"{_P_COL_LABELS.get(p_col, p_col)} < {fdr_threshold} (both)"),
1515        ]
1516    else:  # either
1517        legend_elements = [
1518            _dot(nonsig_color, 0.4, "Not significant"),
1519            _dot(sig_color_a, 0.8, f"Significant in {dataset_names[0]} only"),
1520            _dot(sig_color_b, 0.8, f"Significant in {dataset_names[1]} only"),
1521            _dot(sig_color, 0.9, "Significant in both"),
1522        ]
1523
1524    # Avoid selecting the same column twice (e.g. value_col == p_col == 'p_adj'
1525    # for the p-value correlation call) -- that would duplicate the column
1526    # name within a single dataframe, before merge suffixing even applies.
1527    cols = [label_col, value_col] + ([p_col] if p_col != value_col else [])
1528
1529    for i, gb in enumerate(groups_b):
1530        db_full = ok_b[ok_b[group_col_b] == gb][cols]
1531        for j, ga in enumerate(groups_a):
1532            ax = axes[i, j]
1533            da_full = ok_a[ok_a[group_col_a] == ga][cols]
1534            merged = pd.merge(da_full, db_full, on=label_col, suffixes=("_a", "_b"))
1535
1536            if merged.empty:
1537                ax.text(
1538                    0.5, 0.5, "no shared\nfeatures", ha="center", va="center",
1539                    transform=ax.transAxes, fontsize=8, color="gray",
1540                )
1541                ax.set_xticks([])
1542                ax.set_yticks([])
1543            else:
1544                if p_col == value_col:
1545                    # Both columns are the same underlying values (merge only
1546                    # suffixed them once) -- no separate "_a"/"_b" p_col pair.
1547                    x_raw = merged[f"{value_col}_a"].to_numpy()
1548                    y_raw = merged[f"{value_col}_b"].to_numpy()
1549                    pa, pb = x_raw, y_raw
1550                else:
1551                    x_raw = merged[f"{value_col}_a"].to_numpy()
1552                    y_raw = merged[f"{value_col}_b"].to_numpy()
1553                    pa = merged[f"{p_col}_a"].to_numpy()
1554                    pb = merged[f"{p_col}_b"].to_numpy()
1555                if do_log:
1556                    x = -np.log10(np.clip(x_raw, tiny, None))
1557                    y = -np.log10(np.clip(y_raw, tiny, None))
1558                else:
1559                    x, y = x_raw, y_raw
1560
1561                sig_a = pa < fdr_threshold
1562                sig_b = pb < fdr_threshold
1563
1564                if highlight == "either":
1565                    cat_both = sig_a & sig_b
1566                    cat_a_only = sig_a & ~sig_b
1567                    cat_b_only = ~sig_a & sig_b
1568                elif highlight == "both":
1569                    cat_both = sig_a & sig_b
1570                    cat_a_only = np.zeros(len(merged), dtype=bool)
1571                    cat_b_only = np.zeros(len(merged), dtype=bool)
1572                else:
1573                    cat_both = np.zeros(len(merged), dtype=bool)
1574                    cat_a_only = np.zeros(len(merged), dtype=bool)
1575                    cat_b_only = np.zeros(len(merged), dtype=bool)
1576                sig_mask = cat_both | cat_a_only | cat_b_only
1577                cat_none = ~sig_mask
1578
1579                # Reference line: null effect (0) for effect-size columns,
1580                # or the FDR threshold (on the -log10 scale) for p-value columns.
1581                if do_log:
1582                    ref = -np.log10(fdr_threshold)
1583                    ref_color, ref_style = sig_color, "--"
1584                else:
1585                    ref = 0.0

Call model.plot_xy_data() for each significant check_systematic_shift() hit, restricted to the cells the test actually used.

check_systematic_shift() matches NTC and targeted cells to a ±1 SD window around the NTC mean of log2(x_true) before testing (see its "Matching window" docstring section) -- a much narrower slice than the full dose-response range plot_xy_data() shows by default. This reconstructs that same window (via model.get_shift_window_cells()) and applies it as a subset_meta filter, so the plot shows exactly the cells that drove the test's result.

Loops over the significant features in res (ok and p_col < fdr_threshold, unless features is given explicitly) and calls::

model.plot_xy_data(feature, subset_meta={window_col: True, **your_subset_meta}, **plot_xy_kwargs)

for each, after temporarily adding a boolean window_col column to model.meta marking cells within their own group's matched window (each cell only belongs to one tech_col group, so a single combined column is enough even when plot_xy_kwargs['facet_by'] differs from tech_col). The column is removed again afterwards, even on error.
