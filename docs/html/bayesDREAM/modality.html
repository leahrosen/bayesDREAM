 18class Modality:
 19    """
 20    Stores data and metadata for a single molecular modality.
 21
 22    Attributes
 23    ----------
 24    name : str
 25        Name of the modality (e.g., 'gene', 'transcript', 'splicing')
 26    counts : np.ndarray or pd.DataFrame
 27        The count/measurement data. Shape depends on distribution:
 28        - negbinom/normal/studentt: (features, cells) or (cells, features)
 29        - multinomial: (features, cells, categories)
 30        - binomial: (features, cells) with separate denominator
 31    feature_meta : pd.DataFrame
 32        Feature-level metadata (genes, junctions, etc.)
 33    distribution : str
 34        Distribution family: 'negbinom', 'multinomial', 'binomial', 'normal', 'studentt'
 35    denominator : np.ndarray, optional
 36        For binomial distribution: (features, cells) array of trial counts
 37    """
 38
 39    VALID_DISTRIBUTIONS = {'negbinom', 'multinomial', 'binomial', 'normal', 'studentt'}
 40
 41    def __init__(
 42        self,
 43        name: str,
 44        counts: Union[np.ndarray, pd.DataFrame],
 45        feature_meta: pd.DataFrame,
 46        distribution: Literal['negbinom', 'multinomial', 'binomial', 'normal', 'studentt'],
 47        feature_names: Optional[list] = None,
 48        denominator: Optional[np.ndarray] = None,
 49        cells_axis: int = 1,  # 0 if cells are rows, 1 if cells are columns
 50        cell_names: Optional[list] = None,  # Explicit cell names (when counts is ndarray)
 51        min_count: int = 1,
 52        # Exon skipping specific parameters
 53        inc1: Optional[np.ndarray] = None,
 54        inc2: Optional[np.ndarray] = None,
 55        skip: Optional[np.ndarray] = None,
 56        exon_aggregate_method: Optional[str] = None,
 57    ):
 58        """
 59        Initialize a Modality.
 60
 61        Parameters
 62        ----------
 63        name : str
 64            Modality name
 65        counts : np.ndarray or pd.DataFrame
 66            Count/measurement data
 67        feature_meta : pd.DataFrame
 68            Feature metadata with index matching features in counts
 69        distribution : str
 70            Distribution type
 71        denominator : np.ndarray, optional
 72            For binomial: denominator counts (e.g., total gene expression for SJ usage)
 73        cells_axis : int
 74            Which axis represents cells (0 or 1 for 2D data)
 75        cell_names : list, optional
 76            Explicit cell names/identifiers (used when counts is ndarray, not DataFrame)
 77        inc1 : np.ndarray, optional
 78            For exon skipping: inclusion counts from first junction (d1->a2)
 79        inc2 : np.ndarray, optional
 80            For exon skipping: inclusion counts from second junction (d2->a3)
 81        skip : np.ndarray, optional
 82            For exon skipping: skipping counts (d1->a3)
 83        exon_aggregate_method : str, optional
 84            For exon skipping: how inc1 and inc2 were aggregated ('min' or 'mean')
 85        """
 86        if distribution not in self.VALID_DISTRIBUTIONS:
 87            raise ValueError(f"distribution must be one of {self.VALID_DISTRIBUTIONS}, got {distribution}")
 88
 89        self.name = name
 90        self.distribution = distribution
 91        self.cells_axis = cells_axis
 92
 93        # Handle different count input formats
 94        # Check if counts is sparse matrix
 95        self.is_sparse = sparse.issparse(counts)
 96
 97        if isinstance(counts, pd.DataFrame):
 98            # DataFrame: store both values and DataFrame
 99            self.counts = counts.values
100            self.count_df = counts
101            self.is_sparse = False  # DataFrames are dense
102            # Feature names: use explicit parameter if provided, else from index/columns
103            if cells_axis == 1:
104                self.feature_names = feature_names if feature_names is not None else counts.index.tolist()
105                self.cell_names = counts.columns.tolist()
106            else:
107                self.feature_names = feature_names if feature_names is not None else counts.columns.tolist()
108                self.cell_names = counts.index.tolist()
109        elif self.is_sparse:
110            # Sparse matrix: keep sparse, don't densify!
111            self.counts = counts
112            self.count_df = None
113            self.feature_names = feature_names if feature_names is not None else None
114            self.cell_names = cell_names if cell_names is not None else None
115            pass  # sparse matrix stored as-is
116        else:
117            # Dense array: convert to numpy array
118            self.counts = np.asarray(counts)
119            self.count_df = None
120            self.feature_names = feature_names if feature_names is not None else None
121            self.cell_names = cell_names if cell_names is not None else None
122
123        # If no explicit feature_names and no DataFrame index/columns to derive
124        # from (raw ndarray/sparse counts, e.g. add_custom_modality), fall back
125        # to feature_meta — same priority order used everywhere else in the
126        # codebase, so downstream code can always rely on modality.feature_names
127        # instead of re-deriving its own guess from feature_meta.
128        if self.feature_names is None:
129            self.feature_names = resolve_feature_names(feature_meta, context=f"Modality '{name}'")
130
131        # Disambiguate duplicate feature names (e.g. Ensembl gene_name collisions
132        # from pseudogenes/readthrough transcripts, or a many-to-one fallback
133        # column like 'gene' repeating once per SJ belonging to that gene).
134        # Name-based lookups throughout the codebase (feature_names.index(),
135        # pd.Series indexed by feature_names, get_feature_subset(), plotting by
136        # gene name, etc.) all assume uniqueness.
137        if self.feature_names is not None:
138            deduped = make_names_unique(self.feature_names)
139            if deduped != self.feature_names:
140                n_dup = sum(1 for a, b in zip(deduped, self.feature_names) if a != b)
141                warnings.warn(
142                    f"Modality '{name}' had {n_dup} duplicate feature name(s); "
143                    "disambiguated by appending -1, -2, ... to repeats (scanpy-style). "
144                    "Original identifiers are still available in feature_meta.",
145                    UserWarning
146                )
147            self.feature_names = deduped
148
149        self.feature_meta = feature_meta.copy()
150
151        # For multinomial: ensure the implicit Kth residual slot (position K_max-1) is
152        # always a REAL category, not a padded zero.
153        #
154        # Background: the K-1 residual parameterization computes
155        #   p_{K_max-1} = 1 - sum(p_0 .. p_{K_max-2})
156        # For features with K_actual < K_max the last slot is padded (zero observations).
157        # If the real categories together already account for all probability mass (~1),
158        # then p_{K_max-1} ≈ 0 or negative, and the Dirichlet prior on A inappropriately
159        # assigns mass to the phantom categories that compete with real ones.
160        #
161        # Fix: swap the last real category (column K_actual-1) into the residual slot
162        # (column K_max-1) for any feature where K_actual < K_max.  After reordering:
163        #   columns 0 .. K_actual-2  : real explicit Hills
164        #   columns K_actual-1 .. K_max-2 : phantom (padded, zero data)
165        #   column  K_max-1              : real residual category  ✓
166        #
167        # n_categories (= K_actual per feature) is computed before reordering and is
168        # unaffected by it (still counts non-zero columns, just at different positions).
169        if distribution == 'multinomial':
170            counts_arr = np.asarray(self.counts)  # safe: multinomial must be dense 3D
171            if counts_arr.ndim == 3:
172                # Compute per-feature actual category count from non-zero data (before reordering).
173                # This always overwrites any user-provided n_categories to ensure it is
174                # consistent with the reordering performed below.  Zero-count categories
175                # (present in the tensor but with no reads in any cell) are treated as phantom.
176                n_cats = (counts_arr.sum(axis=1) > 0).sum(axis=1).astype(int)
177                self.feature_meta['n_categories'] = n_cats
178
179                # Reorder: swap last real category to residual position for any feature
180                # with K_actual < K_max.  Work on a copy so original input is untouched.
181                K_max = counts_arr.shape[2]
182                if K_max > 1 and np.any(n_cats < K_max):
183                    counts_arr = counts_arr.copy()
184                    for i in range(counts_arr.shape[0]):
185                        K_actual = int(n_cats[i])
186                        if 0 < K_actual < K_max:
187                            # Swap col K_actual-1 (last real) with col K_max-1 (padded residual slot)
188                            tmp = counts_arr[i, :, K_actual - 1].copy()
189                            counts_arr[i, :, K_actual - 1] = 0.0  # was real, now becomes padding
190                            counts_arr[i, :, K_max - 1] = tmp     # residual slot now real
191                    self.counts = counts_arr
192
193                # ---- Filter: drop multinomial features with ≤1 real category or
194                #              zero ratio-variance across cells ----
195                #
196                # Filter 1: Features with ≤1 real category cannot have multinomial
197                #           proportions (no structure to model).
198                # Filter 2: Features where all real-category ratios (count/cell_total)
199                #           have zero std across cells carry no fitting signal.
200                keep = n_cats > 1  # [T] bool, Filter 1
201
202                for i in range(len(n_cats)):
203                    if not keep[i]:
204                        continue
205                    feat = counts_arr[i]  # [n_cells, K_max]
206                    cell_totals = feat.sum(axis=1, keepdims=True)  # [n_cells, 1]
207                    real_cat_mask = feat.sum(axis=0) > 0  # [K_max]
208                    real_counts = feat[:, real_cat_mask]  # [n_cells, K_actual]
209                    with np.errstate(divide='ignore', invalid='ignore'):
210                        ratios = np.where(
211                            cell_totals > 0,
212                            real_counts / cell_totals,
213                            0.0,
214                        )  # [n_cells, K_actual]
215                    if np.all(ratios.std(axis=0) == 0):
216                        keep[i] = False  # Filter 2: zero ratio variance
217
218                n_dropped = int((~keep).sum())
219                if n_dropped > 0:
220                    n_le1_cat = int((n_cats <= 1).sum())
221                    n_zero_var = n_dropped - n_le1_cat
222                    warnings.warn(
223                        f"[Modality '{name}'] Dropped {n_dropped} multinomial feature(s): "
224                        f"{n_le1_cat} with ≤1 real category, "
225                        f"{n_zero_var} with zero category-ratio variance across cells.",
226                        UserWarning,
227                        stacklevel=2,
228                    )
229                    counts_arr = counts_arr[keep]
230                    if counts_arr.shape[0] == 0:
231                        raise ValueError(
232                            f"[Modality '{name}'] All multinomial features were filtered out "
233                            f"(≤1 real category or zero ratio variance). Check input data."
234                        )
235                    self.feature_meta = self.feature_meta[keep].reset_index(drop=True)
236                    self.feature_meta['n_categories'] = n_cats[keep]
237                    self.counts = counts_arr
238                    if self.feature_names is not None:
239                        self.feature_names = [
240                            fn for fn, k in zip(self.feature_names, keep) if k
241                        ]
242
243        # Handle denominator (can also be sparse)
244        if denominator is not None:
245            if sparse.issparse(denominator):
246                self.denominator = denominator  # Keep sparse
247            else:
248                self.denominator = np.asarray(denominator)
249        else:
250            self.denominator = None
251
252        # Exon skipping specific storage
253        self.inc1 = np.asarray(inc1) if inc1 is not None else None
254        self.inc2 = np.asarray(inc2) if inc2 is not None else None
255        self.skip = np.asarray(skip) if skip is not None else None
256        self.exon_aggregate_method = exon_aggregate_method
257        self._technical_fit_aggregate_method = None  # Track method used during technical fit
258
259        # Store unfiltered versions for recovery when switching methods
260        # (Only used for exon skipping modalities)
261        self._unfiltered_inc1 = None
262        self._unfiltered_inc2 = None
263        self._unfiltered_skip = None
264        self._unfiltered_feature_meta = None
265
266        # Per-modality fitting results storage
267        # Distribution-specific alpha_y parameters (use these directly, not alpha_y_prefit)
268        self.alpha_y_prefit_mult = None     # For negbinom: multiplicative correction [C, T]
269        self.alpha_y_prefit_add = None      # For normal/studentt/binomial/multinomial: additive correction [C, T]
270        self.posterior_samples_ntc = None  # NTC fit: full posterior samples
271        self.posterior_samples_trans = None      # Trans fit: full posterior samples
272
273        # Sum factors DataFrame (negbinom modalities only).
274        # Index: cell barcode. Columns: named sum factor variants
275        # (e.g. 'sum_factor', 'sum_factor_adj', 'sum_factor_new').
276        # Populated by bayesDREAM._init_sum_factors() after model init.
277        # The 'cis' modality holds a reference to the same object as the primary modality.
278        self.sum_factors: Optional[pd.DataFrame] = None
279
280        self.min_count = min_count
281
282        # Validate shapes
283        self._validate()
284
285        # Store dimensionality info
286        self.dims = self._compute_dims()
287
288        # Drop features with zero total counts or zero variance across cells.
289        # This runs at creation and automatically re-runs after cell subsetting
290        # (get_cell_subset returns a new Modality, triggering __init__).
291        self._filter_zero_features()

293    def _filter_zero_features(self) -> None:
294        """
295        Remove features with total counts below min_count or zero std across cells.
296
297        Called at the end of __init__, so it also fires automatically after any cell
298        subsetting (get_cell_subset returns a new Modality which calls __init__).
299        Modifies self in place; updates counts, feature_meta, feature_names, denominator,
300        inc1/inc2/skip, and dims.
301
302        Per-distribution logic
303        ----------------------
304        negbinom   : drop feature if total counts < min_count OR std across cells == 0
305        normal/
306        studentt   : drop feature if std across cells == 0
307                     (sum of values < min_count is also checked for consistency)
308        binomial   : drop feature if denominator sum < min_count (too few observations)
309                     OR ratio std (numer/denom) across cells == 0
310        multinomial: drop feature if total counts (summed over cells+categories) < min_count
311                     (zero-ratio-variance is already handled earlier in __init__)
312        """
313        min_count = self.min_count
314
315        # ------------------------------------------------------------------ #
316        # 1. Compute keep mask per distribution                               #
317        # ------------------------------------------------------------------ #
318        if self.distribution == 'multinomial':
319            counts_arr = np.asarray(self.counts)          # (T, C, K)
320            totals = counts_arr.sum(axis=(1, 2))          # (T,)
321            keep = totals >= min_count
322
323        elif self.distribution == 'binomial':
324            cell_ax = self.cells_axis
325
326            counts_num = (self.counts.toarray() if sparse.issparse(self.counts)
327                          else np.asarray(self.counts))
328            if self.denominator is not None:
329                denom = (self.denominator.toarray() if sparse.issparse(self.denominator)
330                         else np.asarray(self.denominator))
331            else:
332                denom = counts_num
333
334            denom_sums = denom.sum(axis=cell_ax)
335            has_data = denom_sums >= min_count
336
337            with np.errstate(divide='ignore', invalid='ignore'):
338                ratios = np.where(denom > 0, counts_num / denom, np.nan)
339            ratio_stds = np.nanstd(ratios, axis=cell_ax)
340            has_variance = ratio_stds > 0
341
342            keep = has_data & has_variance
343
344        else:  # negbinom, normal, studentt
345            cell_ax = self.cells_axis
346
347            if self.is_sparse:
348                _sp = self.counts.astype(float)
349                totals = np.array(_sp.sum(axis=cell_ax)).flatten()
350                _mean = np.array(_sp.mean(axis=cell_ax)).flatten()
351                _sp2 = _sp.copy()
352                _sp2.data **= 2
353                _sq_mean = np.array(_sp2.mean(axis=cell_ax)).flatten()
354                stds = np.sqrt(np.maximum(_sq_mean - _mean ** 2, 0))
355            else:
356                counts_arr = np.asarray(self.counts)
357                totals = counts_arr.sum(axis=cell_ax)
358                stds = counts_arr.std(axis=cell_ax)
359
360            keep = (totals >= min_count) & (stds > 0)
361
362        n_removed = int((~keep).sum())
363        if n_removed == 0:
364            return
365
366        keep_idx = np.where(keep)[0]
367        n_before = self.dims['n_features']
368
369        print(
370            f"[Modality '{self.name}'] Removed {n_removed}/{n_before} feature(s) "
371            f"with total counts < {min_count} or zero variance across cells."
372        )
373
374        # ------------------------------------------------------------------ #
375        # 2. Apply mask to counts and auxiliary arrays                        #
376        # ------------------------------------------------------------------ #
377        if self.distribution == 'multinomial':
378            self.counts = np.asarray(self.counts)[keep_idx, :, :]
379
380        elif self.cells_axis == 1:
381            if self.is_sparse:
382                self.counts = self.counts[keep_idx, :]
383            else:
384                self.counts = np.asarray(self.counts)[keep_idx, :]
385
386            if self.denominator is not None:
387                if sparse.issparse(self.denominator):
388                    self.denominator = self.denominator[keep_idx, :]
389                else:
390                    self.denominator = np.asarray(self.denominator)[keep_idx, :]
391
392            for _attr in ('inc1', 'inc2', 'skip'):
393                _val = getattr(self, _attr)
394                if _val is not None:
395                    setattr(self, _attr, _val[keep_idx, :])
396
397        else:  # cells_axis == 0
398            if self.is_sparse:
399                self.counts = self.counts[:, keep_idx]
400            else:
401                self.counts = np.asarray(self.counts)[:, keep_idx]
402
403            if self.denominator is not None:
404                if sparse.issparse(self.denominator):
405                    self.denominator = self.denominator[:, keep_idx]
406                else:
407                    self.denominator = np.asarray(self.denominator)[:, keep_idx]
408
409        # count_df is now stale — drop it to avoid inconsistency
410        self.count_df = None
411
412        # ------------------------------------------------------------------ #
413        # 3. Update metadata and dims                                         #
414        # ------------------------------------------------------------------ #
415        self.feature_meta = self.feature_meta.iloc[keep_idx].reset_index(drop=True)
416        if self.feature_names is not None:
417            self.feature_names = [self.feature_names[i] for i in keep_idx]
418
419        self.dims = self._compute_dims()

421    def _validate(self):
422        """Validate data shapes and requirements."""
423        # Get ndim and shape (sparse matrices have these attributes too)
424        counts_ndim = self.counts.ndim if hasattr(self.counts, 'ndim') else len(self.counts.shape)
425        counts_shape = self.counts.shape
426
427        if self.distribution in ['negbinom', 'normal', 'binomial', 'studentt']:
428            if counts_ndim != 2:
429                raise ValueError(f"{self.distribution} modality requires 2D counts, got shape {counts_shape}")
430            n_features = counts_shape[1 - self.cells_axis]
431
432        elif self.distribution == 'multinomial':
433            if counts_ndim != 3:
434                raise ValueError(f"multinomial modality requires 3D counts (features, cells, categories), got shape {counts_shape}")
435            n_features = counts_shape[0]
436        
437        if self.feature_names is not None:
438            if len(self.feature_names) != n_features:
439                raise ValueError(
440                    f"feature_names has length {len(self.feature_names)} but counts has {n_features} features"
441                )
442        
443        # Validate feature_meta matches
444        if len(self.feature_meta) != n_features:
445            raise ValueError(f"feature_meta has {len(self.feature_meta)} rows but counts has {n_features} features")
446
447        # Validate denominator for binomial
448        if self.distribution == 'binomial':
449            if self.denominator is None:
450                raise ValueError("binomial distribution requires denominator")
451            if self.denominator.shape != self.counts.shape:
452                raise ValueError(f"denominator shape {self.denominator.shape} must match counts shape {self.counts.shape}")
453            counts_dense = self.counts.toarray() if sparse.issparse(self.counts) else self.counts
454            denom_dense = self.denominator.toarray() if sparse.issparse(self.denominator) else self.denominator
455            if np.any(counts_dense > denom_dense):
456                raise ValueError(
457                    f"[Modality '{self.name}'] counts exceed denominator for binomial distribution "
458                    f"(found {int(np.sum(counts_dense > denom_dense))} violating entries)"
459                )
460
461        # Validate exon skipping data if provided
462        if self.inc1 is not None or self.inc2 is not None or self.skip is not None:
463            if self.distribution != 'binomial':
464                raise ValueError("inc1/inc2/skip are only valid for binomial distribution (exon skipping)")
465            if self.inc1 is None or self.inc2 is None or self.skip is None:
466                raise ValueError("For exon skipping, must provide all three: inc1, inc2, skip")
467            if self.inc1.shape != self.counts.shape or self.inc2.shape != self.counts.shape or self.skip.shape != self.counts.shape:
468                raise ValueError(f"inc1, inc2, skip must all have same shape as counts: {self.counts.shape}")
469            if self.exon_aggregate_method not in ['min', 'mean']:
470                raise ValueError(f"exon_aggregate_method must be 'min' or 'mean', got: {self.exon_aggregate_method}")
471
472    def _compute_dims(self) -> Dict[str, int]:
473        """Compute dimensionality information."""
474        dims = {}
475
476        if self.distribution in ['negbinom', 'normal', 'binomial', 'studentt']:
477            dims['n_features'] = self.counts.shape[1 - self.cells_axis]
478            dims['n_cells'] = self.counts.shape[self.cells_axis]
479
480        elif self.distribution == 'multinomial':
481            dims['n_features'] = self.counts.shape[0]
482            dims['n_cells'] = self.counts.shape[1]
483            dims['n_categories'] = self.counts.shape[2]
484
485        return dims

487    def to_tensor(self, device: torch.device = torch.device('cpu')) -> torch.Tensor:
488        """Convert counts to torch tensor. Densifies if sparse."""
489        if self.is_sparse:
490            # Convert sparse to dense, then to tensor
491            counts_dense = self.counts.toarray()
492            return torch.tensor(counts_dense, dtype=torch.float32, device=device)
493        else:
494            return torch.tensor(self.counts, dtype=torch.float32, device=device)

496    def get_feature_subset(self, feature_indices: Union[list, np.ndarray]) -> 'Modality':
497        """
498        Return a new Modality with subset of features.
499
500        Parameters
501        ----------
502        feature_indices : list or array
503            Indices or names of features to keep
504
505        Returns
506        -------
507        Modality
508            New modality with subset of features
509        """
510        if isinstance(feature_indices, (list, np.ndarray)) and len(feature_indices) > 0:
511            if isinstance(feature_indices[0], str):
512                # Convert names to indices
513                if self.feature_names is None:
514                    raise ValueError("Cannot subset by name: feature_names not available")
515                feature_indices = [self.feature_names.index(n) for n in feature_indices]
516
517        # Subset counts (maintain sparsity if counts is sparse)
518        if self.distribution in ['negbinom', 'normal', 'binomial', 'studentt']:
519            if self.cells_axis == 1:
520                # Sparse-aware indexing
521                new_counts = self.counts[feature_indices, :]
522                new_denom = self.denominator[feature_indices, :] if self.denominator is not None else None
523                new_inc1 = self.inc1[feature_indices, :] if self.inc1 is not None else None
524                new_inc2 = self.inc2[feature_indices, :] if self.inc2 is not None else None
525                new_skip = self.skip[feature_indices, :] if self.skip is not None else None
526            else:
527                new_counts = self.counts[:, feature_indices]

625    def set_exon_aggregate_method(self, method: str, allow_after_technical_fit: bool = False, recover_filtered: bool = True):
626        """
627        Change the exon skipping aggregation method and recompute inclusion counts.
628
629        This recomputes self.counts (inclusion) and self.denominator (total) based on
630        the new aggregation method. If unfiltered data was stored during modality creation,
631        this will reprocess from that data and potentially recover features that were
632        filtered with the old method but pass filtering with the new method.
633
634        Parameters
635        ----------
636        method : str
637            Aggregation method: 'min' or 'mean'
638        allow_after_technical_fit : bool
639            If True, allow changing method even after technical fit. Default: False.
640            WARNING: Changing aggregation after technical fit invalidates the prefit
641            overdispersion parameters.
642        recover_filtered : bool
643            If True (default), reprocess from unfiltered data to potentially recover
644            features that were filtered with old method. If False or if unfiltered
645            data is not available, just recompute from current data.
646
647        Raises
648        ------
649        ValueError
650            If method is invalid, modality is not exon skipping, or changing after
651            technical fit without explicit permission.
652        """
653        if self.inc1 is None:
654            raise ValueError("This modality does not have exon skipping data (inc1/inc2/skip)")
655
656        if method not in ['min', 'mean']:
657            raise ValueError(f"method must be 'min' or 'mean', got: {method}")
658
659        # Check if technical fit was done with different method
660        if self._technical_fit_aggregate_method is not None:
661            if self._technical_fit_aggregate_method != method and not allow_after_technical_fit:
662                raise ValueError(
663                    f"Technical fit was performed with '{self._technical_fit_aggregate_method}' aggregation. "
664                    f"Cannot change to '{method}' without invalidating prefit parameters. "
665                    f"Set allow_after_technical_fit=True to override (not recommended)."
666                )
667
668        # Check if we can recover from unfiltered data
669        can_recover = (recover_filtered and
670                      self._unfiltered_inc1 is not None and
671                      self._unfiltered_inc2 is not None and
672                      self._unfiltered_skip is not None and
673                      self._unfiltered_feature_meta is not None)
674
675        if can_recover:
676            # Reprocess from unfiltered data with new method
677            print(f"[INFO] Reprocessing from unfiltered data with method='{method}'...")
678
679            # Apply binomial ratio filtering with new method
680            inc1_unfilt = self._unfiltered_inc1
681            inc2_unfilt = self._unfiltered_inc2
682            skip_unfilt = self._unfiltered_skip
683            meta_unfilt = self._unfiltered_feature_meta
684
685            n_events = inc1_unfilt.shape[0]
686
687            # Compute inclusion using new method
688            if method == 'min':
689                inc_for_filter = np.minimum(inc1_unfilt, inc2_unfilt)
690            elif method == 'mean':
691                inc_for_filter = (inc1_unfilt + inc2_unfilt) / 2.0
692
693            tot_for_filter = inc_for_filter + skip_unfilt
694
695            # Check ratio variance for each event
696            valid_events = []
697            n_zero_var = 0
698
699            for i in range(n_events):
700                numer = inc_for_filter[i, :]  # inclusion counts across cells
701                denom = tot_for_filter[i, :]   # total counts across cells
702
703                # Compute ratios, excluding cells where denominator is 0
704                valid_mask = denom > 0
705                if valid_mask.sum() == 0:
706                    n_zero_var += 1
707                    continue
708
709                ratios = numer[valid_mask] / denom[valid_mask]
710                if ratios.std() == 0:
711                    n_zero_var += 1
712                    continue
713
714                valid_events.append(i)
715
716            if n_zero_var > 0:
717                print(f"[INFO] Filtered {n_zero_var} exon skipping event(s) with zero variance in inclusion ratio (method='{method}')")
718
719            if len(valid_events) == 0:
720                raise ValueError(f"No exon skipping events with variable inclusion ratios after refiltering with method='{method}'!")
721
722            # Update to valid events
723            self.inc1 = inc1_unfilt[valid_events, :]
724            self.inc2 = inc2_unfilt[valid_events, :]
725            self.skip = skip_unfilt[valid_events, :]
726            self.feature_meta = meta_unfilt.iloc[valid_events].reset_index(drop=True)
727
728            # Recompute inclusion and total
729            if method == 'min':
730                self.counts = np.minimum(self.inc1, self.inc2)
731            elif method == 'mean':
732                self.counts = (self.inc1 + self.inc2) / 2.0
733
734            self.denominator = self.counts + self.skip
735            self.exon_aggregate_method = method
736
737            # Update dims to reflect new number of features
738            self.dims = self._compute_dims()
739
740            # Report recovery stats
741            n_before = inc1_unfilt.shape[0]
742            n_after = len(valid_events)
743            n_recovered = n_after - (n_before - n_zero_var)  # Difference from current filtered set
744            print(f"[INFO] Recomputed with method='{method}': {n_after}/{n_before} events retained")
745
746        else:
747            # Just recompute from current filtered data (no recovery)
748            if recover_filtered and self._unfiltered_inc1 is None:
749                print(f"[INFO] Unfiltered data not available - recomputing from current filtered data")
750
751            # Recompute inclusion
752            if method == 'min':
753                inclusion = np.minimum(self.inc1, self.inc2)
754            elif method == 'mean':
755                inclusion = (self.inc1 + self.inc2) / 2.0
756
757            # Recompute total
758            total = inclusion + self.skip
759
760            # Update
761            self.counts = inclusion
762            self.denominator = total
763            self.exon_aggregate_method = method
764
765            # Update dims (though feature count shouldn't change in this branch)
766            self.dims = self._compute_dims()
767
768            print(f"[INFO] Recomputed exon skipping inclusion with method='{method}' (no feature recovery)")

 62        Parameters
 63        ----------
 64        name : str
 65            Modality name
 66        counts : np.ndarray or pd.DataFrame
 67            Count/measurement data
 68        feature_meta : pd.DataFrame
 69            Feature metadata with index matching features in counts
 70        distribution : str
 71            Distribution type
 72        denominator : np.ndarray, optional
 73            For binomial: denominator counts (e.g., total gene expression for SJ usage)
 74        cells_axis : int
 75            Which axis represents cells (0 or 1 for 2D data)
 76        cell_names : list, optional
 77            Explicit cell names/identifiers (used when counts is ndarray, not DataFrame)
 78        inc1 : np.ndarray, optional
 79            For exon skipping: inclusion counts from first junction (d1->a2)
 80        inc2 : np.ndarray, optional
 81            For exon skipping: inclusion counts from second junction (d2->a3)
 82        skip : np.ndarray, optional
 83            For exon skipping: skipping counts (d1->a3)
 84        exon_aggregate_method : str, optional
 85            For exon skipping: how inc1 and inc2 were aggregated ('min' or 'mean')
 86        """
 87        if distribution not in self.VALID_DISTRIBUTIONS:
 88            raise ValueError(f"distribution must be one of {self.VALID_DISTRIBUTIONS}, got {distribution}")
 89
 90        self.name = name
 91        self.distribution = distribution
 92        self.cells_axis = cells_axis
 93
 94        # Handle different count input formats
 95        # Check if counts is sparse matrix
 96        self.is_sparse = sparse.issparse(counts)
 97
 98        if isinstance(counts, pd.DataFrame):
 99            # DataFrame: store both values and DataFrame
100            self.counts = counts.values
101            self.count_df = counts
102            self.is_sparse = False  # DataFrames are dense
103            # Feature names: use explicit parameter if provided, else from index/columns
104            if cells_axis == 1:
105                self.feature_names = feature_names if feature_names is not None else counts.index.tolist()
106                self.cell_names = counts.columns.tolist()
107            else:
108                self.feature_names = feature_names if feature_names is not None else counts.columns.tolist()
109                self.cell_names = counts.index.tolist()

111            # Sparse matrix: keep sparse, don't densify!
112            self.counts = counts
113            self.count_df = None
114            self.feature_names = feature_names if feature_names is not None else None
115            self.cell_names = cell_names if cell_names is not None else None
116            pass  # sparse matrix stored as-is
117        else:
118            # Dense array: convert to numpy array
119            self.counts = np.asarray(counts)
120            self.count_df = None
121            self.feature_names = feature_names if feature_names is not None else None
122            self.cell_names = cell_names if cell_names is not None else None
123
124        # If no explicit feature_names and no DataFrame index/columns to derive
125        # from (raw ndarray/sparse counts, e.g. add_custom_modality), fall back
126        # to feature_meta — same priority order used everywhere else in the
127        # codebase, so downstream code can always rely on modality.feature_names
128        # instead of re-deriving its own guess from feature_meta.
129        if self.feature_names is None:
130            self.feature_names = resolve_feature_names(feature_meta, context=f"Modality '{name}'")
131
132        # Disambiguate duplicate feature names (e.g. Ensembl gene_name collisions
133        # from pseudogenes/readthrough transcripts, or a many-to-one fallback
134        # column like 'gene' repeating once per SJ belonging to that gene).
135        # Name-based lookups throughout the codebase (feature_names.index(),
136        # pd.Series indexed by feature_names, get_feature_subset(), plotting by
137        # gene name, etc.) all assume uniqueness.
138        if self.feature_names is not None:
139            deduped = make_names_unique(self.feature_names)
140            if deduped != self.feature_names:
141                n_dup = sum(1 for a, b in zip(deduped, self.feature_names) if a != b)
142                warnings.warn(
143                    f"Modality '{name}' had {n_dup} duplicate feature name(s); "
144                    "disambiguated by appending -1, -2, ... to repeats (scanpy-style). "
145                    "Original identifiers are still available in feature_meta.",
146                    UserWarning
147                )
148            self.feature_names = deduped
149
150        self.feature_meta = feature_meta.copy()
151
152        # For multinomial: ensure the implicit Kth residual slot (position K_max-1) is
153        # always a REAL category, not a padded zero.
154        #
155        # Background: the K-1 residual parameterization computes
156        #   p_{K_max-1} = 1 - sum(p_0 .. p_{K_max-2})
157        # For features with K_actual < K_max the last slot is padded (zero observations).
158        # If the real categories together already account for all probability mass (~1),
159        # then p_{K_max-1} ≈ 0 or negative, and the Dirichlet prior on A inappropriately
160        # assigns mass to the phantom categories that compete with real ones.
161        #
162        # Fix: swap the last real category (column K_actual-1) into the residual slot
163        # (column K_max-1) for any feature where K_actual < K_max.  After reordering:
164        #   columns 0 .. K_actual-2  : real explicit Hills
165        #   columns K_actual-1 .. K_max-2 : phantom (padded, zero data)
166        #   column  K_max-1              : real residual category  ✓
167        #
168        # n_categories (= K_actual per feature) is computed before reordering and is
169        # unaffected by it (still counts non-zero columns, just at different positions).
170        if distribution == 'multinomial':
171            counts_arr = np.asarray(self.counts)  # safe: multinomial must be dense 3D

173                # Compute per-feature actual category count from non-zero data (before reordering).
174                # This always overwrites any user-provided n_categories to ensure it is
175                # consistent with the reordering performed below.  Zero-count categories
176                # (present in the tensor but with no reads in any cell) are treated as phantom.
177                n_cats = (counts_arr.sum(axis=1) > 0).sum(axis=1).astype(int)
178                self.feature_meta['n_categories'] = n_cats
179
180                # Reorder: swap last real category to residual position for any feature
181                # with K_actual < K_max.  Work on a copy so original input is untouched.
182                K_max = counts_arr.shape[2]
183                if K_max > 1 and np.any(n_cats < K_max):
184                    counts_arr = counts_arr.copy()
185                    for i in range(counts_arr.shape[0]):

187                        if 0 < K_actual < K_max:

189                            tmp = counts_arr[i, :, K_actual - 1].copy()
190                            counts_arr[i, :, K_actual - 1] = 0.0  # was real, now becomes padding
191                            counts_arr[i, :, K_max - 1] = tmp     # residual slot now real
192                    self.counts = counts_arr
193
194                # ---- Filter: drop multinomial features with ≤1 real category or
195                #              zero ratio-variance across cells ----
196                #
197                # Filter 1: Features with ≤1 real category cannot have multinomial
198                #           proportions (no structure to model).
199                # Filter 2: Features where all real-category ratios (count/cell_total)
200                #           have zero std across cells carry no fitting signal.
201                keep = n_cats > 1  # [T] bool, Filter 1
202
203                for i in range(len(n_cats)):

205                        continue
206                    feat = counts_arr[i]  # [n_cells, K_max]
207                    cell_totals = feat.sum(axis=1, keepdims=True)  # [n_cells, 1]
208                    real_cat_mask = feat.sum(axis=0) > 0  # [K_max]
209                    real_counts = feat[:, real_cat_mask]  # [n_cells, K_actual]
210                    with np.errstate(divide='ignore', invalid='ignore'):
211                        ratios = np.where(
212                            cell_totals > 0,
213                            real_counts / cell_totals,
214                            0.0,
215                        )  # [n_cells, K_actual]
216                    if np.all(ratios.std(axis=0) == 0):
217                        keep[i] = False  # Filter 2: zero ratio variance
218
219                n_dropped = int((~keep).sum())
220                if n_dropped > 0:
221                    n_le1_cat = int((n_cats <= 1).sum())
222                    n_zero_var = n_dropped - n_le1_cat
223                    warnings.warn(
224                        f"[Modality '{name}'] Dropped {n_dropped} multinomial feature(s): "
225                        f"{n_le1_cat} with ≤1 real category, "
226                        f"{n_zero_var} with zero category-ratio variance across cells.",
227                        UserWarning,
228                        stacklevel=2,
229                    )
230                    counts_arr = counts_arr[keep]
231                    if counts_arr.shape[0] == 0:
232                        raise ValueError(
233                            f"[Modality '{name}'] All multinomial features were filtered out "
234                            f"(≤1 real category or zero ratio variance). Check input data."
235                        )
236                    self.feature_meta = self.feature_meta[keep].reset_index(drop=True)
237                    self.feature_meta['n_categories'] = n_cats[keep]
238                    self.counts = counts_arr
239                    if self.feature_names is not None:
240                        self.feature_names = [
241                            fn for fn, k in zip(self.feature_names, keep) if k
242                        ]
243
244        # Handle denominator (can also be sparse)
245        if denominator is not None:
246            if sparse.issparse(denominator):
247                self.denominator = denominator  # Keep sparse
248            else:
249                self.denominator = np.asarray(denominator)
250        else:
251            self.denominator = None
252
253        # Exon skipping specific storage
254        self.inc1 = np.asarray(inc1) if inc1 is not None else None
255        self.inc2 = np.asarray(inc2) if inc2 is not None else None
256        self.skip = np.asarray(skip) if skip is not None else None
257        self.exon_aggregate_method = exon_aggregate_method
258        self._technical_fit_aggregate_method = None  # Track method used during technical fit
259
260        # Store unfiltered versions for recovery when switching methods
261        # (Only used for exon skipping modalities)
262        self._unfiltered_inc1 = None
263        self._unfiltered_inc2 = None
264        self._unfiltered_skip = None
265        self._unfiltered_feature_meta = None
266
267        # Per-modality fitting results storage
268        # Distribution-specific alpha_y parameters (use these directly, not alpha_y_prefit)
269        self.alpha_y_prefit_mult = None     # For negbinom: multiplicative correction [C, T]
270        self.alpha_y_prefit_add = None      # For normal/studentt/binomial/multinomial: additive correction [C, T]
271        self.posterior_samples_ntc = None  # NTC fit: full posterior samples
272        self.posterior_samples_trans = None      # Trans fit: full posterior samples
273
274        # Sum factors DataFrame (negbinom modalities only).
275        # Index: cell barcode. Columns: named sum factor variants
276        # (e.g. 'sum_factor', 'sum_factor_adj', 'sum_factor_new').
277        # Populated by bayesDREAM._init_sum_factors() after model init.
278        # The 'cis' modality holds a reference to the same object as the primary modality.

294    def _filter_zero_features(self) -> None:
295        """
296        Remove features with total counts below min_count or zero std across cells.
297
298        Called at the end of __init__, so it also fires automatically after any cell
299        subsetting (get_cell_subset returns a new Modality which calls __init__).
300        Modifies self in place; updates counts, feature_meta, feature_names, denominator,
301        inc1/inc2/skip, and dims.
302
303        Per-distribution logic
304        ----------------------
305        negbinom   : drop feature if total counts < min_count OR std across cells == 0
306        normal/
307        studentt   : drop feature if std across cells == 0
308                     (sum of values < min_count is also checked for consistency)
309        binomial   : drop feature if denominator sum < min_count (too few observations)
310                     OR ratio std (numer/denom) across cells == 0
311        multinomial: drop feature if total counts (summed over cells+categories) < min_count
312                     (zero-ratio-variance is already handled earlier in __init__)
313        """
314        min_count = self.min_count
315
316        # ------------------------------------------------------------------ #
317        # 1. Compute keep mask per distribution                               #
318        # ------------------------------------------------------------------ #
319        if self.distribution == 'multinomial':
320            counts_arr = np.asarray(self.counts)          # (T, C, K)
321            totals = counts_arr.sum(axis=(1, 2))          # (T,)
322            keep = totals >= min_count
323
324        elif self.distribution == 'binomial':
325            cell_ax = self.cells_axis
326
327            counts_num = (self.counts.toarray() if sparse.issparse(self.counts)
328                          else np.asarray(self.counts))
329            if self.denominator is not None:
330                denom = (self.denominator.toarray() if sparse.issparse(self.denominator)
331                         else np.asarray(self.denominator))
332            else:
333                denom = counts_num
334
335            denom_sums = denom.sum(axis=cell_ax)
336            has_data = denom_sums >= min_count
337
338            with np.errstate(divide='ignore', invalid='ignore'):
339                ratios = np.where(denom > 0, counts_num / denom, np.nan)
340            ratio_stds = np.nanstd(ratios, axis=cell_ax)
341            has_variance = ratio_stds > 0
342
343            keep = has_data & has_variance
344
345        else:  # negbinom, normal, studentt
346            cell_ax = self.cells_axis
347
348            if self.is_sparse:
349                _sp = self.counts.astype(float)
350                totals = np.array(_sp.sum(axis=cell_ax)).flatten()
351                _mean = np.array(_sp.mean(axis=cell_ax)).flatten()
352                _sp2 = _sp.copy()
353                _sp2.data **= 2
354                _sq_mean = np.array(_sp2.mean(axis=cell_ax)).flatten()
355                stds = np.sqrt(np.maximum(_sq_mean - _mean ** 2, 0))
356            else:
357                counts_arr = np.asarray(self.counts)
358                totals = counts_arr.sum(axis=cell_ax)
359                stds = counts_arr.std(axis=cell_ax)
360
361            keep = (totals >= min_count) & (stds > 0)
362
363        n_removed = int((~keep).sum())
364        if n_removed == 0:
365            return
366
367        keep_idx = np.where(keep)[0]
368        n_before = self.dims['n_features']
369
370        print(
371            f"[Modality '{self.name}'] Removed {n_removed}/{n_before} feature(s) "
372            f"with total counts < {min_count} or zero variance across cells."
373        )
374
375        # ------------------------------------------------------------------ #
376        # 2. Apply mask to counts and auxiliary arrays                        #
377        # ------------------------------------------------------------------ #
378        if self.distribution == 'multinomial':
379            self.counts = np.asarray(self.counts)[keep_idx, :, :]
380
381        elif self.cells_axis == 1:
382            if self.is_sparse:
383                self.counts = self.counts[keep_idx, :]
384            else:
385                self.counts = np.asarray(self.counts)[keep_idx, :]
386
387            if self.denominator is not None:
388                if sparse.issparse(self.denominator):
389                    self.denominator = self.denominator[keep_idx, :]
390                else:
391                    self.denominator = np.asarray(self.denominator)[keep_idx, :]
392
393            for _attr in ('inc1', 'inc2', 'skip'):
394                _val = getattr(self, _attr)
395                if _val is not None:
396                    setattr(self, _attr, _val[keep_idx, :])
397
398        else:  # cells_axis == 0
399            if self.is_sparse:
400                self.counts = self.counts[:, keep_idx]
401            else:
402                self.counts = np.asarray(self.counts)[:, keep_idx]
403
404            if self.denominator is not None:
405                if sparse.issparse(self.denominator):
406                    self.denominator = self.denominator[:, keep_idx]
407                else:
408                    self.denominator = np.asarray(self.denominator)[:, keep_idx]
409
410        # count_df is now stale — drop it to avoid inconsistency
411        self.count_df = None
412
413        # ------------------------------------------------------------------ #
414        # 3. Update metadata and dims                                         #
415        # ------------------------------------------------------------------ #
416        self.feature_meta = self.feature_meta.iloc[keep_idx].reset_index(drop=True)
417        if self.feature_names is not None:
418            self.feature_names = [self.feature_names[i] for i in keep_idx]
419
420        self.dims = self._compute_dims()

422    def _validate(self):
423        """Validate data shapes and requirements."""
424        # Get ndim and shape (sparse matrices have these attributes too)
425        counts_ndim = self.counts.ndim if hasattr(self.counts, 'ndim') else len(self.counts.shape)
426        counts_shape = self.counts.shape
427
428        if self.distribution in ['negbinom', 'normal', 'binomial', 'studentt']:
429            if counts_ndim != 2:
430                raise ValueError(f"{self.distribution} modality requires 2D counts, got shape {counts_shape}")
431            n_features = counts_shape[1 - self.cells_axis]
432
433        elif self.distribution == 'multinomial':
434            if counts_ndim != 3:
435                raise ValueError(f"multinomial modality requires 3D counts (features, cells, categories), got shape {counts_shape}")
436            n_features = counts_shape[0]
437        
438        if self.feature_names is not None:
439            if len(self.feature_names) != n_features:
440                raise ValueError(
441                    f"feature_names has length {len(self.feature_names)} but counts has {n_features} features"
442                )
443        
444        # Validate feature_meta matches
445        if len(self.feature_meta) != n_features:
446            raise ValueError(f"feature_meta has {len(self.feature_meta)} rows but counts has {n_features} features")
447
448        # Validate denominator for binomial
449        if self.distribution == 'binomial':
450            if self.denominator is None:
451                raise ValueError("binomial distribution requires denominator")
452            if self.denominator.shape != self.counts.shape:
453                raise ValueError(f"denominator shape {self.denominator.shape} must match counts shape {self.counts.shape}")
454            counts_dense = self.counts.toarray() if sparse.issparse(self.counts) else self.counts
455            denom_dense = self.denominator.toarray() if sparse.issparse(self.denominator) else self.denominator
456            if np.any(counts_dense > denom_dense):
457                raise ValueError(
458                    f"[Modality '{self.name}'] counts exceed denominator for binomial distribution "
459                    f"(found {int(np.sum(counts_dense > denom_dense))} violating entries)"
460                )
461
462        # Validate exon skipping data if provided
463        if self.inc1 is not None or self.inc2 is not None or self.skip is not None:
464            if self.distribution != 'binomial':
465                raise ValueError("inc1/inc2/skip are only valid for binomial distribution (exon skipping)")
466            if self.inc1 is None or self.inc2 is None or self.skip is None:
467                raise ValueError("For exon skipping, must provide all three: inc1, inc2, skip")
468            if self.inc1.shape != self.counts.shape or self.inc2.shape != self.counts.shape or self.skip.shape != self.counts.shape:
469                raise ValueError(f"inc1, inc2, skip must all have same shape as counts: {self.counts.shape}")
470            if self.exon_aggregate_method not in ['min', 'mean']:
471                raise ValueError(f"exon_aggregate_method must be 'min' or 'mean', got: {self.exon_aggregate_method}")

473    def _compute_dims(self) -> Dict[str, int]:

484            dims['n_categories'] = self.counts.shape[2]
485
486        return dims

626    def set_exon_aggregate_method(self, method: str, allow_after_technical_fit: bool = False, recover_filtered: bool = True):
627        """
628        Change the exon skipping aggregation method and recompute inclusion counts.
629
630        This recomputes self.counts (inclusion) and self.denominator (total) based on
631        the new aggregation method. If unfiltered data was stored during modality creation,
632        this will reprocess from that data and potentially recover features that were
633        filtered with the old method but pass filtering with the new method.
634
635        Parameters
636        ----------
637        method : str
638            Aggregation method: 'min' or 'mean'
639        allow_after_technical_fit : bool
640            If True, allow changing method even after technical fit. Default: False.
641            WARNING: Changing aggregation after technical fit invalidates the prefit
642            overdispersion parameters.
643        recover_filtered : bool
644            If True (default), reprocess from unfiltered data to potentially recover
645            features that were filtered with old method. If False or if unfiltered
646            data is not available, just recompute from current data.
647
648        Raises
649        ------
650        ValueError
651            If method is invalid, modality is not exon skipping, or changing after
652            technical fit without explicit permission.
653        """
654        if self.inc1 is None:
655            raise ValueError("This modality does not have exon skipping data (inc1/inc2/skip)")
656
657        if method not in ['min', 'mean']:
658            raise ValueError(f"method must be 'min' or 'mean', got: {method}")
659
660        # Check if technical fit was done with different method
661        if self._technical_fit_aggregate_method is not None:
662            if self._technical_fit_aggregate_method != method and not allow_after_technical_fit:
663                raise ValueError(
664                    f"Technical fit was performed with '{self._technical_fit_aggregate_method}' aggregation. "
665                    f"Cannot change to '{method}' without invalidating prefit parameters. "
666                    f"Set allow_after_technical_fit=True to override (not recommended)."
667                )
668
669        # Check if we can recover from unfiltered data
670        can_recover = (recover_filtered and
671                      self._unfiltered_inc1 is not None and
672                      self._unfiltered_inc2 is not None and
673                      self._unfiltered_skip is not None and
674                      self._unfiltered_feature_meta is not None)
675
676        if can_recover:
677            # Reprocess from unfiltered data with new method
678            print(f"[INFO] Reprocessing from unfiltered data with method='{method}'...")
679
680            # Apply binomial ratio filtering with new method
681            inc1_unfilt = self._unfiltered_inc1
682            inc2_unfilt = self._unfiltered_inc2
683            skip_unfilt = self._unfiltered_skip
684            meta_unfilt = self._unfiltered_feature_meta
685
686            n_events = inc1_unfilt.shape[0]
687
688            # Compute inclusion using new method
689            if method == 'min':
690                inc_for_filter = np.minimum(inc1_unfilt, inc2_unfilt)
691            elif method == 'mean':
692                inc_for_filter = (inc1_unfilt + inc2_unfilt) / 2.0
693
694            tot_for_filter = inc_for_filter + skip_unfilt
695
696            # Check ratio variance for each event
697            valid_events = []
698            n_zero_var = 0
699
700            for i in range(n_events):
701                numer = inc_for_filter[i, :]  # inclusion counts across cells
702                denom = tot_for_filter[i, :]   # total counts across cells
703
704                # Compute ratios, excluding cells where denominator is 0
705                valid_mask = denom > 0
706                if valid_mask.sum() == 0:
707                    n_zero_var += 1
708                    continue
709
710                ratios = numer[valid_mask] / denom[valid_mask]
711                if ratios.std() == 0:
712                    n_zero_var += 1
713                    continue
714
715                valid_events.append(i)
716
717            if n_zero_var > 0:
718                print(f"[INFO] Filtered {n_zero_var} exon skipping event(s) with zero variance in inclusion ratio (method='{method}')")
719
720            if len(valid_events) == 0:
721                raise ValueError(f"No exon skipping events with variable inclusion ratios after refiltering with method='{method}'!")
722
723            # Update to valid events
724            self.inc1 = inc1_unfilt[valid_events, :]
725            self.inc2 = inc2_unfilt[valid_events, :]
726            self.skip = skip_unfilt[valid_events, :]
727            self.feature_meta = meta_unfilt.iloc[valid_events].reset_index(drop=True)
728
729            # Recompute inclusion and total
730            if method == 'min':
731                self.counts = np.minimum(self.inc1, self.inc2)
732            elif method == 'mean':
733                self.counts = (self.inc1 + self.inc2) / 2.0
734
735            self.denominator = self.counts + self.skip
736            self.exon_aggregate_method = method
737
738            # Update dims to reflect new number of features
739            self.dims = self._compute_dims()
740
741            # Report recovery stats
742            n_before = inc1_unfilt.shape[0]
743            n_after = len(valid_events)
744            n_recovered = n_after - (n_before - n_zero_var)  # Difference from current filtered set
745            print(f"[INFO] Recomputed with method='{method}': {n_after}/{n_before} events retained")
746
747        else:
748            # Just recompute from current filtered data (no recovery)
749            if recover_filtered and self._unfiltered_inc1 is None:
750                print(f"[INFO] Unfiltered data not available - recomputing from current filtered data")
751
752            # Recompute inclusion
753            if method == 'min':
754                inclusion = np.minimum(self.inc1, self.inc2)
755            elif method == 'mean':
756                inclusion = (self.inc1 + self.inc2) / 2.0
757
758            # Recompute total
759            total = inclusion + self.skip
760
761            # Update
762            self.counts = inclusion
763            self.denominator = total
764            self.exon_aggregate_method = method
765
766            # Update dims (though feature count shouldn't change in this branch)
767            self.dims = self._compute_dims()
768
769            print(f"[INFO] Recomputed exon skipping inclusion with method='{method}' (no feature recovery)")

176                # (present in the tensor but with no reads in any cell) are treated as phantom.

213                            real_counts / cell_totals,
214                            0.0,
215                        )  # [n_cells, K_actual]

218

234                            f"(≤1 real category or zero ratio variance). Check input data."
235                        )

237                    self.feature_meta['n_categories'] = n_cats[keep]
238                    self.counts = counts_arr
239                    if self.feature_names is not None:

241                            fn for fn, k in zip(self.feature_names, keep) if k
